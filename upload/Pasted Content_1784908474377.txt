// components/hero/EnsoCircle.tsx
'use client';

import { motion } from 'framer-motion';

interface EnsoCircleProps {
  size: number;
  strokeWidth: number;
  color: string;
  imperfection: number; // 0-1, quanto mais, mais imperfeito
  className?: string;
}

export const EnsoCircle: React.FC<EnsoCircleProps> = ({
  size,
  strokeWidth,
  color,
  imperfection = 0.1,
  className,
}) => {
  // Gera pontos de controle imperfeitos para o círculo
  const generateImperfectPath = (radius: number, imperfection: number) => {
    const points = 8;
    let path = '';
    
    for (let i = 0; i <= points; i++) {
      const angle = (i / points) * Math.PI * 2;
      const imperfectRadius = radius + (Math.random() - 0.5) * radius * imperfection * 2;
      const x = Math.cos(angle) * imperfectRadius;
      const y = Math.sin(angle) * imperfectRadius;
      
      if (i === 0) {
        path += `M ${x} ${y} `;
      } else {
        const cpRadius = imperfectRadius * 1.2;
        const cpAngle = angle - (Math.PI / points);
        const cpX = Math.cos(cpAngle) * cpRadius;
        const cpY = Math.sin(cpAngle) * cpRadius;
        path += `Q ${cpX} ${cpY} ${x} ${y} `;
      }
    }
    
    path += 'Z';
    return path;
  };

  const radius = size / 2;
  const path = generateImperfectPath(radius, imperfection);

  return (
    <motion.svg
      width={size}
      height={size}
      viewBox={`-${size/2} -${size/2} ${size} ${size}`}
      className={className}
      initial={{ pathLength: 0, opacity: 0 }}
      animate={{ pathLength: 1, opacity: 1 }}
      transition={{ duration: 2, ease: "easeInOut" }}
    >
      {/* Círculo principal */}
      <motion.path
        d={path}
        stroke={color}
        strokeWidth={strokeWidth}
        fill="none"
        strokeLinecap="round"
        strokeLinejoin="round"
        initial={{ pathLength: 0 }}
        animate={{ pathLength: 1 }}
        transition={{ duration: 2, delay: 0.5 }}
      />
      
      {/* Pincelada adicional (como tinta escorrendo) */}
      <motion.path
        d={path}
        stroke={color}
        strokeWidth={strokeWidth * 0.5}
        fill="none"
        strokeLinecap="round"
        opacity={0.3}
        initial={{ pathLength: 0 }}
        animate={{ pathLength: 1 }}
        transition={{ duration: 1.5, delay: 0.8 }}
        transform="translate(2, 2)"
      />
      
      {/* Brilho interno */}
      <motion.circle
        cx={0}
        cy={0}
        r={radius * 0.95}
        fill="none"
        stroke={`url(#glowGradient)`}
        strokeWidth={1}
        initial={{ opacity: 0 }}
        animate={{ opacity: 0.5 }}
        transition={{ duration: 1, delay: 2 }}
      />
      
      <defs>
        <radialGradient id="glowGradient">
          <stop offset="0%" stopColor={color} stopOpacity="0.1" />
          <stop offset="100%" stopColor={color} stopOpacity="0" />
        </radialGradient>
      </defs>
    </motion.svg>
  );
};
