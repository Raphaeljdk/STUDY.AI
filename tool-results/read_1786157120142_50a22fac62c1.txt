   385→function formatReviewDate(dateStr: string): string {
   386→  const d = new Date(dateStr);
   387→  const now = new Date();
   388→  const diffMs = d.getTime() - now.getTime();
   389→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   390→
   391→  if (diffDays < 0) return 'Atrasado';
   392→  if (diffDays === 0) return 'Hoje';
   393→  if (diffDays === 1) return 'Amanha';
   394→  if (diffDays <= 7) return `em ${diffDays} dias`;
   395→  return d.toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit' });
   396→}

   621→// ========== NOTEBOOKS LIST ==========
   622→function NotebooksList({ onOpen }: { onOpen: (id: string) => void }) {
   623→  const [notebooks, setNotebooks] = useState<NotebookItem[]>([]);
   624→  const [loading, setLoading] = useState(true);
   625→  const [creating, setCreating] = useState(false);
   626→  const [newTitle, setNewTitle] = useState('');
   627→  const [newColor, setNewColor] = useState(notebookColors[0]);
   628→  const fetchedRef = useRef(false);
   629→
   630→  useEffect(() => {
   631→    if (fetchedRef.current) return;
   632→    fetchedRef.current = true;
   633→    fetch('/api/notebooks').then(r => r.json()).then(d => {
   634→      if (d.notebooks) setNotebooks(d.notebooks);
   635→    }).catch(() => {}).finally(() => setLoading(false));
   636→  }, []);
   637→
   638→  const handleCreate = async (e: React.FormEvent) => {
   639→    e.preventDefault();
   640→    if (!newTitle.trim()) return;
   641→    setCreating(true);
   642→    try {
   643→      const res = await fetch('/api/notebooks', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ title: newTitle, color: newColor }) });
   644→      const data = await res.json();
   645→      if (!res.ok) {
   646→        toast({ title: 'Erro ao criar caderno', description: data.error || 'Tente novamente.', variant: 'destructive' });
   647→        setCreating(false);
   648→        return;
   649→      }
   650→      if (data.notebook) {
   651→        setNotebooks(prev => [data.notebook, ...prev]);
   652→        setNewTitle('');
   653→        setCreating(false);
   654→        onOpen(data.notebook.id);
   655→        toast({ title: 'Caderno criado!', description: `"${data.notebook.title}" esta pronto.` });
   656→      }
   657→    } catch (err) {
   658→      console.error('[NotebooksList] create error:', err);
   659→      toast({ title: 'Erro de conexao', description: 'Nao foi possivel criar o caderno. Verifique sua conexao.', variant: 'destructive' });
   660→    } 
   661→    setCreating(false);
   662→  };
   663→
   664→  const handleDelete = async (id: string, title: string) => {
   665→    if (!confirm(`Deletar "${title}"?`)) return;
   666→    try {
   667→      const res = await fetch(`/api/notebooks/${id}`, { method: 'DELETE' });
   668→      if (res.ok) {
   669→        setNotebooks(prev => prev.filter(n => n.id !== id));
   670→        toast({ title: 'Caderno deletado', description: `"${title}" foi removido.` });
   671→      } else {
   672→        const d = await res.json();
   673→        toast({ title: 'Erro ao deletar', description: d.error || 'Tente novamente.', variant: 'destructive' });
   674→      }
   675→    } catch (err) {
   676→      console.error('[NotebooksList] delete error:', err);

   977→function FlashcardReviewer({ onBack }: { onBack: () => void }) {
   978→  const [cards, setCards] = useState<FlashcardItem[]>([]);
   979→  const [loading, setLoading] = useState(true);
   980→  const [currentIndex, setCurrentIndex] = useState(0);
   981→  const [flipped, setFlipped] = useState(false);
   982→  const [done, setDone] = useState(false);
   983→  const [reviewed, setReviewed] = useState(0);
   984→  const [ratings, setRatings] = useState<number[]>([]);
   985→  const fetchedRef = useRef(false);
   986→
   987→  useEffect(() => {
   988→    if (fetchedRef.current) return;
   989→    fetchedRef.current = true;
   990→    fetch('/api/flashcards?due=true').then(r => r.json()).then(d => {
   991→      if (d.flashcards) setCards(d.flashcards);
   992→    }).catch(() => {}).finally(() => setLoading(false));
   993→  }, []);
   994→
   995→  const handleReview = useCallback(async (quality: number) => {
   996→    const card = cards[currentIndex];
   997→    if (!card) return;
   998→    try {
   999→      await fetch(`/api/flashcards/${card.id}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ review: quality }) });
  1000→    } catch {}
  1001→    setReviewed(prev => prev + 1);
  1002→    setRatings(prev => [...prev, quality]);
  1003→    setFlipped(false);
  1004→    if (currentIndex + 1 >= cards.length) setDone(true);
  1005→    else setCurrentIndex(prev => prev + 1);
  1006→  }, [cards, currentIndex]);
  1007→
  1008→  // Keyboard shortcuts
  1009→  useEffect(() => {
  1010→    if (done || !flipped || cards.length === 0) return;
  1011→    const handleKey = (e: KeyboardEvent) => {
  1012→      if (e.key === '1') handleReview(1);
  1013→      else if (e.key === '2') handleReview(2);
  1014→      else if (e.key === '3') handleReview(3);
  1015→      else if (e.key === '4') handleReview(5);
  1016→    };
  1017→    window.addEventListener('keydown', handleKey);
  1018→    return () => window.removeEventListener('keydown', handleKey);
  1019→  }, [flipped, done, cards.length, handleReview]);
  1020→
  1021→  // Completion stats
  1022→  const avgRating = ratings.length > 0 ? (ratings.reduce((a, b) => a + b, 0) / ratings.length).toFixed(1) : '0';
  1023→  const hardCount = ratings.filter(r => r <= 2).length;
  1024→  const easyCount = ratings.filter(r => r >= 4).length;
  1025→
  1026→  if (loading) return <div className="flex items-center justify-center py-16"><Loader2 size={24} className="animate-spin text-[var(--ws-text-tertiary)]" /></div>;
  1027→  if (cards.length === 0) return (
  1028→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="py-16 text-center">
  1029→      <Check size={48} className="mx-auto mb-4 text-[var(--ws-verdigris)]" strokeWidth={1} />
  1030→      <p className="text-sm text-[var(--ws-text-tertiary)]">Nenhuma carta para revisar agora</p>
  1031→      <button onClick={onBack} className="mt-4 text-sm text-[var(--ws-accent)] hover:underline">Voltar</button>
  1032→    </motion.div>
  1033→  );

  1128→function PomodoroTimer() {
  1129→  const [mode, setMode] = useState<'focus' | 'short' | 'long'>('focus');
  1130→  const [timeLeft, setTimeLeft] = useState(25 * 60);
  1131→  const [isRunning, setIsRunning] = useState(false);
  1132→  const [totalToday, setTotalToday] = useState(0);
  1133→  const [sessions, setSessions] = useState<{ duration: number; createdAt: string }[]>([]);
  1134→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
  1135→  const durations = { focus: 25 * 60, short: 5 * 60, long: 15 * 60 };
  1136→  const modeLabels = { focus: 'Foco', short: 'Pausa Curta', long: 'Pausa Longa' };
  1137→  const fetchedRef = useRef(false);
  1138→
  1139→  useEffect(() => {
  1140→    if (fetchedRef.current) return;
  1141→    fetchedRef.current = true;
  1142→    fetch('/api/sessions').then(r => r.json()).then(d => {
  1143→      if (d.sessions) {
  1144→        const today = new Date().toDateString();
  1145→        setTotalToday(d.sessions.filter((s: any) => new Date(s.createdAt).toDateString() === today).reduce((sum: number, s: any) => sum + s.duration, 0));
  1146→        setSessions(d.sessions.slice(0, 10));
  1147→      }
  1148→    });
  1149→  }, []);
  1150→
  1151→  useEffect(() => {
  1152→    if (isRunning) {
  1153→      intervalRef.current = setInterval(() => {
  1154→        setTimeLeft(prev => {
  1155→          if (prev <= 1) {
  1156→            setIsRunning(false);
  1157→            if (mode === 'focus') {
  1158→              const dur = durations.focus - prev;
  1159→              fetch('/api/sessions', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ duration: dur, type: 'pomodoro' }) });
  1160→              setTotalToday(t => t + dur);
  1161→            }
  1162→            return 0;
  1163→          }
  1164→          return prev - 1;
  1165→        });
  1166→      }, 1000);
  1167→    } else if (intervalRef.current) clearInterval(intervalRef.current);
  1168→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
  1169→  }, [isRunning, mode, durations.focus]);
  1170→
  1171→  const switchMode = (m: 'focus' | 'short' | 'long') => { setMode(m); setTimeLeft(durations[m]); setIsRunning(false); };
  1172→  const minutes = Math.floor(timeLeft / 60);
  1173→  const seconds = timeLeft % 60;
  1174→  const progress = 1 - timeLeft / durations[mode];
  1175→
  1176→  return (
  1177→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
  1178→      <div className="mb-8">
  1179→        <h1 className="font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)] lg:text-3xl"><Timer size={24} className="mr-2 inline text-[var(--ws-accent)]" strokeWidth={1.5} />Pomodoro Timer</h1>
  1180→        <p className="mt-1 text-sm text-[var(--ws-text-tertiary)]">Estude com foco usando a tecnica Pomodoro</p>
  1181→      </div>
  1182→      <div className="mx-auto max-w-md">
  1183→        <div className="mb-8 flex justify-center gap-2">
  1184→          {(['focus', 'short', 'long'] as const).map(m => (
  1185→            <button key={m} onClick={() => switchMode(m)} className={`rounded-ws-button px-4 py-2 text-sm font-medium transition-colors ${mode === m ? 'bg-[var(--ws-accent)] text-[var(--ws-text-on-dark)]' : 'border border-[var(--ws-glass-border)] text-[var(--ws-text-tertiary)] hover:text-[var(--ws-text-secondary)]'}`}>
  1186→              {modeLabels[m]}
  1187→            </button>
  1188→          ))}
  1189→        </div>
  1190→        <WabiSabiCard className="mb-8" hover={false}>
  1191→          <div className="flex flex-col items-center py-8">
  1192→            <div className="relative flex h-56 w-56 items-center justify-center">
  1193→              <svg className="absolute inset-0 -rotate-90" viewBox="0 0 200 200">
  1194→                <circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-glass-border)" strokeWidth="4" />
  1195→                <motion.circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-accent)" strokeWidth="4" strokeLinecap="round" strokeDasharray={2 * Math.PI * 90} initial={{ strokeDashoffset: 2 * Math.PI * 90 }} animate={{ strokeDashoffset: 2 * Math.PI * 90 * (1 - progress) }} transition={{ duration: 0.5 }} />
  1196→              </svg>
  1197→              <div className="text-center">
  1198→                <p className="font-serif-jp text-5xl font-bold text-[var(--ws-text-primary)]">{String(minutes).padStart(2, '0')}:{String(seconds).padStart(2, '0')}</p>

  1208→        <WabiSabiCard hover={false}>
