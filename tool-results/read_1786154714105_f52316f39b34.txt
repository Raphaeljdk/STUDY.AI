   171→// ===== MAIN COMPONENT =====
   172→export function DiscoverView({ onNavigate }: DiscoverViewProps) {
   173→  const [items, setItems] = useState<DiscoverItem[]>([]);
   174→  const [loading, setLoading] = useState(true);
   175→  const [activeFilter, setActiveFilter] = useState('todos');
   176→  const [searchQuery, setSearchQuery] = useState('');
   177→  const [expandedId, setExpandedId] = useState<string | null>(null);
   178→  const [savingId, setSavingId] = useState<string | null>(null);
   179→  const [page, setPage] = useState(1);
   180→  const [totalPages, setTotalPages] = useState(1);
   181→  const [subjects, setSubjects] = useState<Subject[]>([]);
   182→
   183→  const [showAIDialog, setShowAIDialog] = useState(false);
   184→  const [aiType, setAiType] = useState('dica');
   185→  const [aiSubject, setAiSubject] = useState('');
   186→  const [aiDifficulty, setAiDifficulty] = useState('medio');
   187→  const [aiGenerating, setAiGenerating] = useState(false);
   188→
   189→  const [showSubjectDialog, setShowSubjectDialog] = useState(false);
   190→  const [subjectSelectItemId, setSubjectSelectItemId] = useState<string | null>(null);
   191→  const [selectedSubjectId, setSelectedSubjectId] = useState<string>('');
   192→
   193→  // ===== FETCH =====
   194→  const fetchItems = useCallback(async (pageNum: number, type?: string) => {
   195→    try {
   196→      const params = new URLSearchParams({ page: String(pageNum), limit: '20' });
   197→      if (type && type !== 'todos') params.set('type', type);
   198→      const res = await fetch(`/api/discover?${params.toString()}`);
   199→      if (!res.ok) throw new Error('Failed');
   200→      const data = await res.json();
   201→      if (pageNum === 1) {
   202→        setItems(data.items || []);
   203→      } else {
   204→        setItems(prev => [...prev, ...(data.items || [])]);
   205→      }
   206→      setTotalPages(data.pagination?.totalPages || 1);
   207→    } catch {
   208→      toast({ title: 'Erro ao carregar conteudo', description: 'Tente novamente mais tarde.' });
   209→    }
   210→  }, []);
   211→
   212→  const fetchSubjects = useCallback(async () => {
   213→    try {
   214→      const res = await fetch('/api/subjects');
   215→      if (!res.ok) return;
   216→      const data = await res.json();
   217→      setSubjects(Array.isArray(data) ? data : data.subjects || []);
   218→    } catch {
   219→      // silent
   220→    }
   221→  }, []);
   222→
   223→  useEffect(() => {
   224→ const load = async () => {
   225→      setLoading(true);
   226→      setPage(1);
   227→      await fetchItems(1, activeFilter);
   228→      setLoading(false);
   229→    };
   230→    load();
   231→  }, [activeFilter, fetchItems]);
   232→
   233→  useEffect(() => {
   234→    fetchSubjects();
   235→  }, [fetchSubjects]);
   236→
   237→  const loadMore = useCallback(async () => {
   238→    if (loading || page >= totalPages) return;
   239→    const next = page + 1;
   240→    setPage(next);
   241→    await fetchItems(next, activeFilter);
   242→  }, [loading, page, totalPages, activeFilter, fetchItems]);
   243→
   244→  // ===== FILTERED ITEMS =====
   245→  const filteredItems = useMemo(() => {
   246→    if (!searchQuery.trim()) return items;
   247→    const q = searchQuery.toLowerCase();
   248→    return items.filter(
   249→      item =>
   250→        item.title.toLowerCase().includes(q) ||
   251→        item.content.toLowerCase().includes(q) ||
   252→        item.summary?.toLowerCase().includes(q) ||
   253→        item.subject?.toLowerCase().includes(q) ||
   254→        item.tags?.toLowerCase().includes(q)
   255→    );
   256→  }, [items, searchQuery]);
   257→
   258→  // ===== SAVE =====
   259→  const handleSave = async (e: React.MouseEvent, itemId: string) => {
   260→    e.stopPropagation();
   261→    setSavingId(itemId);
   262→    try {
   263→      const res = await fetch(`/api/discover/${itemId}/save`, { method: 'POST' });
   264→      if (!res.ok) throw new Error('Failed');
   265→      const data = await res.json();
   266→      setItems(prev =>
   267→        prev.map(item =>
   268→          item.id === itemId
   269→            ? { ...item, isSaved: data.isSaved, saves: data.isSaved ? item.saves + 1 : Math.max(0, item.saves - 1) }
   270→            : item
   271→        )
   272→      );
   273→      toast({
   274→        title: data.isSaved ? 'Salvo!' : 'Removido dos salvos',
   275→        description: data.isSaved ? 'Conteudo adicionado aos seus salvos.' : 'Conteudo removido dos salvos.',
   276→      });
   277→    } catch {
   278→      toast({ title: 'Erro', description: 'Nao foi possível salvar o conteudo.' });
   279→    } finally {
   280→      setSavingId(null);
   281→    }
   282→  };
   283→
   284→  // ===== AI GENERATE =====
   285→  const handleAIGenerate = async () => {
   286→    if (!aiSubject.trim()) {
   287→      toast({ title: 'Informe o assunto', description: 'Digite um assunto para gerar conteudo.' });
   288→      return;
   289→    }
   290→    setAiGenerating(true);
   291→    try {
   292→      const res = await fetch('/api/discover', {
   293→        method: 'POST',
   294→        headers: { 'Content-Type': 'application/json' },
   295→        body: JSON.stringify({ type: aiType, subject: aiSubject.trim(), difficulty: aiDifficulty, generateWithAI: true }),
   296→      });
   297→      if (!res.ok) {
   298→        const err = await res.json().catch(() => ({}));
   299→        throw new Error(err.error || 'Failed');
   300→      }
   301→      const data = await res.json();
   302→      if (data.item) {
   303→        setItems(prev => [data.item, ...prev]);
   304→        toast({ title: 'Conteudo gerado!', description: data.item.title });
   305→      }
   306→      setShowAIDialog(false);
   307→      setAiSubject('');
   308→    } catch (err: any) {
   309→      toast({ title: 'Erro ao gerar', description: err.message || 'Tente novamente.' });
   310→    } finally {
   311→      setAiGenerating(false);
   312→    }
   313→  };
   314→
   315→  // ===== ADD TO SUBJECT =====
   316→  const openSubjectSelector = (e: React.MouseEvent, itemId: string) => {
   317→    e.stopPropagation();
   318→    setSubjectSelectItemId(itemId);
   319→    setSelectedSubjectId('');
   320→    setShowSubjectDialog(true);
   321→  };
   322→
   323→  const handleAddToSubject = async () => {
   324→    if (!selectedSubjectId || !subjectSelectItemId) return;
   325→    try {
   326→      const item = items.find(i => i.id === subjectSelectItemId);
   327→      if (!item) return;
   328→      const res = await fetch('/api/notes', {
   329→        method: 'POST',
   330→        headers: { 'Content-Type': 'application/json' },
   331→        body: JSON.stringify({ title: `[Discover] ${item.title}`, content: item.content, subjectId: selectedSubjectId }),
   332→      });
   333→      if (res.ok) {
   334→        toast({ title: 'Adicionado a materia!', description: 'Conteudo salvo na materia selecionada.' });
   335→        setShowSubjectDialog(false);
   336→      } else {
   337→        throw new Error('Failed');
   338→      }
   339→    } catch {
   340→      toast({ title: 'Erro', description: 'Nao foi possivel adicionar a materia.' });
   341→    }
   342→  };
   343→
   344→  const handleGenerateFlashcards = (e: React.MouseEvent) => {
   345→    e.stopPropagation();
   346→    toast({ title: 'Flashcards em breve!', description: 'Funcionalidade sera liberada em breve.' });
   347→  };
   348→
   349→  const handleGenerateQuestions = (e: React.MouseEvent) => {
   350→    e.stopPropagation();
   351→    toast({ title: 'Questoes em breve!', description: 'Funcionalidade sera liberada em breve.' });
   352→  };
   353→
   354→  // ===== HELPERS =====
   355→  const getTypeConfig = (type: string) => CARD_TYPES[type] || CARD_TYPES.dica;
   356→  const getDifficultyConfig = (d: string) => DIFFICULTY_CONFIG[d] || DIFFICULTY_CONFIG.medio;
   357→  const formatDuration = (seconds: number | null) => {
   358→    if (!seconds) return null;
   359→    if (seconds < 60) return `${seconds}s`;
   360→    const m = Math.floor(seconds / 60);
   361→    const s = seconds % 60;
   362→    return s > 0 ? `${m}m ${s}s` : `${m}min`;
   363→  };
   364→
   365→  const hasActiveFilter = activeFilter !== 'todos' || !!searchQuery.trim();
   366→
   367→  return (
   368→    <div className="relative min-h-full pb-24">
   369→      {/* Header */}
   370→      <div className="mb-6">
   371→        <h1 className="font-serif-jp text-2xl font-bold mb-1" style={{ color: 'var(--ws-text-primary)' }}>
   372→          Discover
   373→        </h1>
   374→        <p className="text-sm" style={{ color: 'var(--ws-text-tertiary)' }}>
   375→          Explore conteudo educacional gerado pela comunidade e IA
   376→        </p>
   377→      </div>
   378→
   379→      {/* Search */}
   380→      <div className="relative mb-4">
   381→        <Search
   382→          className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2"
   383→          style={{ color: 'var(--ws-text-tertiary)' }}
   384→        />
   385→        <Input
   386→          placeholder="Buscar conteudo..."
   387→          value={searchQuery}
   388→          onChange={e => setSearchQuery(e.target.value)}
   389→          className="pl-10 pr-10 h-11 w-full rounded-ws-button border-[var(--ws-glass-border)] bg-[var(--ws-glass)] backdrop-blur-xl text-[var(--ws-text-primary)] placeholder:text-[var(--ws-text-tertiary)] focus-visible:ring-[var(--ws-accent)]/30"
   390→        />
   391→        {searchQuery && (
   392→          <button
   393→            onClick={() => setSearchQuery('')}
   394→            className="absolute right-3 top-1/2 -translate-y-1/2 rounded-full p-0.5 transition-ws"
   395→            style={{ color: 'var(--ws-text-tertiary)' }}
   396→            aria-label="Limpar busca"
   397→          >
   398→            <X className="h-4 w-4" />
   399→          </button>
   400→        )}
   401→      </div>
   402→
   403→      {/* Filter Bar */}
   404→      <div className="mb-6 -mx-4 px-4 overflow-x-auto no-scrollbar">
   405→        <div className="flex gap-2 pb-1" style={{ minWidth: 'max-content' }}>
   406→          {FILTER_OPTIONS.map(opt => {
   407→            const isActive = activeFilter === opt.key;
   408→            const typeConf = opt.key !== 'todos' ? CARD_TYPES[opt.key] : null;
   409→            return (
   410→              <button
   411→                key={opt.key}
   412→                onClick={() => { setActiveFilter(opt.key); setPage(1); }}
   413→                className="flex items-center gap-1.5 whitespace-nowrap rounded-ws-button px-3.5 py-2 text-xs font-medium transition-ws"
   414→                style={{
   415→                  background: isActive
   416→                    ? typeConf ? typeConf.accentBg : 'color-mix(in srgb, var(--ws-accent) 10%, transparent)'
   417→                    : 'var(--ws-glass)',
   418→                  color: isActive
   419→                    ? typeConf ? typeConf.accent : 'var(--ws-accent)'
   420→                    : 'var(--ws-text-secondary)',
   421→                  border: `1px solid ${isActive
   422→                    ? typeConf ? typeConf.accentBorder : 'color-mix(in srgb, var(--ws-accent) 25%, transparent)'
   423→                    : 'var(--ws-glass-border)'}`,
   424→                  boxShadow: isActive ? 'var(--ws-shadow-soft)' : 'none',
   425→                }}
   426→                aria-pressed={isActive}
   427→              >
   428→                <span>{opt.emoji}</span>
   429→                <span>{opt.label}</span>
   430→              </button>
   431→            );
   432→          })}
   433→        </div>
   434→      </div>
   435→
   436→      {/* Feed */}
   437→      {loading && items.length === 0 ? (
   438→        <div className="space-y-4">
   439→          {Array.from({ length: 4 }).map((_, i) => (
   440→            <SkeletonCard key={i} />
   441→          ))}
   442→        </div>
   443→      ) : filteredItems.length === 0 ? (
   444→        <EmptyState hasFilter={hasActiveFilter} onGenerateClick={() => setShowAIDialog(true)} />
   445→      ) : (
   446→        <div className="space-y-4">
   447→          <AnimatePresence mode="popLayout">
   448→            {filteredItems.map((item, index) => {
   449→              const typeConf = getTypeConfig(item.type);
   450→              const diffConf = getDifficultyConfig(item.difficulty);
   451→              const isExpanded = expandedId === item.id;
   452→              const duration = formatDuration(item.duration);
   453→
   454→              return (
   455→                <motion.div
   456→                  key={item.id}
   457→                  layout
   458→                  initial={{ opacity: 0, y: 16 }}
   459→                  animate={{ opacity: 1, y: 0 }}
   460→                  exit={{ opacity: 0, y: -10 }}
   461→                  transition={{ duration: 0.35, delay: Math.min(index * 0.04, 0.2), ease: [0.4, 0, 0.2, 1] }}
   462→                  className="hover-lift cursor-pointer"
   463→                  style={{ borderRadius: 'var(--ws-radius-card)' }}
   464→                  onClick={() => setExpandedId(isExpanded ? null : item.id)}
   465→                >
   466→                  <div
   467→                    className="border backdrop-blur-xl p-5 transition-ws"
   468→                    style={{
   469→                      borderRadius: 'var(--ws-radius-card)',
   470→                      borderColor: isExpanded ? typeConf.accentBorder : 'var(--ws-glass-border)',
   471→                      background: isExpanded
   472→                        ? `color-mix(in srgb, ${typeConf.accent} 3%, var(--ws-glass))`
   473→                        : 'var(--ws-glass)',
   474→                      boxShadow: isExpanded
   475→                        ? `0 8px 40px color-mix(in srgb, ${typeConf.accent} 8%, transparent)`
   476→                        : 'var(--ws-shadow-soft)',
   477→                    }}
   478→                  >
   479→                    {/* Header */}
   480→                    <div className="flex items-start gap-3 mb-3">
   481→                      <div
   482→                        className="flex h-10 w-10 flex-shrink-0 items-center justify-center text-lg"
   483→                        style={{ background: typeConf.accentBg, borderRadius: '12px 11px 13px 10px' }}
   484→                      >
   485→                        {item.emoji || typeConf.emoji}
   486→                      </div>
   487→                      <div className="flex-1 min-w-0">
   488→                        <div className="flex items-center gap-2 flex-wrap">
