    50→    50→// Lightweight error boundary for individual sections
    51→    51→class SectionErrorBoundary extends Component<{ children: ReactNode; fallback?: ReactNode; name?: string }, { hasError: boolean; error: Error | null }> {
    52→    52→  constructor(props: { children: ReactNode; fallback?: ReactNode; name?: string }) {
    53→    53→    super(props);
    54→    54→    this.state = { hasError: false, error: null };
    55→    55→  }
    56→    56→  static getDerivedStateFromError(error: Error) {
    57→    57→    return { hasError: true, error };
    58→    58→  }
    59→    59→  componentDidCatch(error: Error, info: React.ErrorInfo) {
    60→    60→    console.error(`[SectionErrorBoundary${this.props.name ? ` (${this.props.name})` : ''}]`, error, info.componentStack);
    61→    61→  }
    62→    62→  render() {
    63→    63→    if (this.state.hasError) {
    64→    64→      if (this.props.fallback) return this.props.fallback;
    65→    65→      return (
    66→    66→        <div className="flex min-h-[300px] items-center justify-center rounded-lg border border-dashed border-[var(--ws-glass-border)] p-6">
    67→    67→          <div className="text-center">
    68→    68→            <AlertCircle size={24} className="mx-auto mb-3 text-[var(--ws-accent)]" />
    69→    69→            <p className="text-sm font-medium text-[var(--ws-text-primary)]">Erro ao carregar esta secao</p>
    70→    70→            <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{this.state.error?.message || 'Erro desconhecido'}</p>
    71→    71→            <button
    72→    72→              onClick={() => { this.setState({ hasError: false, error: null }); }}
    73→    73→              className="mt-3 text-xs font-medium text-[var(--ws-accent)] hover:underline"
    74→    74→            >
    75→    75→              Tentar novamente
    76→    76→            </button>
    77→    77→          </div>
    78→    78→        </div>
    79→    79→      );
    80→    80→    }
    81→    81→    return this.props.children;
    82→    82→  }
    83→    83→}

    86→    86→function SafeEditor({ content, onChange, placeholder, onError }: { content: string; onChange: (html: string) => void; placeholder: string; onError?: (msg: string) => void }) {
    87→    87→  const [editorReady, setEditorReady] = useState(false);
    88→    88→  const [editorFailed, setEditorFailed] = useState(false);
    89→    89→  const [EditorComp, setEditorComp] = useState<React.ComponentType<{ content: string; onChange: (html: string) => void; placeholder: string }> | null>(null);
    90→    90→  const saveTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    91→    91→
    92→    92→  useEffect(() => {
    93→    93→    import('./RichTextEditor')
    94→    94→      .then(m => { setEditorComp(() => m.RichTextEditor); setEditorReady(true); })
    95→    95→      .catch((err) => {
    96→    96→        console.warn('[SafeEditor] RichTextEditor failed to load, using fallback:', err);
    97→    97→        setEditorFailed(true);
    98→    98→        onError?.(err?.message || 'Editor falhou ao carregar');
    99→    99→      });
   100→   100→  }, []);
   101→   101→
   102→   102→  const handleChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => {
   103→   103→    const val = e.target.value;
   104→   104→    if (saveTimer.current) clearTimeout(saveTimer.current);
   105→   105→    saveTimer.current = setTimeout(() => onChange(val), 500);
   106→   106→  }, [onChange]);
   107→   107→
   108→   108→  if (editorFailed) {
   109→   109→    return (
   110→   110→      <div>
   111→   111→        <div className="mb-3 flex items-center gap-2 rounded-lg bg-[var(--ws-accent)]/10 px-3 py-2 text-xs text-[var(--ws-accent)]">
   112→   112→          <AlertCircle size={14} /> Editor avancado indisponivel. Usando editor simples.
   113→   113→        </div>
   114→   114→        <textarea
   115→   115→          value={content}
   116→   116→          onChange={handleChange}
   117→   117→          placeholder={placeholder}
   118→   118→          className="min-h-[400px] w-full resize-none bg-transparent p-5 text-sm leading-relaxed text-[var(--ws-text-primary)] placeholder-[var(--ws-text-tertiary)] outline-none"
   119→   119→        />
   120→   120→      </div>
   121→   121→    );
   122→   122→  }
   123→   123→
   124→   124→  if (!editorReady) {
   125→   125→    return (
   126→   126→      <div className="flex min-h-[400px] items-center justify-center">
   127→   127→        <div className="text-center">
   128→   128→          <div className="mx-auto mb-3 h-6 w-6 animate-spin rounded-full border-2 border-[var(--ws-glass-border)] border-t-[var(--ws-accent)]" />
   129→   129→          <p className="text-xs text-[var(--ws-text-tertiary)]">Carregando editor...</p>
   130→   130→        </div>
   131→   131→      </div>
   132→   132→    );
   133→   133→  }
   134→   134→
   135→   135→  if (EditorComp) return <EditorComp content={content} onChange={onChange} placeholder={placeholder} />;
   136→   136→  return null;
   137→   137→}

   174→   174→function useSwipeGestures(onSwipeLeft: () => void, onSwipeRight: () => void) {
   175→   175→  const touchStartX = useRef(0);
   176→   176→  const touchEndX = useRef(0);
   177→   177→
   178→   178→  const onTouchStart = useCallback((e: React.TouchEvent) => {
   179→   179→    touchStartX.current = e.changedTouches[0].screenX;
   180→   180→  }, []);
   181→   181→
   182→   182→  const onTouchEnd = useCallback((e: React.TouchEvent) => {
   183→   183→    touchEndX.current = e.changedTouches[0].screenX;
   184→   184→    const diff = touchEndX.current - touchStartX.current;
   185→   185→    if (Math.abs(diff) > 80) {
   186→   186→      if (diff < 0) onSwipeLeft();
   187→   187→      else onSwipeRight();
   188→   188→    }
   189→   189→  }, [onSwipeLeft, onSwipeRight]);
   190→   190→
   191→   191→  return { onTouchStart, onTouchEnd };
   192→   192→}

   366→   366→function formatReviewDate(dateStr: string): string {
   367→   367→  const d = new Date(dateStr);
   368→   368→  const now = new Date();
   369→   369→  const diffMs = d.getTime() - now.getTime();
   370→   370→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   371→   371→
   372→   372→  if (diffDays < 0) return 'Atrasado';
   373→   373→  if (diffDays === 0) return 'Hoje';
   374→   374→  if (diffDays === 1) return 'Amanha';
   375→   375→  if (diffDays <= 7) return `em ${diffDays} dias`;
   376→   376→  return d.toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit' });
   377→   377→}

   958→   958→function FlashcardReviewer({ onBack }: { onBack: () => void }) {
   959→   959→  const [cards, setCards] = useState<FlashcardItem[]>([]);
   960→   960→  const [loading, setLoading] = useState(true);
   961→   961→  const [currentIndex, setCurrentIndex] = useState(0);
   962→   962→  const [flipped, setFlipped] = useState(false);
   963→   963→  const [done, setDone] = useState(false);
   964→   964→  const [reviewed, setReviewed] = useState(0);
   965→   965→  const [ratings, setRatings] = useState<number[]>([]);
   966→   966→  const fetchedRef = useRef(false);
   967→   967→
   968→   968→  useEffect(() => {
   969→   969→    if (fetchedRef.current) return;
   970→   970→    fetchedRef.current = true;
   971→   971→    fetch('/api/flashcards?due=true').then(r => r.json()).then(d => {
   972→   972→      if (d.flashcards) setCards(d.flashcards);
   973→   973→    }).catch(() => {}).finally(() => setLoading(false));
   974→   974→  }, []);
   975→   975→
   976→   976→  const handleReview = useCallback(async (quality: number) => {
   977→   977→    const card = cards[currentIndex];
   978→   978→    if (!card) return;
   979→   979→    try {
   980→   980→      await fetch(`/api/flashcards/${card.id}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ review: quality }) });
   981→   981→    } catch {}
   982→   982→    setReviewed(prev => prev + 1);
   983→   983→    setRatings(prev => [...prev, quality]);
   984→   984→    setFlipped(false);
   985→   985→    if (currentIndex + 1 >= cards.length) setDone(true);
   986→   986→    else setCurrentIndex(prev => prev + 1);
   987→   987→  }, [cards, currentIndex]);
   988→   988→
   989→   989→  // Keyboard shortcuts
   990→   990→  useEffect(() => {
   991→   991→    if (done || !flipped || cards.length === 0) return;
   992→   992→    const handleKey = (e: KeyboardEvent) => {
   993→   993→      if (e.key === '1') handleReview(1);
   994→   994→      else if (e.key === '2') handleReview(2);
   995→   995→      else if (e.key === '3') handleReview(3);
   996→   996→      else if (e.key === '4') handleReview(5);
   997→   997→    };
   998→   998→    window.addEventListener('keydown', handleKey);
   999→   999→    return () => window.removeEventListener('keydown', handleKey);
  1000→  1000→  }, [flipped, done, cards.length, handleReview]);
  1001→  1001→
  1002→  1002→  // Completion stats
  1003→  1003→  const avgRating = ratings.length > 0 ? (ratings.reduce((a, b) => a + b, 0) / ratings.length).toFixed(1) : '0';
  1004→  1004→  const hardCount = ratings.filter(r => r <= 2).length;
  1005→  1005→  const easyCount = ratings.filter(r => r >= 4).length;
  1006→  1006→
  1007→  1007→  if (loading) return <div className="flex items-center justify-center py-16"><Loader2 size={24} className="animate-spin text-[var(--ws-text-tertiary)]" /></div>;
  1008→  1008→  if (cards.length === 0) return (
  1009→  1009→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="py-16 text-center">
