   196→function getMasteryColor(mastery: number): string {
   197→  if (mastery >= 80) return '#22c55e';
   198→  if (mastery >= 60) return '#84cc16';
   199→  if (mastery >= 40) return '#f59e0b';
   200→  if (mastery >= 20) return '#f97316';
   201→  return '#ef4444';
   202→}
   203→
   204→function getMasteryLabel(mastery: number): string {
   205→  if (mastery >= 90) return 'Dominio total';
   206→  if (mastery >= 70) return 'Avancado';
   207→  if (mastery >= 50) return 'Intermediario';
   208→  if (mastery >= 25) return 'Iniciante';
   209→  return 'Nao iniciado';
   210→}
   211→
   212→function formatRelativeDate(dateStr: string | null | undefined): string {
   213→  if (!dateStr) return '—';
   214→  try {
   215→    return formatDistanceToNow(parseISO(dateStr), {
   216→      addSuffix: true,
   217→      locale: ptBR,
   218→    });
   219→  } catch {
   220→    return '—';
   221→  }
   222→}

   252→// Main export
   253→export function SubjectsView({ onNavigate }: SubjectsViewProps) {
   254→  // ===== STATE =====
   255→  const [subjects, setSubjects] = useState<SubjectWithMastery[]>([]);
   256→  const [loading, setLoading] = useState(true);
   257→  const [selectedSubjectId, setSelectedSubjectId] = useState<string | null>(null);
   258→  const [subjectDetail, setSubjectDetail] = useState<SubjectDetail | null>(null);
   259→  const [detailLoading, setDetailLoading] = useState(false);
   260→  const [detailTopics, setDetailTopics] = useState<Topic[]>([]);
   261→
   262→  // Add/Edit dialog state
   263→  const [showAddDialog, setShowAddDialog] = useState(false);
   264→  const [showEditDialog, setShowEditDialog] = useState(false);
   265→  const [formSubmitting, setFormSubmitting] = useState(false);
   266→  const [formName, setFormName] = useState('');
   267→  const [formDescription, setFormDescription] = useState('');
   268→  const [formColor, setFormColor] = useState('#f59e0b');
   269→  const [formIcon, setFormIcon] = useState('book');
   270→
   271→  // Delete confirmation
   272→  const [showDeleteDialog, setShowDeleteDialog] = useState(false);
   273→  const [deleteTarget, setDeleteTarget] = useState<string | null>(null);
   274→  const [deleting, setDeleting] = useState(false);
   275→
   276→  // Add topic form
   277→  const [showAddTopic, setShowAddTopic] = useState(false);
   278→  const [topicName, setTopicName] = useState('');
   279→  const [topicDescription, setTopicDescription] = useState('');
   280→  const [topicSubmitting, setTopicSubmitting] = useState(false);
   281→
   282→  // ===== FETCH SUBJECTS =====
   283→  const fetchSubjects = useCallback(async () => {
   284→    try {
   285→      setLoading(true);
   286→      const res = await fetch('/api/subjects');
   287→      if (!res.ok) throw new Error('Erro ao carregar materias');
   288→      const data = await res.json();
   289→
   290→      // Enrich subjects with mastery and pending task data
   291→      const enriched: SubjectWithMastery[] = await Promise.all(
   292→        (data.subjects || []).map(async (s: SubjectList) => {
   293→          try {
   294→            const detailRes = await fetch(`/api/subjects/${s.id}`);
   295→            if (!detailRes.ok) {
   296→              return {
   297→                ...s,
   298→                mastery: 0,
   299→                lastStudied: null,
   300→                pendingTasks: 0,
   301→              };
   302→            }
   303→            const detailData = await detailRes.json();
   304→            const detail: SubjectDetail = detailData.subject;
   305→            const topics = detail.topics || [];
   306→            const avgMastery =
   307→              topics.length > 0
   308→                ? Math.round(topics.reduce((sum: number, t: Topic) => sum + t.mastery, 0) / topics.length)
   309→                : 0;
   310→
   311→            // Get pending tasks count - we need to fetch tasks separately
   312→            let pendingTasks = 0;
   313→            let lastStudied: string | null = null;
   314→
   315→            // Use sessions count as a proxy - if sessions exist, it was studied
   316→            if (detail._count?.sessions > 0) {
   317→              lastStudied = detail.updatedAt || detail.createdAt;
   318→            }
   319→
   320→            return {
   321→              ...s,
   322→              mastery: avgMastery,
   323→              lastStudied,
   324→              pendingTasks,
   325→            };
   326→          } catch {
   327→            return {
   328→              ...s,
   329→              mastery: 0,
   330→              lastStudied: null,
   331→              pendingTasks: 0,
   332→            };
   333→          }
   334→        })
   335→      );
   336→
   337→      setSubjects(enriched);
   338→    } catch (err: any) {
   339→      toast({
   340→        title: 'Erro ao carregar',
   341→        description: err.message || 'Nao foi possivel carregar suas materias.',
   342→        variant: 'destructive',
   343→      });
   344→    } finally {
   345→      setLoading(false);
   346→    }
   347→  }, []);
   348→
   349→  useEffect(() => {
   350→    fetchSubjects();
   351→  }, [fetchSubjects]);
   352→
   353→  // ===== FETCH SUBJECT DETAIL =====
   354→  const fetchSubjectDetail = useCallback(async (subjectId: string) => {
   355→    try {
   356→      setDetailLoading(true);
   357→      const res = await fetch(`/api/subjects/${subjectId}`);
   358→      if (!res.ok) throw new Error('Materia nao encontrada');
   359→      const data = await res.json();
   360→      setSubjectDetail(data.subject);
   361→      setDetailTopics(data.subject.topics || []);
   362→    } catch (err: any) {
   363→      toast({
   364→        title: 'Erro',
   365→        description: err.message || 'Nao foi possivel carregar os detalhes.',
   366→        variant: 'destructive',
   367→      });
   368→    } finally {
   369→      setDetailLoading(false);
   370→    }
   371→  }, []);
   372→
   373→  useEffect(() => {
   374→    if (selectedSubjectId) {
   375→      fetchSubjectDetail(selectedSubjectId);
   376→      setShowAddTopic(false);
   377→      setTopicName('');
   378→      setTopicDescription('');
   379→    } else {
   380→      setSubjectDetail(null);
   381→      setDetailTopics([]);
   382→    }
   383→  }, [selectedSubjectId, fetchSubjectDetail]);
   384→
   385→  // ===== HANDLERS =====
   386→
   387→  // Open add dialog
   388→  const openAddDialog = () => {
   389→    setFormName('');
   390→    setFormDescription('');
   391→    setFormColor('#f59e0b');
   392→    setFormIcon('book');
   393→    setShowAddDialog(true);
   394→  };
   395→
   396→  // Open edit dialog
   397→  const openEditDialog = () => {
   398→    if (!subjectDetail) return;
   399→    setFormName(subjectDetail.name);
   400→    setFormDescription(subjectDetail.description || '');
   401→    setFormColor(subjectDetail.color);
   402→    setFormIcon(subjectDetail.icon);
   403→    setShowEditDialog(true);
   404→  };
   405→
   406→  // Submit add
   407→  const handleAddSubject = async () => {
   408→    if (!formName.trim()) {
   409→      toast({ title: 'Campo obrigatorio', description: 'Informe o nome da materia.', variant: 'destructive' });
   410→      return;
   411→    }
   412→    try {
   413→      setFormSubmitting(true);
   414→      const res = await fetch('/api/subjects', {
   415→        method: 'POST',
   416→        headers: { 'Content-Type': 'application/json' },
   417→        body: JSON.stringify({
   418→          name: formName.trim(),
   419→          description: formDescription.trim() || null,
   420→          color: formColor,
   421→          icon: formIcon,
   422→        }),
   423→      });
   424→      if (!res.ok) {
   425→        const err = await res.json();
   426→        throw new Error(err.error || 'Erro ao criar materia');
   427→      }
   428→      toast({ title: 'Materia criada!', description: `${formName.trim()} foi adicionada com sucesso.` });
   429→      setShowAddDialog(false);
   430→      fetchSubjects();
   431→    } catch (err: any) {
   432→      toast({ title: 'Erro ao criar', description: err.message, variant: 'destructive' });
   433→    } finally {
   434→      setFormSubmitting(false);
   435→    }
   436→  };
   437→
   438→  // Submit edit
   439→  const handleEditSubject = async () => {
   440→    if (!subjectDetail || !formName.trim()) {
   441→      toast({ title: 'Campo obrigatorio', description: 'Informe o nome da materia.', variant: 'destructive' });
   442→      return;
   443→    }
   444→    try {
   445→      setFormSubmitting(true);
   446→      const res = await fetch(`/api/subjects/${subjectDetail.id}`, {
   447→        method: 'PATCH',
   448→        headers: { 'Content-Type': 'application/json' },
   449→        body: JSON.stringify({
   450→          name: formName.trim(),
   451→          description: formDescription.trim() || null,
   452→          color: formColor,
   453→          icon: formIcon,
   454→        }),
   455→      });
   456→      if (!res.ok) {
   457→        const err = await res.json();
   458→        throw new Error(err.error || 'Erro ao atualizar materia');
   459→      }
   460→      toast({ title: 'Materia atualizada!', description: `${formName.trim()} foi salva com sucesso.` });
   461→      setShowEditDialog(false);
   462→      fetchSubjectDetail(subjectDetail.id);
   463→      fetchSubjects();
   464→    } catch (err: any) {
   465→      toast({ title: 'Erro ao atualizar', description: err.message, variant: 'destructive' });
   466→    } finally {
   467→      setFormSubmitting(false);
   468→    }
   469→  };
   470→
   471→  // Delete
   472→  const handleDelete = async () => {
   473→    if (!deleteTarget) return;
   474→    try {
   475→      setDeleting(true);
   476→      const res = await fetch(`/api/subjects/${deleteTarget}`, { method: 'DELETE' });
   477→      if (!res.ok) {
   478→        const err = await res.json();
   479→        throw new Error(err.error || 'Erro ao excluir materia');
   480→      }
   481→      toast({ title: 'Materia excluida', description: 'A materia foi removida com sucesso.' });
   482→      setShowDeleteDialog(false);
   483→      setDeleteTarget(null);
   484→      if (selectedSubjectId === deleteTarget) {
   485→        setSelectedSubjectId(null);
   486→      }
   487→      fetchSubjects();
   488→    } catch (err: any) {
   489→      toast({ title: 'Erro ao excluir', description: err.message, variant: 'destructive' });
   490→    } finally {
   491→      setDeleting(false);
   492→    }
   493→  };
   494→
   495→  // Add topic
   496→  const handleAddTopic = async () => {
   497→    if (!selectedSubjectId || !topicName.trim()) {
   498→      toast({ title: 'Campo obrigatorio', description: 'Informe o nome do topico.', variant: 'destructive' });
   499→      return;
   500→    }
   501→    try {
   502→      setTopicSubmitting(true);
   503→      const res = await fetch(`/api/subjects/${selectedSubjectId}/topics`, {
   504→        method: 'POST',
   505→        headers: { 'Content-Type': 'application/json' },
   506→        body: JSON.stringify({
   507→          name: topicName.trim(),
   508→          description: topicDescription.trim() || null,
   509→        }),
   510→      });
   511→      if (!res.ok) {
   512→        const err = await res.json();
   513→        throw new Error(err.error || 'Erro ao criar topico');
   514→      }
   515→      toast({ title: 'Topico adicionado!', description: `${topicName.trim()} foi criado com sucesso.` });
   516→      setTopicName('');
   517→      setTopicDescription('');
   518→      setShowAddTopic(false);
   519→      fetchSubjectDetail(selectedSubjectId);
   520→      fetchSubjects();
   521→    } catch (err: any) {
   522→      toast({ title: 'Erro ao criar topico', description: err.message, variant: 'destructive' });
   523→    } finally {
   524→      setTopicSubmitting(false);
   525→    }
   526→  };
   527→
   528→  // ===== RENDER: SUBJECT FORM DIALOG =====
   529→  const renderFormDialog = (
   530→    open: boolean,
   531→    onOpenChange: (v: boolean) => void,
   532→    title: string,
   533→    description: string,
   534→    onSubmit: () => void,
   535→    submitting: boolean,
   536→  ) => (
   537→    <Dialog open={open} onOpenChange={onOpenChange}>
   538→      <DialogContent className="sm:max-w-[480px] border-[var(--ws-glass-border)] bg-[var(--ws-glass)] backdrop-blur-xl"
   539→        style={{ borderRadius: 'var(--ws-radius-card)' }}
   540→      >
   541→        <DialogHeader>
   542→          <DialogTitle className="text-[var(--ws-text-primary)]">{title}</DialogTitle>
   543→          <DialogDescription className="text-[var(--ws-text-tertiary)]">{description}</DialogDescription>
   544→        </DialogHeader>
   545→
   546→        <div className="space-y-4 py-2">
   547→          {/* Name */}
   548→          <div className="space-y-2">
   549→            <Label htmlFor="subject-name" className="text-[var(--ws-text-secondary)]">Nome da materia *</Label>
   550→            <Input
   551→              id="subject-name"
   552→              value={formName}
   553→              onChange={(e) => setFormName(e.target.value)}
   554→              placeholder="Ex: Matematica, Portugues, Fisica..."
   555→              className="border-[var(--ws-glass-border)] bg-[var(--ws-bg)] text-[var(--ws-text-primary)] placeholder:text-[var(--ws-text-tertiary)]"
   556→              onKeyDown={(e) => e.key === 'Enter' && onSubmit()}
   557→            />
   558→          </div>
   559→
   560→          {/* Description */}
   561→          <div className="space-y-2">
   562→            <Label htmlFor="subject-desc" className="text-[var(--ws-text-secondary)]">Descricao</Label>
   563→            <Textarea
   564→              id="subject-desc"
   565→              value={formDescription}
   566→              onChange={(e) => setFormDescription(e.target.value)}
   567→              placeholder="Opcional: uma breve descricao da materia..."
   568→              rows={2}
   569→              className="border-[var(--ws-glass-border)] bg-[var(--ws-bg)] text-[var(--ws-text-primary)] placeholder:text-[var(--ws-text-tertiary)] resize-none"
   570→            />
   571→          </div>
   572→
   573→          {/* Color picker */}
   574→          <div className="space-y-2">
   575→            <Label className="text-[var(--ws-text-secondary)]">Cor</Label>
   576→            <div className="flex flex-wrap gap-2">
   577→              {COLOR_PRESETS.map((c) => (
   578→                <button
   579→                  key={c.value}
   580→                  type="button"
   581→                  onClick={() => setFormColor(c.value)}
   582→                  className="relative h-8 w-8 rounded-full transition-all duration-200 hover:scale-110 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-offset-[var(--ws-bg)]"
   583→                  style={{
   584→                    backgroundColor: c.value,
   585→                    focusRingColor: c.value,
   586→                    boxShadow: formColor === c.value ? `0 0 0 3px var(--ws-bg), 0 0 0 5px ${c.value}` : 'none',
   587→                  }}
   588→                  title={c.name}
   589→                  aria-label={`Cor ${c.name}`}
   590→                />
   591→              ))}
   592→            </div>
   593→          </div>
   594→
   595→          {/* Icon picker */}
