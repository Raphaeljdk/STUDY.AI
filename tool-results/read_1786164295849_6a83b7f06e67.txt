    64→// Lightweight error boundary for individual sections
    65→class SectionErrorBoundary extends Component<{ children: ReactNode; fallback?: ReactNode; name?: string }, { hasError: boolean; error: Error | null }> {
    66→  constructor(props: { children: ReactNode; fallback?: ReactNode; name?: string }) {
    67→    super(props);
    68→    this.state = { hasError: false, error: null };
    69→  }
    70→  static getDerivedStateFromError(error: Error) {
    71→    return { hasError: true, error };
    72→  }
    73→  componentDidCatch(error: Error, info: React.ErrorInfo) {
    74→    console.error(`[SectionErrorBoundary${this.props.name ? ` (${this.props.name})` : ''}]`, error, info.componentStack);
    75→  }
    76→  render() {
    77→    if (this.state.hasError) {
    78→      if (this.props.fallback) return this.props.fallback;
    79→      return (
    80→        <div className="flex min-h-[300px] items-center justify-center rounded-lg border border-dashed border-[var(--ws-glass-border)] p-6">
    81→          <div className="text-center">
    82→            <AlertCircle size={24} className="mx-auto mb-3 text-[var(--ws-accent)]" />
    83→            <p className="text-sm font-medium text-[var(--ws-text-primary)]">Erro ao carregar esta secao</p>
    84→            <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{this.state.error?.message || 'Erro desconhecido'}</p>
    85→            <button
    86→              onClick={() => { this.setState({ hasError: false, error: null }); }}
    87→              className="mt-3 text-xs font-medium text-[var(--ws-accent)] hover:underline"
    88→            >
    89→              Tentar novamente
    90→            </button>
    91→          </div>
    92→        </div>
    93→      );
    94→    }
    95→    return this.props.children;
    96→  }
    97→}

    99→// Safe rich text editor with fallback to plain textarea
   100→function SafeEditor({ content, onChange, placeholder, onError }: { content: string; onChange: (html: string) => void; placeholder: string; onError?: (msg: string) => void }) {
   101→  const [editorReady, setEditorReady] = useState(false);
   102→  const [editorFailed, setEditorFailed] = useState(false);
   103→  const [EditorComp, setEditorComp] = useState<React.ComponentType<{ content: string; onChange: (html: string) => void; placeholder: string }> | null>(null);
   104→  const saveTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   105→
   106→  useEffect(() => {
   107→    import('./RichTextEditor')
   108→      .then(m => { setEditorComp(() => m.RichTextEditor); setEditorReady(true); })
   109→      .catch((err) => {
   110→        console.warn('[SafeEditor] RichTextEditor failed to load, using fallback:', err);
   111→        setEditorFailed(true);
   112→        onError?.(err?.message || 'Editor falhou ao carregar');
   113→      });
   114→  }, []);
   115→
   116→  const handleChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => {
   117→    const val = e.target.value;
   118→    if (saveTimer.current) clearTimeout(saveTimer.current);
   119→    saveTimer.current = setTimeout(() => onChange(val), 500);
   120→  }, [onChange]);
   121→
   122→  if (editorFailed) {
   123→    return (
   124→      <div>
   125→        <div className="mb-3 flex items-center gap-2 rounded-lg bg-[var(--ws-accent)]/10 px-3 py-2 text-xs text-[var(--ws-accent)]">
   126→          <AlertCircle size={14} /> Editor avancado indisponivel. Usando editor simples.
   127→        </div>
   128→        <textarea
   129→          value={content}
   130→          onChange={handleChange}
   131→          placeholder={placeholder}
   132→          className="min-h-[400px] w-full resize-none bg-transparent p-5 text-sm leading-relaxed text-[var(--ws-text-primary)] placeholder-[var(--ws-text-tertiary)] outline-none"
   133→        />
   134→      </div>
   135→    );
   136→  }
   137→
   138→  if (!editorReady) {
   139→    return (
   140→      <div className="flex min-h-[400px] items-center justify-center">
   141→        <div className="text-center">
   142→          <div className="mx-auto mb-3 h-6 w-6 animate-spin rounded-full border-2 border-[var(--ws-glass-border)] border-t-[var(--ws-accent)]" />
   143→          <p className="text-xs text-[var(--ws-text-tertiary)]">Carregando editor...</p>
   144→        </div>
   145→      </div>
   146→    );
   147→  }
   148→
   149→  if (EditorComp) return <EditorComp content={content} onChange={onChange} placeholder={placeholder} />;
   150→  return null;
   151→}

   174→// ========== ACTIVE USERS HOOK ==========
   175→function useActiveUsers() {
   176→  const [count, setCount] = useState(0);
   177→  useEffect(() => {
   178→    const base = 87;
   179→    const tick = () => setCount(base + Math.floor(Math.random() * 30));
   180→    tick();
   181→    const id = setInterval(tick, 5000);
   182→    return () => clearInterval(id);
   183→  }, []);
   184→  return count;
   185→}

   187→// ========== SWIPE HOOK ==========
   188→function useSwipeGestures(onSwipeLeft: () => void, onSwipeRight: () => void) {
   189→  const touchStartX = useRef(0);
   190→  const touchEndX = useRef(0);
   191→
   192→  const onTouchStart = useCallback((e: React.TouchEvent) => {
   193→    touchStartX.current = e.changedTouches[0].screenX;
   194→  }, []);
   195→
   196→  const onTouchEnd = useCallback((e: React.TouchEvent) => {
   197→    touchEndX.current = e.changedTouches[0].screenX;
   198→    const diff = touchEndX.current - touchStartX.current;
   199→    if (Math.abs(diff) > 80) {
   200→      if (diff < 0) onSwipeLeft();
   201→      else onSwipeRight();
   202→    }
   203→  }, [onSwipeLeft, onSwipeRight]);
   204→
   205→  return { onTouchStart, onTouchEnd };
   206→}

   417→  const h = new Date().getHours();
   418→  if (h < 12) return 'Comece o dia com foco e determinacao. Cada estudo e um passo adiante.';
   419→  if (h < 18) return 'A tarde e perfeita para revisar e consolidar o conhecimento.';

   423→function formatReviewDate(dateStr: string): string {
   424→  const d = new Date(dateStr);
   425→  const now = new Date();
   426→  const diffMs = d.getTime() - now.getTime();
   427→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   428→
   429→  if (diffDays < 0) return 'Atrasado';
   430→  if (diffDays === 0) return 'Hoje';
   431→  if (diffDays === 1) return 'Amanha';
   432→  if (diffDays <= 7) return `em ${diffDays} dias`;
   433→  return d.toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit' });
   434→}

   668→  useEffect(() => {
   669→    if (fetchedRef.current) return;
   670→    fetchedRef.current = true;

  1015→function FlashcardReviewer({ onBack }: { onBack: () => void }) {
  1016→  const [cards, setCards] = useState<FlashcardItem[]>([]);
  1017→  const [loading, setLoading] = useState(true);
  1018→  const [currentIndex, setCurrentIndex] = useState(0);
  1019→  const [flipped, setFlipped] = useState(false);
  1020→  const [done, setDone] = useState(false);
  1021→  const [reviewed, setReviewed] = useState(0);
  1022→  const [ratings, setRatings] = useState<number[]>([]);
  1023→  const fetchedRef = useRef(false);
  1024→
  1025→  useEffect(() => {
  1026→    if (fetchedRef.current) return;
  1027→    fetchedRef.current = true;
  1028→    fetch('/api/flashcards?due=true').then(r => r.json()).then(d => {
  1029→      if (d.flashcards) setCards(d.flashcards);
  1030→    }).catch(() => {}).finally(() => setLoading(false));
  1031→  }, []);
  1032→
  1033→  const handleReview = useCallback(async (quality: number) => {
  1034→    const card = cards[currentIndex];
  1035→    if (!card) return;
  1036→    try {
  1037→      await fetch(`/api/flashcards/${card.id}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ review: quality }) });
  1038→    } catch {}
  1039→    setReviewed(prev => prev + 1);
  1040→    setRatings(prev => [...prev, quality]);
  1041→    setFlipped(false);
  1042→    if (currentIndex + 1 >= cards.length) setDone(true);
  1043→    else setCurrentIndex(prev => prev + 1);
  1044→  }, [cards, currentIndex]);
  1045→
  1046→  // Keyboard shortcuts
  1047→  useEffect(() => {
  1048→    if (done || !flipped || cards.length === 0) return;
  1049→    const handleKey = (e: KeyboardEvent) => {
  1050→      if (e.key === '1') handleReview(1);
  1051→      else if (e.key === '2') handleReview(2);
  1052→      else if (e.key === '3') handleReview(3);
  1053→      else if (e.key === '4') handleReview(5);

  1083→        <WabiSabiCard hover={false}>
  1084→          <p className="font-serif-jp text-xl font-bold text-[var(--ws-gold)]">{hardCount}</p>
  1085→          <p className="mt-1 text-[10px] text-[var(--ws-text-tertiary)]">Dificeis</p>

  1166→function PomodoroTimer() {
  1167→  const [mode, setMode] = useState<'focus' | 'short' | 'long'>('focus');
  1168→  const [timeLeft, setTimeLeft] = useState(25 * 60);
  1169→  const [isRunning, setIsRunning] = useState(false);
  1170→  const [totalToday, setTotalToday] = useState(0);
  1171→  const [sessions, setSessions] = useState<{ duration: number; createdAt: string }[]>([]);
  1172→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
  1173→  const durations = { focus: 25 * 60, short: 5 * 60, long: 15 * 60 };
  1174→  const modeLabels = { focus: 'Foco', short: 'Pausa Curta', long: 'Pausa Longa' };
  1175→  const fetchedRef = useRef(false);
  1176→
  1177→  useEffect(() => {
  1178→    if (fetchedRef.current) return;
  1179→    fetchedRef.current = true;
  1180→    fetch('/api/sessions').then(r => r.json()).then(d => {
  1181→      if (d.sessions) {
  1182→        const today = new Date().toDateString();
  1183→        setTotalToday(d.sessions.filter((s: any) => new Date(s.createdAt).toDateString() === today).reduce((sum: number, s: any) => sum + s.duration, 0));
  1184→        setSessions(d.sessions.slice(0, 10));
  1185→      }
  1186→    });
  1187→  }, []);
  1188→
  1189→  useEffect(() => {
  1190→    if (isRunning) {
  1191→      intervalRef.current = setInterval(() => {
  1192→        setTimeLeft(prev => {
  1193→          if (prev <= 1) {
  1194→            setIsRunning(false);
  1195→            if (mode === 'focus') {
  1196→              const dur = durations.focus - prev;
  1197→              fetch('/api/sessions', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ duration: dur, type: 'pomodoro' }) });
  1198→              setTotalToday(t => t + dur);
  1199→            }
  1200→            return 0;
  1201→          }
  1202→          return prev - 1;
  1203→        });
  1204→      }, 1000);
  1205→    } else if (intervalRef.current) clearInterval(intervalRef.current);
  1206→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
  1207→  }, [isRunning, mode, durations.focus]);
  1208→
  1209→  const switchMode = (m: 'focus' | 'short' | 'long') => { setMode(m); setTimeLeft(durations[m]); setIsRunning(false); };
  1210→  const minutes = Math.floor(timeLeft / 60);
  1211→  const seconds = timeLeft % 60;
  1212→  const progress = 1 - timeLeft / durations[mode];
  1213→
  1214→  return (
  1215→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
  1216→      <div className="mb-8">
  1217→        <h1 className="font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)] lg:text-3xl"><Timer size={24} className="mr-2 inline text-[var(--ws-accent)]" strokeWidth={1.5} />Pomodoro Timer</h1>
  1218→        <p className="mt-1 text-sm text-[var(--ws-text-tertiary)]">Estude com foco usando a tecnica Pomodoro</p>
  1219→      </div>
  1220→      <div className="mx-auto max-w-md">
  1221→        <div className="mb-8 flex justify-center gap-2">
  1222→          {(['focus', 'short', 'long'] as const).map(m => (
  1223→            <button key={m} onClick={() => switchMode(m)} className={`rounded-ws-button px-4 py-2 text-sm font-medium transition-colors ${mode === m ? 'bg-[var(--ws-accent)] text-[var(--ws-text-on-dark)]' : 'border border-[var(--ws-glass-border)] text-[var(--ws-text-tertiary)] hover:text-[var(--ws-text-secondary)]'}`}>
  1224→              {modeLabels[m]}
  1225→            </button>
  1226→          ))}
  1227→        </div>
  1228→        <WabiSabiCard className="mb-8" hover={false}>
  1229→          <div className="flex flex-col items-center py-8">
  1230→            <div className="relative flex h-56 w-56 items-center justify-center">
  1231→              <svg className="absolute inset-0 -rotate-90" viewBox="0 0 200 200">
  1232→                <circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-glass-border)" strokeWidth="4" />
  1233→                <motion.circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-accent)" strokeWidth="4" strokeLinecap="round" strokeDasharray={2 * Math.PI * 90} initial={{ strokeDashoffset: 2 * Math.PI * 90 }} animate={{ strokeDashoffset: 2 * Math.PI * 90 * (1 - progress) }} transition={{ duration: 0.5 }} />
  1234→              </svg>
  1235→              <div className="text-center">
  1236→                <p className="font-serif-jp text-5xl font-bold text-[var(--ws-text-primary)]">{String(minutes).padStart(2, '0')}:{String(seconds).padStart(2, '0')}</p>
  1237→                <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{modeLabels[mode]}</p>
  1238→              </div>
  1239→            </div>
  1240→            <div className="mt-8 flex gap-4">
  1241→              <button onClick={() => switchMode(mode)} className="rounded-ws-button p-3 text-[var(--ws-text-tertiary)] transition-colors hover:bg-[color-mix(in_srgb,var(--ws-ink)_5%,transparent)] hover:text-[var(--ws-text-primary)]" title="Reiniciar"><RotateCcw size={18} /></button>

  1244→          </div>
  1245→        </WabiSabiCard>
  1246→        <WabiSabiCard hover={false}>
  1247→          <h3 className="mb-3 text-sm font-semibold text-[var(--ws-text-secondary)]"><Clock size={14} className="mr-1.5 inline" /> Hoje</h3>
  1248→          <p className="font-serif-jp text-3xl font-bold text-[var(--ws-text-primary)]">{Math.round(totalToday / 60)} min</p>
  1249→          <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">de tempo de foco</p>

  1251→            <div className="mt-4 max-h-48 space-y-2 overflow-y-auto border-t border-[var(--ws-glass-border)] pt-4">
