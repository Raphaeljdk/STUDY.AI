   316→// ========== NOTEBOOKS LIST ==========
   317→function NotebooksList({ onOpen }: { onOpen: (id: string) => void }) {
   318→  const [notebooks, setNotebooks] = useState<NotebookItem[]>([]);
   319→  const [loading, setLoading] = useState(true);
   320→  const [creating, setCreating] = useState(false);
   321→  const [newTitle, setNewTitle] = useState('');
   322→  const [newColor, setNewColor] = useState(notebookColors[0]);
   323→  const [msg, setMsg] = useState<{ type: 'success' | 'error'; text: string } | null>(null);
   324→
   325→  const fetchedRef = useRef(false);
   326→
   327→  useEffect(() => {
   328→    if (fetchedRef.current) return;
   329→    fetchedRef.current = true;
   330→    fetch('/api/notebooks')
   331→      .then(r => r.json())
   332→      .then(data => { if (data.notebooks) setNotebooks(data.notebooks); })
   333→      .catch(() => {})
   334→      .finally(() => setLoading(false));
   335→  }, []);
   336→
   337→  const handleCreate = async (e: React.FormEvent) => {
   338→    e.preventDefault();
   339→    if (!newTitle.trim()) return;
   340→    setCreating(true);
   341→    try {
   342→      const res = await fetch('/api/notebooks', {
   343→        method: 'POST',
   344→        headers: { 'Content-Type': 'application/json' },
   345→        body: JSON.stringify({ title: newTitle, color: newColor }),
   346→      });
   347→      const data = await res.json();
   348→      if (data.notebook) {
   349→        setNotebooks(prev => [data.notebook, ...prev]);
   350→        setNewTitle('');
   351→        setCreating(false);
   352→        onOpen(data.notebook.id);
   353→      }
   354→    } catch { setMsg({ type: 'error', text: 'Erro ao criar caderno' }); }
   355→    setCreating(false);
   356→  };
   357→
   358→  const handleDelete = async (id: string, title: string) => {
   359→    if (!confirm(`Deletar o caderno "${title}"?`)) return;
   360→    try {
   361→      await fetch(`/api/notebooks/${id}`, { method: 'DELETE' });
   362→      setNotebooks(prev => prev.filter(n => n.id !== id));
   363→    } catch { setMsg({ type: 'error', text: 'Erro ao deletar' }); }
   364→  };
   365→
   366→  return (
   367→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }}>
   368→      <div className="mb-8">
   369→        <h1 className="font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)] lg:text-3xl">
   370→          Meus Cadernos
   371→        </h1>
   372→        <p className="mt-1 text-sm text-[var(--ws-text-tertiary)]">Organize seus estudos por assunto</p>
   373→      </div>
   374→
   375→      {msg && (
   376→        <div className={`mb-4 rounded-ws-button px-4 py-2.5 text-sm ${
   377→          msg.type === 'success' ? 'bg-[color-mix(in_srgb,var(--ws-verdigris)_12%,transparent)] text-[var(--ws-verdigris)]' : 'bg-[color-mix(in_srgb,var(--ws-accent)_10%,transparent)] text-[var(--ws-accent)]'
   378→        }`}>{msg.text}</div>
   379→      )}
   380→
   381→      {/* Create form */}
   382→      <WabiSabiCard className="mb-8" hover={false}>
   383→        <form onSubmit={handleCreate} className="flex flex-col gap-4 sm:flex-row sm:items-end">
   384→          <div className="flex-1">
   385→            <label className="mb-1.5 block text-sm font-medium text-[var(--ws-text-secondary)]">Novo Caderno</label>
   386→            <input
   387→              type="text"
   388→              value={newTitle}
   389→              onChange={e => setNewTitle(e.target.value)}
   390→              placeholder="Ex: Matematica, Historia, Biologia..."
   391→              className="w-full rounded-ws-button border border-[var(--ws-glass-border)] bg-[var(--ws-bg)] px-4 py-3 text-sm text-[var(--ws-text-primary)] placeholder-[var(--ws-text-tertiary)] outline-none transition-colors focus:border-[var(--ws-accent)]/30"
   392→            />
   393→          </div>
   394→          <div>
   395→            <label className="mb-1.5 block text-sm font-medium text-[var(--ws-text-secondary)]">Cor</label>
   396→            <div className="flex gap-2">
   397→              {notebookColors.map(c => (
   398→                <button
   399→                  key={c}
   400→                  type="button"
   401→                  onClick={() => setNewColor(c)}
   402→                  className={`h-8 w-8 rounded-full border-2 transition-transform ${newColor === c ? 'scale-110 border-[var(--ws-text-primary)]' : 'border-transparent'}`}
   403→                  style={{ backgroundColor: c }}
   404→                />

   845→      .then(d => {
   846→        if (d.flashcards) setCards(d.flashcards);
   847→      })
   848→      .finally(() => setLoading(false));
   849→  }, []);
   850→
   851→  const handleReview = async (quality: number) => {
   852→    const card = cards[currentIndex];
   853→    if (!card) return;
   854→
   855→    try {
   856→      await fetch(`/api/flashcards/${card.id}`, {
   857→        method: 'PATCH',
   858→        headers: { 'Content-Type': 'application/json' },
   859→        body: JSON.stringify({ review: quality }),
   860→      });
   861→    } catch { /* ignore */ }
   862→
   863→    const newReviewed = reviewed + 1;
   864→    setReviewed(newReviewed);
   865→    setFlipped(false);
   866→
   867→    if (currentIndex + 1 >= cards.length) {
   868→      setDone(true);
   869→    } else {
   870→      setCurrentIndex(prev => prev + 1);
   871→    }
   872→  };
   873→
   874→  if (loading) {
   875→    return <div className="flex items-center justify-center py-16"><Loader2 size={24} className="animate-spin text-[var(--ws-text-tertiary)]" /></div>;
   876→  }
   877→
   878→  if (cards.length === 0) {
   879→    return (
   880→      <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="py-16 text-center">
   881→        <Check size={48} className="mx-auto mb-4 text-[var(--ws-verdigris)]" strokeWidth={1} />
   882→        <p className="text-sm text-[var(--ws-text-tertiary)]">Nenhuma carta para revisar agora</p>
   883→        <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">Volte mais tarde ou crie novos flashcards</p>
   884→        <button onClick={onBack} className="mt-4 text-sm text-[var(--ws-accent)] hover:underline">Voltar</button>
   885→      </motion.div>
   886→    );
   887→  }
   888→
   889→  if (done) {
   890→    return (
   891→      <motion.div initial={{ opacity: 0, scale: 0.95 }} animate={{ opacity: 1, scale: 1 }} className="py-16 text-center">
   892→        <EnsoCircle size={80} strokeWidth={2} color="var(--ws-verdigris)" imperfection={0.1} animate={false} />
   893→        <h2 className="mt-6 font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)]">Sessao Completa!</h2>
   894→        <p className="mt-2 text-sm text-[var(--ws-text-tertiary)]">Voce revisou {reviewed} cartas</p>
   895→        <ZenButton variant="primary" size="md" className="mt-6" onClick={onBack}>Voltar aos Flashcards</ZenButton>
   896→      </motion.div>
   897→    );
   898→  }
   899→
   900→  const card = cards[currentIndex];
   901→
   902→  return (
   903→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }}>
   904→      <div className="mb-6 flex items-center justify-between">
   905→        <button onClick={onBack} className="flex items-center gap-2 text-sm text-[var(--ws-text-tertiary)] hover:text-[var(--ws-text-primary)]">
   906→          <ArrowLeft size={16} /> Voltar
   907→        </button>
   908→        <span className="text-xs text-[var(--ws-text-tertiary)]">{currentIndex + 1} de {cards.length}</span>
   909→      </div>
   910→
   911→      {/* Progress bar */}
   912→      <div className="mb-8 h-1 w-full rounded-full bg-[var(--ws-glass-border)]">
   913→        <motion.div
   914→          className="h-1 rounded-full bg-[var(--ws-accent)]"
   915→          initial={{ width: 0 }}
   916→          animate={{ width: `${((currentIndex + 1) / cards.length) * 100}%` }}
   917→          transition={{ duration: 0.3 }}
   918→        />
   919→      </div>
   920→
   921→      {/* Card */}
   922→      <div className="mx-auto max-w-xl">
   923→        <motion.div
   924→          className="perspective-1000 cursor-pointer"
   925→          onClick={() => !flipped && setFlipped(true)}
   926→          style={{ perspective: 1000 }}
   927→        >
   928→          <motion.div
   929→            className="min-h-[280px]"
   930→            animate={{ rotateY: flipped ? 180 : 0 }}
   931→            transition={{ duration: 0.6, type: 'spring', stiffness: 100 }}
   932→            style={{ transformStyle: 'preserve-3d' }}
   933→          >
   934→            {/* Front */}
   935→            <div
   936→              className="absolute inset-0 flex flex-col items-center justify-center rounded-ws-organic border border-[var(--ws-glass-border)] bg-[var(--ws-glass)] p-8 backdrop-blur-xl"
   937→              style={{ backfaceVisibility: 'hidden' }}
   938→            >
   939→              <p className="mb-4 text-xs font-medium text-[var(--ws-accent)]">PERGUNTA</p>
   940→              <p className="text-center text-lg font-medium leading-relaxed text-[var(--ws-text-primary)]">{card.front}</p>
   941→              <p className="mt-6 text-xs text-[var(--ws-text-tertiary)]">Clique para ver a resposta</p>
   942→            </div>
   943→
   944→            {/* Back */}
   945→            <div
   946→              className="absolute inset-0 flex flex-col items-center justify-center rounded-ws-organic border border-[var(--ws-accent)]/20 bg-[var(--ws-bg)] p-8"
   947→              style={{ backfaceVisibility: 'hidden', transform: 'rotateY(180deg)' }}
   948→            >
   949→              <p className="mb-4 text-xs font-medium text-[var(--ws-verdigris)]">RESPOSTA</p>
   950→              <p className="text-center text-base leading-relaxed text-[var(--ws-text-primary)]">{card.back}</p>
   951→            </div>
   952→          </motion.div>
   953→        </motion.div>
   954→
   955→        {/* Rating buttons */}
   956→        <AnimatePresence>
   957→          {flipped && (
   958→            <motion.div
   959→              initial={{ opacity: 0, y: 20 }}
   960→              animate={{ opacity: 1, y: 0 }}
   961→              className="mt-8 flex flex-wrap justify-center gap-3"
   962→            >
   963→              <ReviewButton label="Novamente" sub="1 min" quality={1} color="var(--ws-accent)" onClick={() => handleReview(1)} />
   964→              <ReviewButton label="Dificil" sub="6 min" quality={2} color="#d35400" onClick={() => handleReview(2)} />
   965→              <ReviewButton label="Bom" sub="10 min" quality={3} color="#27ae60" onClick={() => handleReview(3)} />
   966→              <ReviewButton label="Facil" sub="4 dias" quality={5} color="var(--ws-verdigris)" onClick={() => handleReview(5)} />
   967→            </motion.div>
   968→          )}
   969→        </AnimatePresence>
   970→      </div>
   971→    </motion.div>
   972→  );
   973→}

   987→// ========== POMODORO TIMER ==========
   988→function PomodoroTimer() {
   989→  const [mode, setMode] = useState<'focus' | 'short' | 'long'>('focus');
   990→  const [timeLeft, setTimeLeft] = useState(25 * 60);
   991→  const [isRunning, setIsRunning] = useState(false);
   992→  const [sessions, setSessions] = useState<{ duration: number; createdAt: string; type: string }[]>([]);
   993→  const [totalToday, setTotalToday] = useState(0);
   994→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   995→
   996→  const durations = { focus: 25 * 60, short: 5 * 60, long: 15 * 60 };
   997→  const modeLabels = { focus: 'Foco', short: 'Pausa Curta', long: 'Pausa Longa' };
   998→
   999→  useEffect(() => {
  1000→    fetch('/api/sessions').then(r => r.json()).then(d => {
  1001→      if (d.sessions) {
  1002→        const today = new Date().toDateString();
  1003→        const todaySessions = d.sessions.filter((s: any) => new Date(s.createdAt).toDateString() === today);
  1004→        setTotalToday(todaySessions.reduce((sum: number, s: any) => sum + s.duration, 0));
  1005→        setSessions(d.sessions.slice(0, 10));
  1006→      }
  1007→    });
  1008→  }, []);
  1009→
  1010→  useEffect(() => {
  1011→    if (isRunning) {
  1012→      intervalRef.current = setInterval(() => {
  1013→        setTimeLeft(prev => {
  1014→          if (prev <= 1) {
  1015→            setIsRunning(false);
  1016→            if (mode === 'focus') {
  1017→              // Save session
  1018→              const duration = durations.focus - prev;
  1019→              fetch('/api/sessions', {
  1020→                method: 'POST',
  1021→                headers: { 'Content-Type': 'application/json' },
  1022→                body: JSON.stringify({ duration, type: 'pomodoro' }),
  1023→              });
  1024→              setTotalToday(t => t + duration);
  1025→            }
  1026→            return 0;
  1027→          }
  1028→          return prev - 1;
  1029→        });
  1030→      }, 1000);
  1031→    } else if (intervalRef.current) {
  1032→      clearInterval(intervalRef.current);
  1033→    }
  1034→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
  1035→  }, [isRunning, mode, durations.focus]);
  1036→
  1037→  const switchMode = (m: 'focus' | 'short' | 'long') => {
  1038→    setMode(m);
  1039→    setTimeLeft(durations[m]);
  1040→    setIsRunning(false);
  1041→  };
  1042→
  1043→  const minutes = Math.floor(timeLeft / 60);
  1044→  const seconds = timeLeft % 60;
  1045→  const progress = 1 - timeLeft / durations[mode];
  1046→  const totalTimeMin = Math.round(totalToday / 60);
  1047→
  1048→  return (
  1049→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }}>
  1050→      <div className="mb-8">
  1051→        <h1 className="font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)] lg:text-3xl">
  1052→          <Timer size={24} className="mr-2 inline text-[var(--ws-accent)]" strokeWidth={1.5} />Pomodoro Timer
  1053→        </h1>
  1054→        <p className="mt-1 text-sm text-[var(--ws-text-tertiary)]">Estude com foco usando a tecnica Pomodoro</p>
  1055→      </div>
  1056→
  1057→      <div className="mx-auto max-w-md">
  1058→        {/* Mode selector */}
  1059→        <div className="mb-8 flex justify-center gap-2">
  1060→          {(['focus', 'short', 'long'] as const).map(m => (
  1061→            <button
  1062→              key={m}
  1063→              onClick={() => switchMode(m)}
  1064→              className={`rounded-ws-button px-4 py-2 text-sm font-medium transition-colors ${
  1065→                mode === m
  1066→                  ? 'bg-[var(--ws-accent)] text-[var(--ws-text-on-dark)]'
  1067→                  : 'border border-[var(--ws-glass-border)] text-[var(--ws-text-tertiary)] hover:text-[var(--ws-text-secondary)]'
  1068→              }`}
  1069→            >
  1070→              {modeLabels[m]}
  1071→            </button>
  1072→          ))}
  1073→        </div>
  1074→
  1075→        {/* Timer circle */}
  1076→        <WabiSabiCard className="mb-8" hover={false}>
  1077→          <div className="flex flex-col items-center py-8">
  1078→            <div className="relative flex h-56 w-56 items-center justify-center">
  1079→              {/* Progress ring */}
  1080→              <svg className="absolute inset-0 -rotate-90" viewBox="0 0 200 200">
  1081→                <circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-glass-border)" strokeWidth="4" />
  1082→                <motion.circle
  1083→                  cx="100" cy="100" r="90" fill="none" stroke="var(--ws-accent)" strokeWidth="4" strokeLinecap="round"
  1084→                  strokeDasharray={2 * Math.PI * 90}
  1085→                  animate={{ strokeDashoffset: 2 * Math.PI * 90 * (1 - progress) }}
  1086→                  transition={{ duration: 0.5 }}
  1087→                />
  1088→              </svg>
  1089→              <div className="text-center">
  1090→                <p className="font-serif-jp text-5xl font-bold text-[var(--ws-text-primary)]">
  1091→                  {String(minutes).padStart(2, '0')}:{String(seconds).padStart(2, '0')}
  1092→                </p>
  1093→                <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{modeLabels[mode]}</p>
  1094→              </div>
  1095→            </div>
  1096→
  1097→            {/* Controls */}
  1098→            <div className="mt-8 flex gap-4">
  1099→              <button
  1100→                onClick={() => switchMode(mode)}
  1101→                className="rounded-ws-button p-3 text-[var(--ws-text-tertiary)] transition-colors hover:bg-[color-mix(in_srgb,var(--ws-ink)_5%,transparent)] hover:text-[var(--ws-text-primary)]"
  1102→                title="Reiniciar"
  1103→              >
  1104→                <RotateCcw size={18} />
  1105→              </button>
  1106→              <ZenButton
  1107→                variant="primary"
  1108→                size="lg"
  1109→                onClick={() => setIsRunning(!isRunning)}
  1110→                className="px-12"
  1111→              >
  1112→                {isRunning ? 'Pausar' : 'Iniciar'}
  1113→              </ZenButton>
  1114→            </div>
  1115→          </div>
  1116→        </WabiSabiCard>
  1117→
  1118→        {/* Today's stats */}
  1119→        <WabiSabiCard hover={false}>
  1120→          <h3 className="mb-3 text-sm font-semibold text-[var(--ws-text-secondary)]">
  1121→            <Clock size={14} className="mr-1.5 inline" /> Hoje
  1122→          </h3>
  1123→          <p className="font-serif-jp text-3xl font-bold text-[var(--ws-text-primary)]">{totalTimeMin} min</p>
  1124→          <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">de tempo de foco</p>
  1125→
  1126→          {sessions.length > 0 && (

  1131→                  <span className="font-medium text-[var(--ws-text-primary)]">{Math.round(s.duration / 60)} min</span>
  1132→                </div>
  1133→              ))}
  1134→            </div>
  1135→          )}
  1136→        </WabiSabiCard>
  1137→      </div>
  1138→    </motion.div>
  1139→  );
