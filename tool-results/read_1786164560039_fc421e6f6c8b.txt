    64→/* ========================================================================
    65→   Toolbar Button Component (outside render to avoid re-creation)
    66→   ======================================================================== */
    67→
    68→function TBtn({
    69→  title, active, onClick, disabled, children, className = '',
    70→}: {
    71→  title: string;
    72→  active?: boolean;
    73→  onClick: () => void;
    74→  disabled?: boolean;
    75→  children: React.ReactNode;
    76→  className?: string;
    77→}) {
    78→  return (
    79→    <button
    80→      title={title}
    81→      onClick={onClick}
    82→      disabled={disabled}
    83→      className={`flex h-8 w-8 items-center justify-center rounded-lg transition-colors ${
    84→        active
    85→          ? 'bg-amber-500 text-white shadow-sm'
    86→          : 'text-gray-600 hover:bg-gray-100 hover:text-gray-900'
    87→      } ${disabled ? 'opacity-30 cursor-not-allowed' : ''} ${className}`}
    88→    >
    89→      {children}
    90→    </button>
    91→  );
    92→}
    93→}
    94→

    99→function Dropdown({
   100→  label, icon, items, value,
   101→}: {
   102→  label: string;
   103→  icon: React.ReactNode;
   104→  items: { value: string; label: string }[];
   105→  value?: string;
   106→}) {
   107→  const [open, setOpen] = useState(false);
   108→  const ref = useRef<HTMLDivElement>(null);
   109→
   110→  useEffect(() => {
   111→    if (!open) return;
   112→    const handler = (e: MouseEvent) => {
   113→      if (ref.current && !ref.current.contains(e.target as Node)) setOpen(false);
   114→    };
   115→    document.addEventListener('mousedown', handler);
   116→    return () => document.removeEventListener('mousedown', handler);
   117→  }, [open]);
   118→
   119→  return (
   120→    <div ref={ref} className="relative">
   121→      <button
   122→        onClick={() => setOpen(!open)}
   123→        className="flex h-8 items-center gap-1 rounded-lg border border-gray-200 px-2 text-xs font-medium text-gray-700 hover:bg-gray-50 transition-colors"
   124→        title={label}
   125→      >
   126→        {icon}
   127→        <ChevronDown size={10} className="text-gray-400" />
   128→      </button>
   129→      {open && (
   130→        <div className="absolute left-0 top-full z-50 mt-1 max-h-72 w-56 overflow-y-auto rounded-lg border border-gray-200 bg-white py-1 shadow-xl">
   131→          {items.map((item) => (
   132→            <button
   133→              key={item.value}
   134→              onClick={() => { setOpen(false); }}
   135→              className={`w-full px-3 py-1.5 text-left text-sm hover:bg-gray-100 transition-colors ${
   136→                value === item.value ? 'bg-amber-50 font-semibold text-amber-700' : 'text-gray-700'
   137→              }`}
   138→            >
   139→              {item.label}
   140→            </button>
   141→          ))}
   142→        </div>
   143→      )}
   144→    </div>
   145→  );
   146→}

   151→
   152→export default function DocumentEditor({
   153→  content = '',
   154→  paperStyle = 'blank',
   155→  paperColor = '#ffffff',
   156→  onChange,
   157→}: DocumentEditorProps) {
   158→  const editorRef = useRef<any>(null);
   159→  const [wordCount, setWordCount] = useState(0);
   160→  const [charCount, setCharCount] = useState(0);
   161→  const [showFontPicker, setShowFontPicker] = useState(false);
   162→  const [showSizePicker, setShowSizePicker] = useState(false);
   163→  const [showColorPicker, setShowColorPicker] = useState(false);
   164→  const [showHighlightPicker, setShowHighlightPicker] = useState(false);
   165→  const fontRef = useRef<HTMLDivElement>(null);
   166→  const sizeRef = useRef<HTMLDivElement>(null);
   167→  const colorRef = useRef<HTMLDivElement>(null);
   168→  const highlightRef = useRef<HTMLDivElement>(null);
   169→
   170→  const editor = useEditor({
   171→    extensions: [
   172→      StarterKit.configure({
   173→        heading: { levels: [1, 2, 3, 4, 5, 6] },
   174→      }),
   175→      Underline,
   176→      TextAlign.configure({ types: ['heading', 'paragraph'] }),
   177→      Highlight.configure({ multicolor: true }),
   178→      Placeholder.configure({ placeholder: 'Comece a digitar seu texto aqui...' }),
   179→      TaskList,
   180→      TaskItem.configure({ nested: true }),
   181→      Color,
   182→      TextStyle,
   183→    ],
   184→    content: content || '',
   185→    editorProps: {
   186→      attributes: {
   187→        class: 'document-editor-content',
   188→      },
   189→    },
   190→    onUpdate: ({ editor }) => {
   191→      const html = editor.getHTML();
   192→      onChange?.(html);
   193→      updateCounts(editor);
   194→    },
   195→    immediatelyRender: false,
   196→  });
   197→
   198→  editorRef.current = editor;
   199→
   200→  /* ---- Close dropdowns on outside click ---- */
   201→  useEffect(() => {
   202→    const handler = (e: MouseEvent) => {
   203→      if (fontRef.current && !fontRef.current.contains(e.target as Node)) setShowFontPicker(false);
   204→      if (sizeRef.current && !sizeRef.current.contains(e.target as Node)) setShowSizePicker(false);
   205→      if (colorRef.current && !colorRef.current.contains(e.target as Node)) setShowColorPicker(false);
   206→      if (highlightRef.current && !highlightRef.current.contains(e.target as Node)) setShowHighlightPicker(false);
   207→    };
   208→    document.addEventListener('mousedown', handler);
   209→    return () => document.removeEventListener('mousedown', handler);
   210→  }, []);
   211→
   212→  /* ---- Word/char count ---- */
   213→  const updateCounts = useCallback((ed: any) => {
   214→    const text = ed?.getText() || '';
   215→    setCharCount(text.length);
   216→    setWordCount(text.trim() ? text.trim().split(/\s+/).length : 0);
   217→  }, []);
   218→
   219→  useEffect(() => {
   220→    if (editor) updateCounts(editor);
   221→  }, [editor, updateCounts]);
   222→
   223→  /* ---- Font family ---- */
   224→  const applyFontFamily = useCallback((font: string) => {
   225→    if (!editor) return;
   226→    editor.chain().focus().setMark('textStyle', { fontFamily: font }).run();
   227→    setShowFontPicker(false);
   228→  }, [editor]);
   229→
   230→  /* ---- Font size via style attribute ---- */
   231→  const applyFontSize = useCallback((size: number) => {
   232→    if (!editor) return;
   233→    editor.chain().focus().setMark('textStyle', { fontSize: `${size}px` }).run();
   234→    setShowSizePicker(false);
   235→  }, [editor]);
   236→
   237→  /* ---- Heading ---- */
   238→  const applyHeading = useCallback((level: number | null) => {
   239→    if (!editor) return;
   240→    if (level) {
   241→      editor.chain().focus().toggleHeading({ level: level as 1 | 2 | 3 | 4 | 5 | 6 }).run();
   242→    } else {
   243→      editor.chain().focus().setParagraph().run();
   244→    }
   245→  }, [editor]);
   246→
   247→  /* ---- Line height ---- */
   248→  const applyLineHeight = useCallback((lh: number) => {
   249→    if (!editor) return;
   250→    editor.chain().focus().setMark('textStyle', { lineHeight: lh }).run();
   251→  }, [editor]);
   252→
   253→  /* ---- Print ---- */
   254→  const handlePrint = useCallback(() => {
   255→    if (!editor) return;
   256→    const printWindow = window.open('', '_blank');
   257→    if (!printWindow) return;
   258→    printWindow.document.write(`
   259→      <html><head><title>Documento</title>
   260→      <style>body{font-family:Inter,sans-serif;padding:40px;max-width:800px;margin:auto;line-height:1.6;color:#111}
   261→      h1{font-size:28px;font-weight:700;margin:20px 0 10px}h2{font-size:22px;font-weight:600;margin:18px 0 8px}h3{font-size:18px;font-weight:600;margin:16px 0 6px}
   262→      p{margin:8px 0}ul,ol{padding-left:24px;margin:8px 0}li{margin:4px 0}blockquote{border-left:3px solid #ddd;padding-left:16px;color:#555;margin:12px 0}
   263→      code{background:#f3f4f6;padding:2px 6px;border-radius:4px;font-size:14px}pre{background:#1e1e1e;color:#d4d4d4;padding:16px;border-radius:8px;overflow-x:auto}
   264→      mark{background:#fef08a;padding:1px 3px;border-radius:2px}.task-list{list-style:none;padding:0}.task-list li{display:flex;align-items:center;gap:8px}
   265→      hr{border:none;border-top:1px solid #e5e7eb;margin:16px 0}</style></head>
   266→      <body>${editor.getHTML()}</body></html>
   267→    `);
   268→    printWindow.document.close();
   269→    printWindow.print();
   270→  }, [editor]);
   271→
   272→  /* ---- Indent/Outdent ---- */
   273→  const handleIndent = useCallback(() => {
   274→    if (!editor) return;
   275→    // For list items, sink them
   276→    if (editor.isActive('listItem')) {
   277→      editor.chain().focus().sinkListItem('listItem').run();
   278→    }
   279→  }, [editor]);
   280→
   281→  const handleOutdent = useCallback(() => {
   282→    if (!editor) return;
   283→    if (editor.isActive('listItem')) {
   284→      editor.chain().focus().liftListItem('listItem').run();
   285→    }
   286→  }, [editor]);
   287→
   288→  if (!editor) {
   289→    return (
   290→      <div className="flex h-full items-center justify-center">
   291→        <div className="text-center">
   292→          <div className="mx-auto mb-3 h-6 w-6 animate-spin rounded-full border-2 border-gray-300 border-t-amber-500" />
   293→          <p className="text-sm text-gray-400">Carregando editor...</p>
   294→        </div>
   295→      </div>
   296→    );
   297→  }
   298→
   299→  const currentFont = 'Inter';
   300→  const currentSize = 16;
   301→
   302→  /* ---- Paper styles ---- */
   303→  const paperBg = paperColor || '#ffffff';
   304→  let contentClass = 'prose prose-sm sm:prose-base max-w-none';
   305→  if (paperStyle === 'lined') {
   306→    contentClass += ' lined-paper';
   307→  } else if (paperStyle === 'grid') {
   308→    contentClass += ' grid-paper';
   309→  } else if (paperStyle === 'dotted') {
   310→    contentClass += ' dotted-paper';
   311→  }
   312→
   313→  return (
   314→    <div className="flex h-full w-full flex-col" style={{ backgroundColor: '#f3f4f6' }}>
   315→      {/* ====== WORD-LIKE RIBBON TOOLBAR ====== */}
   316→      <div className="sticky top-0 z-50 border-b border-gray-200 bg-white/95 backdrop-blur-md">
   317→        {/* Row 1: Main formatting */}
   318→        <div className="flex flex-wrap items-center gap-0.5 px-2 py-1.5">
   319→          {/* Undo/Redo */}
   320→          <TBtn title="Desfazer (Ctrl+Z)" onClick={() => editor.chain().focus().undo().run()} disabled={!editor.can().undo()}>
   321→            <Undo2 size={15} />
   322→          </TBtn>
   323→          <TBtn title="Refazer (Ctrl+Shift+Z)" onClick={() => editor.chain().focus().redo().run()} disabled={!editor.can().redo()}>
   324→            <Redo2 size={15} />
   325→          </TBtn>
   326→
   327→          <Divider />
   328→
   329→          {/* Font Family */}
   330→          <div ref={fontRef} className="relative">
   331→            <button
   332→              onClick={() => { setShowFontPicker(!showFontPicker); setShowSizePicker(false); }}
   333→              className="flex h-8 max-w-[130px] items-center gap-1 truncate rounded-lg border border-gray-200 px-2 text-xs font-medium text-gray-700 hover:bg-gray-50 transition-colors"
   334→              title="Fonte"
   335→            >
   336→              <Type size={12} className="shrink-0" />
   337→              <span className="truncate">{currentFont}</span>
   338→              <ChevronDown size={10} className="shrink-0 text-gray-400" />
   339→            </button>
   340→            {showFontPicker && (
   341→              <div className="absolute left-0 top-full z-50 mt-1 max-h-72 w-48 overflow-y-auto rounded-lg border border-gray-200 bg-white py-1 shadow-xl">
   342→                {FONT_FAMILIES.map((f) => (
   343→                  <button
   344→                    key={f.value}
   345→                    onClick={() => applyFontFamily(f.value)}
   346→                    className={`w-full px-3 py-2 text-left text-sm hover:bg-gray-100 transition-colors ${currentFont === f.label ? 'bg-amber-50 font-semibold text-amber-700' : 'text-gray-700'}`}
   347→                    style={{ fontFamily: f.value }}
   348→                  >
   349→                    {f.label}
   350→                  </button>
   351→                ))}
   352→              </div>
   353→            )}
   354→          </div>
   355→
   356→          {/* Font Size */}
   357→          <div ref={sizeRef} className="relative">
   358→            <button
   359→              onClick={() => { setShowSizePicker(!showSizePicker); setShowFontPicker(false); }}
   360→              className="flex h-8 w-14 items-center justify-center gap-1 rounded-lg border border-gray-200 text-xs font-medium text-gray-700 hover:bg-gray-50 transition-colors"
   361→              title="Tamanho da fonte"
   362→            >
   363→              {currentSize}
   364→              <ChevronDown size={10} className="text-gray-400" />
   365→            </button>
   366→            {showSizePicker && (
   367→              <div className="absolute left-0 top-full z-50 mt-1 max-h-72 w-16 overflow-y-auto rounded-lg border border-gray-200 bg-white py-1 shadow-xl">
   368→                {FONT_SIZES.map((s) => (
   369→                  <button
   370→                    key={s}
   371→                    onClick={() => applyFontSize(s)}
   372→                    className={`w-full px-2 py-1 text-center text-sm hover:bg-gray-100 transition-colors ${currentSize === s ? 'bg-amber-50 font-bold text-amber-700' : 'text-gray-700'}`}
   373→                  >
   374→                    {s}
   375→                  </button>
   376→                ))}
   377→              </div>
   378→            )}
   379→          </div>
   380→
   381→          <Divider />
   382→
   383→          {/* Heading Styles */}
   384→          <TBtn title="Titulo 1" active={editor.isActive('heading', { level: 1 })} onClick={() => applyHeading(1)}>
   385→            <Heading1 size={15} />
   386→          </TBtn>
   387→          <TBtn title="Titulo 2" active={editor.isActive('heading', { level: 2 })} onClick={() => applyHeading(2)}>
   388→            <Heading2 size={15} />
   389→          </TBtn>
   390→          <TBtn title="Titulo 3" active={editor.isActive('heading', { level: 3 })} onClick={() => applyHeading(3)}>
   391→            <Heading3 size={15} />
   392→          </TBtn>
   393→          <TBtn title="Corpo de texto" active={!editor.isActive('heading')} onClick={() => applyHeading(null)}>
   394→            <Pilcrow size={15} />
   395→          </TBtn>
   396→
   397→          <Divider />
   398→
   399→          {/* Text Formatting */}
   400→          <TBtn title="Negrito (Ctrl+B)" active={editor.isActive('bold')} onClick={() => editor.chain().focus().toggleBold().run()}>
   401→            <Bold size={15} />
   402→          </TBtn>
   403→          <TBtn title="Italico (Ctrl+I)" active={editor.isActive('italic')} onClick={() => editor.chain().focus().toggleItalic().run()}>
   404→            <Italic size={15} />
   405→          </TBtn>
   406→          <TBtn title="Sublinhado (Ctrl+U)" active={editor.isActive('underline')} onClick={() => editor.chain().focus().toggleUnderline().run()}>
   407→            <UnderlineIcon size={15} />
   408→          </TBtn>
   409→          <TBtn title="Tachado" active={editor.isActive('strike')} onClick={() => editor.chain().focus().toggleStrike().run()}>
   410→            <Strikethrough size={15} />
   411→          </TBtn>
   412→          <TBtn title="Codigo" active={editor.isActive('code')} onClick={() => editor.chain().focus().toggleCode().run()}>
   413→            <Code size={15} />
   414→          </TBtn>
   415→
   416→          <Divider />
   417→
   418→          {/* Text Color */}
   419→          <div ref={colorRef} className="relative">
   420→            <TBtn title="Cor do texto" onClick={() => { setShowColorPicker(!showColorPicker); setShowHighlightPicker(false); }}>
   421→              <div className="relative">
   422→                <Palette size={15} />
   423→                <div className="absolute -bottom-0.5 left-0 right-0 h-1 rounded-full bg-black" />
   424→              </div>
   425→            </TBtn>
   426→            {showColorPicker && (
   427→              <div className="absolute left-0 top-full z-50 mt-1 w-44 rounded-lg border border-gray-200 bg-white p-2 shadow-xl">
   428→                <p className="mb-1.5 text-[10px] font-medium text-gray-400 uppercase">Cor do texto</p>
   429→                <div className="grid grid-cols-6 gap-1">
   430→                  {TEXT_COLORS.map((c) => (
   431→                    <button
   432→                      key={c}
   433→                      onClick={() => { editor.chain().focus().setColor(c).run(); setShowColorPicker(false); }}
   434→                      className={`h-6 w-6 rounded-md border transition-transform hover:scale-110 ${c === '#ffffff' ? 'border-gray-300' : 'border-transparent'}`}
   435→                      style={{ backgroundColor: c }}
   436→                      title={c}
   437→                    />
   438→                  ))}
   439→                </div>
   440→              </div>
   441→            )}
   442→          </div>
   443→
   444→          {/* Highlight Color */}
   445→          <div ref={highlightRef} className="relative">
   446→            <TBtn title="Realce" onClick={() => { setShowHighlightPicker(!showHighlightPicker); setShowColorPicker(false); }}>
   447→              <Highlighter size={15} />
   448→            </TBtn>
   449→            {showHighlightPicker && (
   450→              <div className="absolute left-0 top-full z-50 mt-1 w-44 rounded-lg border border-gray-200 bg-white p-2 shadow-xl">
   451→                <p className="mb-1.5 text-[10px] font-medium text-gray-400 uppercase">Cor de realce</p>
   452→                <div className="grid grid-cols-5 gap-1">
   453→                  {HIGHLIGHT_COLORS.map((c) => (
   454→                    <button
   455→                      key={c}
   456→                      onClick={() => { 
   457→                        if (c === 'transparent') {
   458→                          editor.chain().focus().unsetHighlight().run();
   459→                        } else {
   460→                          editor.chain().focus().toggleHighlight({ color: c }).run();
   461→                        }
   462→                        setShowHighlightPicker(false);
   463→                      }}
   464→                      className={`h-6 w-6 rounded-md border border-gray-200 transition-transform hover:scale-110 ${c === 'transparent' ? 'bg-white' : ''}`}
   465→                      style={c !== 'transparent' ? { backgroundColor: c } : {}}
   466→                      title={c === 'transparent' ? 'Sem realce' : c}
   467→                    />
   468→                  ))}
   469→                </div>
   470→              </div>
   471→            )}
   472→          </div>
   473→
   474→          <Divider />
   475→
   476→          {/* Alignment */}
   477→          <TBtn title="Alinhar a esquerda" active={editor.isActive({ textAlign: 'left' })} onClick={() => editor.chain().focus().setTextAlign('left').run()}>
   478→            <AlignLeft size={15} />
   479→          </TBtn>
   480→          <TBtn title="Centralizar" active={editor.isActive({ textAlign: 'center' })} onClick={() => editor.chain().focus().setTextAlign('center').run()}>
   481→            <AlignCenter size={15} />
   482→          </TBtn>
   483→          <TBtn title="Alinhar a direita" active={editor.isActive({ textAlign: 'right' })} onClick={() => editor.chain().focus().setTextAlign('right').run()}>
   484→            <AlignRight size={15} />
   485→          </TBtn>
   486→          <TBtn title="Justificar" active={editor.isActive({ textAlign: 'justify' })} onClick={() => editor.chain().focus().setTextAlign('justify').run()}>
   487→            <AlignJustify size={15} />
   488→          </TBtn>
   489→
   490→          <Divider />
   491→
   492→          {/* Lists */}
   493→          <TBtn title="Lista com marcadores" active={editor.isActive('bulletList')} onClick={() => editor.chain().focus().toggleBulletList().run()}>
   494→            <List size={15} />
