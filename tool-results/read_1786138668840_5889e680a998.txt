    36→// Lightweight error boundary for individual sections
    37→class SectionErrorBoundary extends Component<{ children: ReactNode; fallback?: ReactNode; name?: string }, { hasError: boolean; error: Error | null }> {
    38→  constructor(props: { children: ReactNode; fallback?: ReactNode; name?: string }) {
    39→    super(props);
    40→    this.state = { hasError: false, error: null };
    41→  }
    42→  static getDerivedStateFromError(error: Error) {
    43→    return { hasError: true, error };
    44→  }
    45→  componentDidCatch(error: Error, info: React.ErrorInfo) {
    46→    console.error(`[SectionErrorBoundary${this.props.name ? ` (${this.props.name})` : ''}]`, error, info.componentStack);
    47→  }
    48→  render() {
    49→    if (this.state.hasError) {
    50→      if (this.props.fallback) return this.props.fallback;
    51→      return (
    52→        <div className="flex min-h-[300px] items-center justify-center rounded-lg border border-dashed border-[var(--ws-glass-border)] p-6">
    53→          <div className="text-center">
    54→            <AlertCircle size={24} className="mx-auto mb-3 text-[var(--ws-accent)]" />
    55→            <p className="text-sm font-medium text-[var(--ws-text-primary)]">Erro ao carregar esta secao</p>
    56→            <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{this.state.error?.message || 'Erro desconhecido'}</p>
    57→            <button
    58→              onClick={() => { this.setState({ hasError: false, error: null }); }}
    59→              className="mt-3 text-xs font-medium text-[var(--ws-accent)] hover:underline"
    60→            >
    61→              Tentar novamente
    62→            </button>
    63→          </div>
    64→        </div>
    65→      );
    66→    }
    67→    return this.props.children;
    68→  }

    72→function SafeEditor({ content, onChange, placeholder, onError }: { content: string; onChange: (html: string) => void; placeholder: string; onError?: (msg: string) => void }) {
    73→  const [editorReady, setEditorReady] = useState(false);
    74→  const [editorFailed, setEditorFailed] = useState(false);
    75→  const [EditorComp, setEditorComp] = useState<React.ComponentType<{ content: string; onChange: (html: string) => void; placeholder: string }> | null>(null);
    76→  const saveTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    77→
    78→  useEffect(() => {
    79→    import('./RichTextEditor')
    80→      .then(m => { setEditorComp(() => m.RichTextEditor); setEditorReady(true); })

    83→        setEditorFailed(true);
    84→        onError?.(err?.message || 'Editor falhou ao carregar');
    85→      });
    86→  }, []);
    87→
    88→  const handleChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => {
    89→    const val = e.target.value;
    90→    if (saveTimer.current) clearTimeout(saveTimer.current);
    91→    saveTimer.current = setTimeout(() => onChange(val), 500);
    92→  }, [onChange]);
    93→
    94→  if (editorFailed) {
    95→    return (
    96→      <div>
    97→        <div className="mb-3 flex items-center gap-2 rounded-lg bg-[var(--ws-accent)]/10 px-3 py-2 text-xs text-[var(--ws-accent)]">
    98→          <AlertCircle size={14} /> Editor avancado indisponivel. Usando editor simples.
    99→        </div>
   100→        <textarea
   101→          value={content}
   102→          onChange={handleChange}
   103→          placeholder={placeholder}
   104→          className="min-h-[400px] w-full resize-none bg-transparent p-5 text-sm leading-relaxed text-[var(--ws-text-primary)] placeholder-[var(--ws-text-tertiary)] outline-none"
   105→        />
   106→      </div>
   107→    );
   108→  }
   109→
   110→  if (!editorReady) {
   111→    return (
   112→      <div className="flex min-h-[400px] items-center justify-center">
   113→        <div className="text-center">
   114→          <div className="mx-auto mb-3 h-6 w-6 animate-spin rounded-full border-2 border-[var(--ws-glass-border)] border-t-[var(--ws-accent)]" />
   115→          <p className="text-xs text-[var(--ws-text-tertiary)]">Carregando editor...</p>
   116→        </div>
   117→      </div>
   118→    );
   119→  }
   120→
   121→  if (EditorComp) return <EditorComp content={content} onChange={onChange} placeholder={placeholder} />;
   122→  return null;
   123→}

   159→// ========== SWIPE HOOK ==========
   160→function useSwipeGestures(onSwipeLeft: () => void, onSwipeRight: () => void) {
   161→  const touchStartX = useRef(0);
   162→  const touchEndX = useRef(0);
   163→
   164→  const onTouchStart = useCallback((e: React.TouchEvent) => {
   165→    touchStartX.current = e.changedTouches[0].screenX;
   166→  }, []);
   167→
   168→  const onTouchEnd = useCallback((e: React.TouchEvent) => {
   169→    touchEndX.current = e.changedTouches[0].screenX;
   170→    const diff = touchEndX.current - touchStartX.current;
   171→    if (Math.abs(diff) > 80) {
   172→      if (diff < 0) onSwipeLeft();
   173→      else onSwipeRight();
   174→    }
   175→  }, [onSwipeLeft, onSwipeRight]);
   176→
   177→  return { onTouchStart, onTouchEnd };
   178→}

   916→    if (fetchedRef.current) return;

   924→    const card = cards[currentIndex];

  1055→// ========== POMODORO TIMER ==========
  1056→function PomodoroTimer() {
  1057→  const [mode, setMode] = useState<'focus' | 'short' | 'long'>('focus');
  1058→  const [timeLeft, setTimeLeft] = useState(25 * 60);
  1059→  const [isRunning, setIsRunning] = useState(false);
  1060→  const [totalToday, setTotalToday] = useState(0);
  1061→  const [sessions, setSessions] = useState<{ duration: number; createdAt: string }[]>([]);
  1062→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
  1063→  const durations = { focus: 25 * 60, short: 5 * 60, long: 15 * 60 };
  1064→  const modeLabels = { focus: 'Foco', short: 'Pausa Curta', long: 'Pausa Longa' };
  1065→  const fetchedRef = useRef(false);
  1066→
  1067→  useEffect(() => {
  1068→    if (fetchedRef.current) return;
  1069→    fetchedRef.current = true;
  1070→    fetch('/api/sessions').then(r => r.json()).then(d => {
  1071→      if (d.sessions) {
  1072→        const today = new Date().toDateString();
  1073→        setTotalToday(d.sessions.filter((s: any) => new Date(s.createdAt).toDateString() === today).reduce((sum: number, s: any) => sum + s.duration, 0));
  1074→        setSessions(d.sessions.slice(0, 10));
  1075→      }
  1076→    });
  1077→  }, []);
  1078→
  1079→  useEffect(() => {
  1080→    if (isRunning) {
  1081→      intervalRef.current = setInterval(() => {
  1082→        setTimeLeft(prev => {
  1083→          if (prev <= 1) {
  1084→            setIsRunning(false);
  1085→            if (mode === 'focus') {
  1086→              const dur = durations.focus - prev;
  1087→              fetch('/api/sessions', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ duration: dur, type: 'pomodoro' }) });
  1088→              setTotalToday(t => t + dur);
  1089→            }
  1090→            return 0;
  1091→          }
  1092→          return prev - 1;
  1093→        });
  1094→      }, 1000);
  1095→    } else if (intervalRef.current) clearInterval(intervalRef.current);
  1096→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
  1097→  }, [isRunning, mode, durations.focus]);
  1098→
  1099→  const switchMode = (m: 'focus' | 'short' | 'long') => { setMode(m); setTimeLeft(durations[m]); setIsRunning(false); };
  1100→  const minutes = Math.floor(timeLeft / 60);
  1101→  const seconds = timeLeft % 60;
  1102→  const progress = 1 - timeLeft / durations[mode];
  1103→
  1104→  return (
  1105→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
  1106→      <div className="mb-8">
  1107→        <h1 className="font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)] lg:text-3xl"><Timer size={24} className="mr-2 inline text-[var(--ws-accent)]" strokeWidth={1.5} />Pomodoro Timer</h1>
  1108→        <p className="mt-1 text-sm text-[var(--ws-text-tertiary)]">Estude com foco usando a tecnica Pomodoro</p>
  1109→      </div>
  1110→      <div className="mx-auto max-w-md">
  1111→        <div className="mb-8 flex justify-center gap-2">
  1112→          {(['focus', 'short', 'long'] as const).map(m => (
  1113→            <button key={m} onClick={() => switchMode(m)} className={`rounded-ws-button px-4 py-2 text-sm font-medium transition-colors ${mode === m ? 'bg-[var(--ws-accent)] text-[var(--ws-text-on-dark)]' : 'border border-[var(--ws-glass-border)] text-[var(--ws-text-tertiary)] hover:text-[var(--ws-text-secondary)]'}`}>
  1114→              {modeLabels[m]}
  1115→            </button>
  1116→          ))}
  1117→        </div>
  1118→        <WabiSabiCard className="mb-8" hover={false}>
  1119→          <div className="flex flex-col items-center py-8">
  1120→            <div className="relative flex h-56 w-56 items-center justify-center">
  1121→              <svg className="absolute inset-0 -rotate-90" viewBox="0 0 200 200">
  1122→                <circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-glass-border)" strokeWidth="4" />
  1123→                <motion.circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-accent)" strokeWidth="4" strokeLinecap="round" strokeDasharray={2 * Math.PI * 90} initial={{ strokeDashoffset: 2 * Math.PI * 90 }} animate={{ strokeDashoffset: 2 * Math.PI * 90 * (1 - progress) }} transition={{ duration: 0.5 }} />
  1124→              </svg>
  1125→              <div className="text-center">
  1126→                <p className="font-serif-jp text-5xl font-bold text-[var(--ws-text-primary)]">{String(minutes).padStart(2, '0')}:{String(seconds).padStart(2, '0')}</p>
  1127→                <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{modeLabels[mode]}</p>
  1128→              </div>
  1129→            </div>
  1130→            <div className="mt-8 flex gap-4">
  1131→              <button onClick={() => switchMode(mode)} className="rounded-ws-button p-3 text-[var(--ws-text-tertiary)] transition-colors hover:bg-[color-mix(in_srgb,var(--ws-ink)_5%,transparent)] hover:text-[var(--ws-text-primary)]" title="Reiniciar"><RotateCcw size={18} /></button>
  1132→              <ZenButton variant="primary" size="lg" onClick={() => setIsRunning(!isRunning)} className="px-12">{isRunning ? 'Pausar' : 'Iniciar'}</ZenButton>
  1133→            </div>
  1134→          </div>
  1135→        </WabiSabiCard>
  1136→        <WabiSabiCard hover={false}>
  1137→          <h3 className="mb-3 text-sm font-semibold text-[var(--ws-text-secondary)]"><Clock size={14} className="mr-1.5 inline" /> Hoje</h3>
  1138→          <p className="font-serif-jp text-3xl font-bold text-[var(--ws-text-primary)]">{Math.round(totalToday / 60)} min</p>
  1139→          <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">de tempo de foco</p>
  1140→          {sessions.length > 0 && (
  1141→            <div className="mt-4 max-h-48 space-y-2 overflow-y-auto border-t border-[var(--ws-glass-border)] pt-4">
  1142→              {sessions.map((s, i) => (
  1143→                <div key={i} className="flex items-center justify-between text-xs">
  1144→                  <span className="text-[var(--ws-text-tertiary)]">{new Date(s.createdAt).toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' })}</span>
  1145→                  <span className="font-medium text-[var(--ws-text-primary)]">{Math.round(s.duration / 60)} min</span>
  1146→                </div>
  1147→              ))}
  1148→            </div>
  1149→          )}
  1150→        </WabiSabiCard>
  1151→      </div>
  1152→    </motion.div>
