    24→// Lightweight error boundary for individual sections
    25→class SectionErrorBoundary extends Component<{ children: ReactNode; fallback?: ReactNode; name?: string }, { hasError: boolean; error: Error | null }> {
    26→  constructor(props: { children: ReactNode; fallback?: ReactNode; name?: string }) {
    27→    super(props);
    28→    this.state = { hasError: false, error: null };
    29→  }
    30→  static getDerivedStateFromError(error: Error) {
    31→    return { hasError: true, error };
    32→  }
    33→  componentDidCatch(error: Error, info: React.ErrorInfo) {
    34→    console.error(`[SectionErrorBoundary${this.props.name ? ` (${this.props.name})` : ''}]`, error, info.componentStack);
    35→  }
    36→  render() {
    37→    if (this.state.hasError) {
    38→      if (this.props.fallback) return this.props.fallback;
    39→      return (
    40→        <div className="flex min-h-[300px] items-center justify-center rounded-lg border border-dashed border-[var(--ws-glass-border)] p-6">
    41→          <div className="text-center">
    42→            <AlertCircle size={24} className="mx-auto mb-3 text-[var(--ws-accent)]" />
    43→            <p className="text-sm font-medium text-[var(--ws-text-primary)]">Erro ao carregar esta secao</p>
    44→            <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{this.state.error?.message || 'Erro desconhecido'}</p>
    45→            <button
    46→              onClick={() => { this.setState({ hasError: false, error: null }); }}
    47→              className="mt-3 text-xs font-medium text-[var(--ws-accent)] hover:underline"
    48→            >
    49→              Tentar novamente
    50→            </button>
    51→          </div>
    52→        </div>
    53→      );
    54→    }
    55→    return this.props.children;
    56→  }
    57→}

    59→// Safe rich text editor with fallback to plain textarea
    60→function SafeEditor({ content, onChange, placeholder, onError }: { content: string; onChange: (html: string) => void; placeholder: string; onError?: (msg: string) => void }) {
    61→  const [editorReady, setEditorReady] = useState(false);
    62→  const [editorFailed, setEditorFailed] = useState(false);
    63→  const [EditorComp, setEditorComp] = useState<React.ComponentType<{ content: string; onChange: (html: string) => void; placeholder: string }> | null>(null);
    64→  const saveTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    65→
    66→  useEffect(() => {
    67→    import('./RichTextEditor')
    68→      .then(m => { setEditorComp(() => m.RichTextEditor); setEditorReady(true); })
    69→      .catch((err) => {
    70→        console.warn('[SafeEditor] RichTextEditor failed to load, using fallback:', err);
    71→        setEditorFailed(true);
    72→        onError?.(err?.message || 'Editor falhou ao carregar');
    73→      });
    74→  }, []);
    75→
    76→  const handleChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => {
    77→    const val = e.target.value;
    78→    if (saveTimer.current) clearTimeout(saveTimer.current);
    79→    saveTimer.current = setTimeout(() => onChange(val), 500);
    80→  }, [onChange]);
    81→
    82→  if (editorFailed) {
    83→    return (
    84→      <div>
    85→        <div className="mb-3 flex items-center gap-2 rounded-lg bg-[var(--ws-accent)]/10 px-3 py-2 text-xs text-[var(--ws-accent)]">
    86→          <AlertCircle size={14} /> Editor avancado indisponivel. Usando editor simples.
    87→        </div>
    88→        <textarea
    89→          value={content}
    90→          onChange={handleChange}
    91→          placeholder={placeholder}
    92→          className="min-h-[400px] w-full resize-none bg-transparent p-5 text-sm leading-relaxed text-[var(--ws-text-primary)] placeholder-[var(--ws-text-tertiary)] outline-none"
    93→        />
    94→      </div>
    95→    );
    96→  }
    97→
    98→  if (!editorReady) {
    99→    return (
   100→      <div className="flex min-h-[400px] items-center justify-center">
   101→        <div className="text-center">
   102→          <div className="mx-auto mb-3 h-6 w-6 animate-spin rounded-full border-2 border-[var(--ws-glass-border)] border-t-[var(--ws-accent)]" />
   103→          <p className="text-xs text-[var(--ws-text-tertiary)]">Carregando editor...</p>
   104→        </div>
   105→      </div>
   106→    );
   107→  }
   108→
   109→  if (EditorComp) return <EditorComp content={content} onChange={onChange} placeholder={placeholder} />;
   110→  return null;
   111→}

   135→function useActiveUsers() {
   136→  const [count, setCount] = useState(0);
   137→  useEffect(() => {
   138→    const base = 87;
   139→    const tick = () => setCount(base + Math.floor(Math.random() * 30));
   140→    tick();
   141→    const id = setInterval(tick, 5000);
   142→    return () => clearInterval(id);
   143→  }, []);
   144→  return count;
   145→}

   147→// ========== SWIPE HOOK ==========
   148→function useSwipeGestures(onSwipeLeft: () => void, onSwipeRight: () => void) {
   149→  const touchStartX = useRef(0);
   150→  const touchEndX = useRef(0);
   151→
   152→  const onTouchStart = useCallback((e: React.TouchEvent) => {
   153→    touchStartX.current = e.changedTouches[0].screenX;
   154→  }, []);
   155→
   156→  const onTouchEnd = useCallback((e: React.TouchEvent) => {
   157→    touchEndX.current = e.changedTouches[0].screenX;
   158→    const diff = touchEndX.current - touchStartX.current;
   159→    if (Math.abs(diff) > 80) {
   160→      if (diff < 0) onSwipeLeft();
   161→      else onSwipeRight();
   162→    }
   163→  }, [onSwipeLeft, onSwipeRight]);
   164→
   165→  return { onTouchStart, onTouchEnd };
   166→}

   301→function formatReviewDate(dateStr: string): string {
   302→  const d = new Date(dateStr);
   303→  const now = new Date();
   304→  const diffMs = d.getTime() - now.getTime();
   305→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   306→
   307→  if (diffDays < 0) return 'Atrasado';
   308→  if (diffDays === 0) return 'Hoje';
   309→  if (diffDays === 1) return 'Amanha';
   310→  if (diffDays <= 7) return `em ${diffDays} dias`;
   311→  return d.toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit' });
   312→}

  1094→// ========== FLASHCARD REVIEWER ==========
  1095→function FlashcardReviewer({ onBack }: { onBack: () => void }) {
  1096→  const [cards, setCards] = useState<FlashcardItem[]>([]);
  1097→  const [loading, setLoading] = useState(true);
  1098→  const [currentIndex, setCurrentIndex] = useState(0);
  1099→  const [flipped, setFlipped] = useState(false);
  1100→  const [done, setDone] = useState(false);
  1101→  const [reviewed, setReviewed] = useState(0);
  1102→  const [ratings, setRatings] = useState<number[]>([]);
  1103→  const fetchedRef = useRef(false);
  1104→
  1105→  useEffect(() => {
  1106→    if (fetchedRef.current) return;
  1107→    fetchedRef.current = true;
  1108→    fetch('/api/flashcards?due=true').then(r => r.json()).then(d => {
  1109→      if (d.flashcards) setCards(d.flashcards);
  1110→    }).catch(() => {}).finally(() => setLoading(false));
  1111→  }, []);
  1112→
  1113→  const handleReview = useCallback(async (quality: number) => {
  1114→    const card = cards[currentIndex];
  1115→    if (!card) return;
  1116→    try {
  1117→      await fetch(`/api/flashcards/${card.id}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ review: quality }) });
  1118→    } catch {}
  1119→    setReviewed(prev => prev + 1);
  1120→    setRatings(prev => [...prev, quality]);
  1121→    setFlipped(false);
  1122→    if (currentIndex + 1 >= cards.length) setDone(true);
  1123→    else setCurrentIndex(prev => prev + 1);
  1124→  }, [cards, currentIndex]);
  1125→
  1126→  // Keyboard shortcuts
  1127→  useEffect(() => {
  1128→    if (done || !flipped || cards.length === 0) return;
  1129→    const handleKey = (e: KeyboardEvent) => {
  1130→      if (e.key === '1') handleReview(1);
  1131→      else if (e.key === '2') handleReview(2);
  1132→      else if (e.key === '3') handleReview(3);
  1133→      else if (e.key === '4') handleReview(5);
  1134→    };
  1135→    window.addEventListener('keydown', handleKey);
  1136→    return () => window.removeEventListener('keydown', handleKey);
  1137→  }, [flipped, done, cards.length, handleReview]);
  1138→
  1139→  // Completion stats

  1246→function PomodoroTimer() {
  1247→  const [mode, setMode] = useState<'focus' | 'short' | 'long'>('focus');
  1248→  const [timeLeft, setTimeLeft] = useState(25 * 60);
  1249→  const [isRunning, setIsRunning] = useState(false);
  1250→  const [totalToday, setTotalToday] = useState(0);
  1251→  const [sessions, setSessions] = useState<{ duration: number; createdAt: string }[]>([]);
  1252→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
  1253→  const durations = { focus: 25 * 60, short: 5 * 60, long: 15 * 60 };
  1254→  const modeLabels = { focus: 'Foco', short: 'Pausa Curta', long: 'Pausa Longa' };
  1255→  const fetchedRef = useRef(false);
  1256→
  1257→  useEffect(() => {
  1258→    if (fetchedRef.current) return;
  1259→    fetchedRef.current = true;
  1260→    fetch('/api/sessions').then(r => r.json()).then(d => {
  1261→      if (d.sessions) {
  1262→        const today = new Date().toDateString();
  1263→        setTotalToday(d.sessions.filter((s: any) => new Date(s.createdAt).toDateString() === today).reduce((sum: number, s: any) => sum + s.duration, 0));
  1264→        setSessions(d.sessions.slice(0, 10));
  1265→      }
  1266→    });
  1267→  }, []);
  1268→
  1269→  useEffect(() => {
  1270→    if (isRunning) {
  1271→      intervalRef.current = setInterval(() => {
  1272→        setTimeLeft(prev => {
  1273→          if (prev <= 1) {
  1274→            setIsRunning(false);
  1275→            if (mode === 'focus') {
  1276→              const dur = durations.focus - prev;
  1277→              fetch('/api/sessions', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ duration: dur, type: 'pomodoro' }) });
  1278→              setTotalToday(t => t + dur);
  1279→            }
  1280→            return 0;
  1281→          }
  1282→          return prev - 1;
  1283→        });
  1284→      }, 1000);
  1285→    } else if (intervalRef.current) clearInterval(intervalRef.current);
  1286→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
  1287→  }, [isRunning, mode, durations.focus]);
  1288→
  1289→  const switchMode = (m: 'focus' | 'short' | 'long') => { setMode(m); setTimeLeft(durations[m]); setIsRunning(false); };
  1290→  const minutes = Math.floor(timeLeft / 60);
  1291→  const seconds = timeLeft % 60;
  1292→  const progress = 1 - timeLeft / durations[mode];
  1293→
  1294→  return (
  1295→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
  1296→      <div className="mb-8">
  1297→        <h1 className="font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)] lg:text-3xl"><Timer size={24} className="mr-2 inline text-[var(--ws-accent)]" strokeWidth={1.5} />Pomodoro Timer</h1>
  1298→        <p className="mt-1 text-sm text-[var(--ws-text-tertiary)]">Estude com foco usando a tecnica Pomodoro</p>
  1299→      </div>
  1300→      <div className="mx-auto max-w-md">
  1301→        <div className="mb-8 flex justify-center gap-2">
  1302→          {(['focus', 'short', 'long'] as const).map(m => (
  1303→            <button key={m} onClick={() => switchMode(m)} className={`rounded-ws-button px-4 py-2 text-sm font-medium transition-colors ${mode === m ? 'bg-[var(--ws-accent)] text-[var(--ws-text-on-dark)]' : 'border border-[var(--ws-glass-border)] text-[var(--ws-text-tertiary)] hover:text-[var(--ws-text-secondary)]'}`}>
  1304→              {modeLabels[m]}
  1305→            </button>
  1306→          ))}
  1307→        </div>
  1308→        <WabiSabiCard className="mb-8" hover={false}>
  1309→          <div className="flex flex-col items-center py-8">
  1310→            <div className="relative flex h-56 w-56 items-center justify-center">
  1311→              <svg className="absolute inset-0 -rotate-90" viewBox="0 0 200 200">

  1313→                <motion.circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-accent)" strokeWidth="4" strokeLinecap="round" strokeDasharray={2 * Math.PI * 90} initial={{ strokeDashoffset: 2 * Math.PI * 90 }} animate={{ strokeDashoffset: 2 * Math.PI * 90 * (1 - progress) }} transition={{ duration: 0.5 }} />
  1314→              </svg>
  1315→              <div className="text-center">
  1316→                <p className="font-serif-jp text-5xl font-bold text-[var(--ws-text-primary)]">{String(minutes).padStart(2, '0')}:{String(seconds).padStart(2, '0')}</p>

  1325→        </WabiSabiCard>
  1326→        <WabiSabiCard hover={false}>
  1327→          <h3 className="mb-3 text-sm font-semibold text-[var(--ws-text-secondary)]"><Clock size={14} className="mr-1.5 inline" /> Hoje</h3>
