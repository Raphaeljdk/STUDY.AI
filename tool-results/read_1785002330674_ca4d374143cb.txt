    20→// Lightweight error boundary for individual sections
    21→class SectionErrorBoundary extends Component<{ children: ReactNode; fallback?: ReactNode; name?: string }, { hasError: boolean; error: Error | null }> {
    22→  constructor(props: { children: ReactNode; fallback?: ReactNode; name?: string }) {
    23→    super(props);
    24→    this.state = { hasError: false, error: null };
    25→  }
    26→  static getDerivedStateFromError(error: Error) {
    27→    return { hasError: true, error };
    28→  }
    29→  componentDidCatch(error: Error, info: React.ErrorInfo) {
    30→    console.error(`[SectionErrorBoundary${this.props.name ? ` (${this.props.name})` : ''}]`, error, info.componentStack);
    31→  }
    32→  render() {
    33→    if (this.state.hasError) {
    34→      if (this.props.fallback) return this.props.fallback;
    35→      return (
    36→        <div className="flex min-h-[300px] items-center justify-center rounded-lg border border-dashed border-[var(--ws-glass-border)] p-6">
    37→          <div className="text-center">
    38→            <AlertCircle size={24} className="mx-auto mb-3 text-[var(--ws-accent)]" />
    39→            <p className="text-sm font-medium text-[var(--ws-text-primary)]">Erro ao carregar esta secao</p>
    40→            <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{this.state.error?.message || 'Erro desconhecido'}</p>
    41→            <button
    42→              onClick={() => { this.setState({ hasError: false, error: null }); }}
    43→              className="mt-3 text-xs font-medium text-[var(--ws-accent)] hover:underline"
    44→            >
    45→              Tentar novamente
    46→            </button>
    47→          </div>
    48→        </div>
    49→      );
    50→    }
    51→    return this.props.children;
    52→  }
    53→}

    55→// Safe rich text editor with fallback to plain textarea
    56→function SafeEditor({ content, onChange, placeholder, onError }: { content: string; onChange: (html: string) => void; placeholder: string; onError?: (msg: string) => void }) {
    57→  const [editorReady, setEditorReady] = useState(false);
    58→  const [editorFailed, setEditorFailed] = useState(false);
    59→  const [EditorComp, setEditorComp] = useState<React.ComponentType<{ content: string; onChange: (html: string) => void; placeholder: string }> | null>(null);
    60→  const saveTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    61→
    62→  useEffect(() => {
    63→    import('./RichTextEditor')
    64→      .then(m => { setEditorComp(() => m.RichTextEditor); setEditorReady(true); })
    65→      .catch((err) => {
    66→        console.warn('[SafeEditor] RichTextEditor failed to load, using fallback:', err);
    67→        setEditorFailed(true);
    68→        onError?.(err?.message || 'Editor falhou ao carregar');
    69→      });
    70→  }, []);
    71→
    72→  const handleChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => {
    73→    const val = e.target.value;
    74→    if (saveTimer.current) clearTimeout(saveTimer.current);
    75→    saveTimer.current = setTimeout(() => onChange(val), 500);
    76→  }, [onChange]);
    77→
    78→  if (editorFailed) {
    79→    return (
    80→      <div>
    81→        <div className="mb-3 flex items-center gap-2 rounded-lg bg-[var(--ws-accent)]/10 px-3 py-2 text-xs text-[var(--ws-accent)]">
    82→          <AlertCircle size={14} /> Editor avancado indisponivel. Usando editor simples.
    83→        </div>
    84→        <textarea
    85→          value={content}
    86→          onChange={handleChange}
    87→          placeholder={placeholder}
    88→          className="min-h-[400px] w-full resize-none bg-transparent p-5 text-sm leading-relaxed text-[var(--ws-text-primary)] placeholder-[var(--ws-text-tertiary)] outline-none"
    89→        />
    90→      </div>
    91→    );
    92→  }
    93→
    94→  if (!editorReady) {
    95→    return (
    96→      <div className="flex min-h-[400px] items-center justify-center">
    97→        <div className="text-center">
    98→          <div className="mx-auto mb-3 h-6 w-6 animate-spin rounded-full border-2 border-[var(--ws-glass-border)] border-t-[var(--ws-accent)]" />
    99→          <p className="text-xs text-[var(--ws-text-tertiary)]">Carregando editor...</p>
   100→        </div>
   101→      </div>
   102→    );
   103→  }
   104→
   105→  if (EditorComp) return <EditorComp content={content} onChange={onChange} placeholder={placeholder} />;
   106→  return null;
   107→}

   205→// ========== HELPER: Time-based greeting ==========
   206→function getGreeting(): string {
   207→  const h = new Date().getHours();
   208→  if (h < 12) return 'Bom dia';
   209→  if (h < 18) return 'Boa tarde';
   210→  return 'Boa noite';
   211→}

   213→function getGreetingSubtext(): string {
   214→  const h = new Date().getHours();
   215→  if (h < 12) return 'Comece o dia com foco e determinacao. Cada estudo e um passo adiante.';
   216→  if (h < 18) return 'A tarde e perfeita para revisar e consolidar o conhecimento.';
   217→  return 'Boas sessoes noturnas fixam melhor o que voce aprendeu.';
   218→}

   220→function formatReviewDate(dateStr: string): string {
   221→  const d = new Date(dateStr);
   222→  const now = new Date();
   223→  const diffMs = d.getTime() - now.getTime();
   224→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   225→
   226→  if (diffDays < 0) return 'Atrasado';
   227→  if (diffDays === 0) return 'Hoje';
   228→  if (diffDays === 1) return 'Amanha';
   229→  if (diffDays <= 7) return `em ${diffDays} dias`;
   230→  return d.toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit' });
   231→}

   549→function NotebookEditor({ notebookId, onBack }: { notebookId: string; onBack: () => void }) {

   551→  const [loading, setLoading] = useState(true);

   555→  const [editTitle, setEditTitle] = useState(false);
   556→  const [showFcForm, setShowFcForm] = useState(false);
   557→  const [fcFront, setFcFront] = useState('');
   558→  const [fcBack, setFcBack] = useState('');
   559→  const [creatingFc, setCreatingFc] = useState(false);
   560→  const [generating, setGenerating] = useState(false);
   561→  const [editorError, setEditorError] = useState<string | null>(null);
   562→  const saveTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   563→  const fetchedRef = useRef(false);
   564→
   565→  useEffect(() => {
   566→    if (fetchedRef.current) return;
   567→    fetchedRef.current = true;
   568→    fetch(`/api/notebooks/${notebookId}`)
   569→      .then(r => r.json())
   570→      .then(d => {
   571→        if (d.notebook) { setNotebook(d.notebook); setTitle(d.notebook.title); setContent(d.notebook.content || ''); }
   572→        else { toast({ title: 'Caderno nao encontrado', description: 'O caderno solicitado nao existe.', variant: 'destructive' }); }
   573→      })
   574→      .catch(err => {
   575→        console.error('[NotebookEditor] fetch error:', err);
   576→        toast({ title: 'Erro ao carregar', description: 'Nao foi possivel carregar o caderno.', variant: 'destructive' });
   577→      })
   578→      .finally(() => setLoading(false));
   579→  }, [notebookId]);
   580→
   581→  const saveContent = useCallback(async (newContent: string, newTitle?: string) => {
   582→    setSaving(true);
   583→    try {
   584→      const data: any = { content: newContent };
   585→      if (newTitle !== undefined) data.title = newTitle;
   586→      const res = await fetch(`/api/notebooks/${notebookId}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) });
   587→      const d = await res.json();
   588→      if (d.notebook) setNotebook(d.notebook);
   589→    } catch (err) {
   590→      console.error('[NotebookEditor] save error:', err);
   591→    }
   592→    setSaving(false);
   593→  }, [notebookId]);
   594→

   596→    setContent(html);
   597→    if (saveTimerRef.current) clearTimeout(saveTimerRef.current);

   602→    if (title.trim()) { saveContent(content, title.trim()); setEditTitle(false); }

   606→    e.preventDefault();
   607→    if (!fcFront.trim() || !fcBack.trim()) return;
   608→    setCreatingFc(true);

   610→      const res = await fetch('/api/flashcards', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ front: fcFront, back: fcBack, notebookId }) });
   611→      const data = await res.json();

   613→        setNotebook(prev => prev ? { ...prev, flashcards: [data.flashcard, ...(prev.flashcards || [])] } : prev);
   614→        setFcFront(''); setFcBack(''); setShowFcForm(false);
   615→      } else {
   616→        toast({ title: 'Erro ao criar flashcard', description: data.error || 'Tente novamente.', variant: 'destructive' });
   617→      }
   618→    } catch (err) {
   619→      console.error('[NotebookEditor] create fc error:', err);
   620→      toast({ title: 'Erro de conexao', description: 'Nao foi possivel criar o flashcard.', variant: 'destructive' });
   621→    }
   622→    setCreatingFc(false);

   627→      await fetch(`/api/flashcards/${fcId}`, { method: 'DELETE' });

   629→    } catch (err) {
   630→      console.error('[NotebookEditor] delete fc error:', err);
   631→    }
   632→  };
   633→

   645→        for (const fc of data.flashcards) {

   652→        }

   687→        </div>

   946→function FlashcardReviewer({ onBack }: { onBack: () => void }) {
   947→  const [cards, setCards] = useState<FlashcardItem[]>([]);
   948→  const [loading, setLoading] = useState(true);
   949→  const [currentIndex, setCurrentIndex] = useState(0);
   950→  const [flipped, setFlipped] = useState(false);
   951→  const [done, setDone] = useState(false);
   952→  const [reviewed, setReviewed] = useState(0);
   953→  const [ratings, setRatings] = useState<number[]>([]);
   954→  const fetchedRef = useRef(false);
   955→
   956→  useEffect(() => {
   957→    if (fetchedRef.current) return;
   958→    fetchedRef.current = true;
   959→    fetch('/api/flashcards?due=true').then(r => r.json()).then(d => {
   960→      if (d.flashcards) setCards(d.flashcards);
   961→    }).catch(() => {}).finally(() => setLoading(false));
   962→  }, []);
   963→
   964→  const handleReview = useCallback(async (quality: number) => {
   965→    const card = cards[currentIndex];
   966→    if (!card) return;
   967→    try {
   968→      await fetch(`/api/flashcards/${card.id}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ review: quality }) });
   969→    } catch {}
   970→    setReviewed(prev => prev + 1);
   971→    setRatings(prev => [...prev, quality]);
   972→    setFlipped(false);
   973→    if (currentIndex + 1 >= cards.length) setDone(true);
   974→    else setCurrentIndex(prev => prev + 1);
   975→  }, [cards, currentIndex]);
   976→
   977→  // Keyboard shortcuts
   978→  useEffect(() => {
   979→    if (done || !flipped || cards.length === 0) return;

   981→      if (e.key === '1') handleReview(1);
   982→      else if (e.key === '2') handleReview(2);

   984→      else if (e.key === '4') handleReview(5);

  1012→          <p className="mt-1 text-[10px] text-[var(--ws-text-tertiary)]">Media</p>
  1013→        </WabiSabiCard>
  1014→        <WabiSabiCard hover={false}>
  1015→          <p className="font-serif-jp text-xl font-bold text-[var(--ws-gold)]">{hardCount}</p>
  1016→          <p className="mt-1 text-[10px] text-[var(--ws-text-tertiary)]">Dificeis</p>

  1097→function PomodoroTimer() {
  1098→  const [mode, setMode] = useState<'focus' | 'short' | 'long'>('focus');
  1099→  const [timeLeft, setTimeLeft] = useState(25 * 60);
  1100→  const [isRunning, setIsRunning] = useState(false);
  1101→  const [totalToday, setTotalToday] = useState(0);
  1102→  const [sessions, setSessions] = useState<{ duration: number; createdAt: string }[]>([]);
  1103→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
  1104→  const durations = { focus: 25 * 60, short: 5 * 60, long: 15 * 60 };
  1105→  const modeLabels = { focus: 'Foco', short: 'Pausa Curta', long: 'Pausa Longa' };
  1106→  const fetchedRef = useRef(false);
  1107→
  1108→  useEffect(() => {
  1109→    if (fetchedRef.current) return;
  1110→    fetchedRef.current = true;
  1111→    fetch('/api/sessions').then(r => r.json()).then(d => {
  1112→      if (d.sessions) {
  1113→        const today = new Date().toDateString();
  1114→        setTotalToday(d.sessions.filter((s: any) => new Date(s.createdAt).toDateString() === today).reduce((sum: number, s: any) => sum + s.duration, 0));
  1115→        setSessions(d.sessions.slice(0, 10));
  1116→      }
  1117→    });
  1118→  }, []);
  1119→
  1120→  useEffect(() => {
  1121→    if (isRunning) {
  1122→      intervalRef.current = setInterval(() => {
  1123→        setTimeLeft(prev => {
  1124→          if (prev <= 1) {
  1125→            setIsRunning(false);
  1126→            if (mode === 'focus') {
  1127→              const dur = durations.focus - prev;
  1128→              fetch('/api/sessions', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ duration: dur, type: 'pomodoro' }) });
  1129→              setTotalToday(t => t + dur);
  1130→            }
  1131→            return 0;
  1132→          }
  1133→          return prev - 1;
  1134→        });
  1135→      }, 1000);
  1136→    } else if (intervalRef.current) clearInterval(intervalRef.current);
  1137→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
  1138→  }, [isRunning, mode, durations.focus]);
  1139→
  1140→  const switchMode = (m: 'focus' | 'short' | 'long') => { setMode(m); setTimeLeft(durations[m]); setIsRunning(false); };
  1141→  const minutes = Math.floor(timeLeft / 60);
  1142→  const seconds = timeLeft % 60;
  1143→  const progress = 1 - timeLeft / durations[mode];
  1144→
  1145→  return (
  1146→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }}>
  1147→      <div className="mb-8">
  1148→        <h1 className="font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)] lg:text-3xl"><Timer size={24} className="mr-2 inline text-[var(--ws-accent)]" strokeWidth={1.5} />Pomodoro Timer</h1>
  1149→        <p className="mt-1 text-sm text-[var(--ws-text-tertiary)]">Estude com foco usando a tecnica Pomodoro</p>
  1150→      </div>
  1151→      <div className="mx-auto max-w-md">
  1152→        <div className="mb-8 flex justify-center gap-2">
  1153→          {(['focus', 'short', 'long'] as const).map(m => (
  1154→            <button key={m} onClick={() => switchMode(m)} className={`rounded-ws-button px-4 py-2 text-sm font-medium transition-colors ${mode === m ? 'bg-[var(--ws-accent)] text-[var(--ws-text-on-dark)]' : 'border border-[var(--ws-glass-border)] text-[var(--ws-text-tertiary)] hover:text-[var(--ws-text-secondary)]'}`}>
  1155→              {modeLabels[m]}
  1156→            </button>
  1157→          ))}
  1158→        </div>
  1159→        <WabiSabiCard className="mb-8" hover={false}>
  1160→          <div className="flex flex-col items-center py-8">
  1161→            <div className="relative flex h-56 w-56 items-center justify-center">
  1162→              <svg className="absolute inset-0 -rotate-90" viewBox="0 0 200 200">
  1163→                <circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-glass-border)" strokeWidth="4" />
  1164→                <motion.circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-accent)" strokeWidth="4" strokeLinecap="round" strokeDasharray={2 * Math.PI * 90} animate={{ strokeDashoffset: 2 * Math.PI * 90 * (1 - progress) }} transition={{ duration: 0.5 }} />
  1165→              </svg>
  1166→              <div className="text-center">
  1167→                <p className="font-serif-jp text-5xl font-bold text-[var(--ws-text-primary)]">{String(minutes).padStart(2, '0')}:{String(seconds).padStart(2, '0')}</p>
  1168→                <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{modeLabels[mode]}</p>
  1169→              </div>
  1170→            </div>
  1171→            <div className="mt-8 flex gap-4">
  1172→              <button onClick={() => switchMode(mode)} className="rounded-ws-button p-3 text-[var(--ws-text-tertiary)] transition-colors hover:bg-[color-mix(in_srgb,var(--ws-ink)_5%,transparent)] hover:text-[var(--ws-text-primary)]" title="Reiniciar"><RotateCcw size={18} /></button>
  1173→              <ZenButton variant="primary" size="lg" onClick={() => setIsRunning(!isRunning)} className="px-12">{isRunning ? 'Pausar' : 'Iniciar'}</ZenButton>
  1174→            </div>
  1175→          </div>
  1176→        </WabiSabiCard>
  1177→        <WabiSabiCard hover={false}>
  1178→          <h3 className="mb-3 text-sm font-semibold text-[var(--ws-text-secondary)]"><Clock size={14} className="mr-1.5 inline" /> Hoje</h3>
  1179→          <p className="font-serif-jp text-3xl font-bold text-[var(--ws-text-primary)]">{Math.round(totalToday / 60)} min</p>
  1180→          <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">de tempo de foco</p>

  1182→            <div className="mt-4 max-h-48 space-y-2 overflow-y-auto border-t border-[var(--ws-glass-border)] pt-4">
  1183→              {sessions.map((s, i) => (

  1187→                </div>
  1188→              ))}
  1189→            </div>
  1190→          )}
  1191→        </WabiSabiCard>
