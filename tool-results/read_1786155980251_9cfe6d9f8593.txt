   288→n              className="rounded-full px-3 py-1 text-xs font-semibold"

   378→                        {step.name}
   379→                      </h3>

   383→                    </div>
   384→                  </div>
   385→
   386→                  <AnimatePresence>

   400→                            className="mt-3 rounded-ws-button"

   404→                            <Check className="mr-1.5 h-3.5 w-3.5" /> Marcar como concluido
   405→                          </Button>
   406→                        )}
   407→                      </motion.div>
   408→                    )}
   409→                  </AnimatePresence>
   410→                </div>
   411→              </motion.div>
   412→            );
   413→          })}
   414→        </div>
   415→      </motion.div>
   416→    );
   417→  }
   418→
   419→  // ===== LIST VIEW =====
   420→  return (
   421→    <motion.div
   422→      className="mx-auto max-w-2xl space-y-6 px-4 pb-8 pt-4"
   423→      variants={containerVariants}
   424→      initial="hidden"
   425→      animate="show"
   426→    >
   427→      {/* Header */}
   428→      <motion.div variants={itemVariants} className="text-center">
   429→        <div className="mb-2 inline-flex items-center gap-2 rounded-full border border-[var(--ws-glass-border)] bg-[var(--ws-glass)] px-4 py-1.5 backdrop-blur-xl">
   430→          <Route className="h-4 w-4" style={{ color: 'var(--ws-accent)' }} />
   431→          <span className="text-xs font-medium" style={{ color: 'var(--ws-text-secondary)' }}>Trilhas de Aprendizagem</span>
   432→        </div>
   433→        <h1 className="font-serif-jp text-3xl font-bold" style={{ color: 'var(--ws-text-primary)' }}>
   434→          Roadmaps
   435→        </h1>
   436→        <p className="mt-1 text-sm" style={{ color: 'var(--ws-text-tertiary)' }}>

   438→        </p>

   440→
   441→      {/* Generate button */}
   442→      <motion.div variants={itemVariants}>
   443→        <Button
   444→          onClick={() => setGenerateOpen(true)}
   445→          className="w-full rounded-ws-button py-6 text-base font-semibold"
   446→          style={{ background: 'var(--ws-accent)', color: 'var(--ws-text-on-dark)' }}
   447→          whileHover={{ scale: 1.01 }}
   448→          whileTap={{ scale: 0.98 }}
   449→        >
   450→          <Sparkles className="mr-2 h-5 w-5" /> Gerar nova trilha
   451→        </Button>
   452→      </motion.div>
   453→
   454→      {/* Suggested roadmaps (when empty) */}
   455→      {roadmaps.length === 0 && !loading && (
   456→        <motion.div variants={itemVariants}>
   457→          <WabiSabiCard hover={false}>
   458→            <div className="mb-4 flex items-center gap-2">
   459→              <Lightbulb className="h-5 w-5" style={{ color: 'var(--ws-gold)' }} />
   460→              <div>
   461→                <h2 className="font-serif-jp text-lg font-semibold" style={{ color: 'var(--ws-text-primary)' }}>Sugestoes populares</h2>
   462→                <p className="text-xs" style={{ color: 'var(--ws-text-tertiary)' }}>Comece com uma trilha pre-estruturada</p>
   463→              </div>
   464→            </div>
   465→
   466→            <div className="space-y-2">
   467→              {SUGGESTED_ROADMAPS.map((s) => (
   468→                <button
   469→                  key={s.topic}
   470→                  className="flex w-full items-center gap-3 rounded-ws-button border border-[var(--ws-glass-border)] p-3 text-left transition-all hover:border-[var(--ws-accent)] hover-lift"
   471→                  style={{ background: 'var(--ws-glass)' }}
   472→                  onClick={() => handleSuggestedRoadmap(s)}
