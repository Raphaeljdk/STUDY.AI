    62→// ===== HELPERS =====
    63→function getGreeting(): string {
    64→  const h = new Date().getHours();
    65→  if (h < 12) return 'Bom dia';
    66→  if (h < 18) return 'Boa tarde';
    67→  return 'Boa noite';
    68→}
    69→
    70→function getGreetingSubtext(): string {
    71→  const h = new Date().getHours();
    72→  if (h < 12) return 'Comece o dia com foco e determinacao. Cada estudo e um passo adiante.';
    73→  if (h < 18) return 'A tarde e perfeita para revisar e consolidar o conhecimento.';
    74→  return 'Boas sessoes noturnas fixam melhor o que voce aprendeu.';
    75→}
    76→
    77→function getLevelName(level: number): string {
    78→  if (level <= 1) return 'Iniciante';
    79→  if (level <= 2) return 'Aprendiz';
    80→  if (level <= 3) return 'Estudante';
    81→  if (level <= 4) return 'Dedicado';
    82→  if (level <= 5) return 'Especialista';
    83→  return 'Mestre';
    84→}
    85→
    86→function getXpForLevel(level: number): number {
    87→  // XP thresholds: 0->100->250->500->850->1300...
    88→  const thresholds = [0, 100, 250, 500, 850, 1300];
    89→  if (level <= 0) return 0;
    90→  if (level >= thresholds.length) {
    91→    // Beyond defined levels: 450 per level
    92→    const lastDefined = thresholds[thresholds.length - 1];
    93→    return lastDefined + (level - (thresholds.length - 1)) * 450;
    94→  }
    95→  return thresholds[level];
    96→}
    97→
    98→function getPriorityIndicator(priority: string) {
    99→  switch (priority?.toUpperCase()) {
   100→    case 'ALTA': return { dot: 'bg-red-500', label: 'Alta', emoji: '🔴' };
   101→    case 'MEDIA': return { dot: 'bg-amber-500', label: 'Media', emoji: '🟡' };
   102→    case 'BAIXA': return { dot: 'bg-green-500', label: 'Baixa', emoji: '🟢' };
   103→    default: return { dot: 'bg-gray-400', label: priority || 'Media', emoji: '🟡' };
   104→  }
   105→}

   110→    case 'DELIVERY': return 'Entrega';
   111→    case 'EVENT': return 'Evento';
   112→    case 'STUDY': return 'Sessao';
