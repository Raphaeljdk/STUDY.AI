   388→function formatReviewDate(dateStr: string): string {
   389→  const d = new Date(dateStr);
   390→  const now = new Date();
   391→  const diffMs = d.getTime() - now.getTime();
   392→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   393→
   394→  if (diffDays < 0) return 'Atrasado';
   395→  if (diffDays === 0) return 'Hoje';
   396→  if (diffDays === 1) return 'Amanha';
   397→  if (diffDays <= 7) return `em ${diffDays} dias`;
   398→  return d.toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit' });
   399→}

   979→// ========== FLASHCARD REVIEWER ==========
   980→function FlashcardReviewer({ onBack }: { onBack: () => void }) {
   981→  const [cards, setCards] = useState<FlashcardItem[]>([]);
   982→  const [loading, setLoading] = useState(true);
   983→  const [currentIndex, setCurrentIndex] = useState(0);
   984→  const [flipped, setFlipped] = useState(false);
   985→  const [done, setDone] = useState(false);
   986→  const [reviewed, setReviewed] = useState(0);
   987→  const [ratings, setRatings] = useState<number[]>([]);
   988→  const fetchedRef = useRef(false);
   989→
   990→  useEffect(() => {
   991→    if (fetchedRef.current) return;
   992→    fetchedRef.current = true;
   993→    fetch('/api/flashcards?due=true').then(r => r.json()).then(d => {
   994→      if (d.flashcards) setCards(d.flashcards);
   995→    }).catch(() => {}).finally(() => setLoading(false));
   996→  }, []);
   997→
   998→  const handleReview = useCallback(async (quality: number) => {
   999→    const card = cards[currentIndex];
  1000→    if (!card) return;
  1001→    try {
  1002→      await fetch(`/api/flashcards/${card.id}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ review: quality }) });
  1003→    } catch {}
  1004→    setReviewed(prev => prev + 1);
  1005→    setRatings(prev => [...prev, quality]);
  1006→    setFlipped(false);
  1007→    if (currentIndex + 1 >= cards.length) setDone(true);
  1008→    else setCurrentIndex(prev => prev + 1);
  1009→  }, [cards, currentIndex]);
  1010→
  1011→  // Keyboard shortcuts
  1012→  useEffect(() => {
  1013→    if (done || !flipped || cards.length === 0) return;
  1014→    const handleKey = (e: KeyboardEvent) => {
  1015→      if (e.key === '1') handleReview(1);
  1016→      else if (e.key === '2') handleReview(2);
  1017→      else if (e.key === '3') handleReview(3);
  1018→      else if (e.key === '4') handleReview(5);
  1019→    };
  1020→    window.addEventListener('keydown', handleKey);
  1021→    return () => window.removeEventListener('keydown', handleKey);
  1022→  }, [flipped, done, cards.length, handleReview]);

  1034→      <button onClick={onBack} className="mt-4 text-sm text-[var(--ws-accent)] hover:underline">Voltar</button>
  1035→    </motion.div>
  1036→  );
  1037→  if (done) return (
  1038→    <motion.div initial={{ opacity: 0, scale: 0.95 }} animate={{ opacity: 1, scale: 1 }} className="mx-auto max-w-md py-16 text-center">
  1039→      <EnsoCircle size={80} strokeWidth={2} color="var(--ws-verdigris)" imperfection={0.1} animate={false} />

  1041→      <p className="mt-2 text-sm text-[var(--ws-text-tertiary)]">Voce revisou {reviewed} cartas</p>

  1043→      <div className="mt-8 grid grid-cols-3 gap-4">
  1044→        <WabiSabiCard hover={false}>
  1045→          <p className="font-serif-jp text-xl font-bold text-[var(--ws-accent)]">{avgRating}</p>
  1046→          <p className="mt-1 text-[10px] text-[var(--ws-text-tertiary)]">Media</p>
  1047→        </WabiSabiCard>
  1048→        <WabiSabiCard hover={false}>
  1049→          <p className="font-serif-jp text-xl font-bold text-[var(--ws-gold)]">{hardCount}</p>
  1050→          <p className="mt-1 text-[10px] text-[var(--ws-text-tertiary)]">Dificeis</p>
  1051→        </WabiSabiCard>
  1052→        <WabiSabiCard hover={false}>
  1053→          <p className="font-serif-jp text-xl font-bold text-[var(--ws-verdigris)]">{easyCount}</p>

  1061→
  1062→  const card = cards[currentIndex];
  1063→  return (
  1064→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
  1065→      {/* Progress header */}
  1066→      <div className="mb-6">
  1067→        <div className="mb-4 flex items-center justify-between">
  1068→          <button onClick={onBack} className="flex items-center gap-2 text-sm text-[var(--ws-text-tertiary)] hover:text-[var(--ws-text-primary)]">
  1069→            <ArrowLeft size={16} /> Voltar

  1072→            <span className="font-serif-jp text-sm font-bold text-[var(--ws-text-primary)]">{currentIndex + 1} <span className="font-normal text-[var(--ws-text-tertiary)]">de {cards.length}</span></span>

  1131→function PomodoroTimer() {
  1132→  const [mode, setMode] = useState<'focus' | 'short' | 'long'>('focus');
  1133→  const [timeLeft, setTimeLeft] = useState(25 * 60);
  1134→  const [isRunning, setIsRunning] = useState(false);
  1135→  const [totalToday, setTotalToday] = useState(0);
  1136→  const [sessions, setSessions] = useState<{ duration: number; createdAt: string }[]>([]);
  1137→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
  1138→  const durations = { focus: 25 * 60, short: 5 * 60, long: 15 * 60 };
  1139→  const modeLabels = { focus: 'Foco', short: 'Pausa Curta', long: 'Pausa Longa' };
  1140→  const fetchedRef = useRef(false);
  1141→
  1142→  useEffect(() => {
  1143→    if (fetchedRef.current) return;
  1144→    fetchedRef.current = true;
  1145→    fetch('/api/sessions').then(r => r.json()).then(d => {
  1146→      if (d.sessions) {
  1147→        const today = new Date().toDateString();
  1148→        setTotalToday(d.sessions.filter((s: any) => new Date(s.createdAt).toDateString() === today).reduce((sum: number, s: any) => sum + s.duration, 0));
  1149→        setSessions(d.sessions.slice(0, 10));
  1150→      }
  1151→    });
  1152→  }, []);
  1153→
  1154→  useEffect(() => {
  1155→    if (isRunning) {
  1156→      intervalRef.current = setInterval(() => {
  1157→        setTimeLeft(prev => {
  1158→          if (prev <= 1) {
  1159→            setIsRunning(false);
  1160→            if (mode === 'focus') {
  1161→              const dur = durations.focus - prev;
  1162→              fetch('/api/sessions', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ duration: dur, type: 'pomodoro' }) });
  1163→              setTotalToday(t => t + dur);
  1164→            }
  1165→            return 0;
  1166→          }
  1167→          return prev - 1;
  1168→        });
  1169→      }, 1000);
  1170→    } else if (intervalRef.current) clearInterval(intervalRef.current);
  1171→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
  1172→  }, [isRunning, mode, durations.focus]);
  1173→
  1174→  const switchMode = (m: 'focus' | 'short' | 'long') => { setMode(m); setTimeLeft(durations[m]); setIsRunning(false); };
  1175→  const minutes = Math.floor(timeLeft / 60);
  1176→  const seconds = timeLeft % 60;
  1177→  const progress = 1 - timeLeft / durations[mode];
  1178→
  1179→  return (
  1180→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
  1181→      <div className="mb-8">
  1182→        <h1 className="font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)] lg:text-3xl"><Timer size={24} className="mr-2 inline text-[var(--ws-accent)]" strokeWidth={1.5} />Pomodoro Timer</h1>
  1183→        <p className="mt-1 text-sm text-[var(--ws-text-tertiary)]">Estude com foco usando a tecnica Pomodoro</p>
  1184→      </div>
  1185→      <div className="mx-auto max-w-md">
  1186→        <div className="mb-8 flex justify-center gap-2">
  1187→          {(['focus', 'short', 'long'] as const).map(m => (
  1188→            <button key={m} onClick={() => switchMode(m)} className={`rounded-ws-button px-4 py-2 text-sm font-medium transition-colors ${mode === m ? 'bg-[var(--ws-accent)] text-[var(--ws-text-on-dark)]' : 'border border-[var(--ws-glass-border)] text-[var(--ws-text-tertiary)] hover:text-[var(--ws-text-secondary)]'}`}>
  1189→              {modeLabels[m]}
  1190→            </button>
  1191→          ))}
  1192→        </div>
  1193→        <WabiSabiCard className="mb-8" hover={false}>
  1194→          <div className="flex flex-col items-center py-8">
  1195→            <div className="relative flex h-56 w-56 items-center justify-center">
  1196→              <svg className="absolute inset-0 -rotate-90" viewBox="0 0 200 200">
  1197→                <circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-glass-border)" strokeWidth="4" />
  1198→                <motion.circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-accent)" strokeWidth="4" strokeLinecap="round" strokeDasharray={2 * Math.PI * 90} initial={{ strokeDashoffset: 2 * Math.PI * 90 }} animate={{ strokeDashoffset: 2 * Math.PI * 90 * (1 - progress) }} transition={{ duration: 0.5 }} />
  1199→              </svg>
  1200→              <div className="text-center">
  1201→                <p className="font-serif-jp text-5xl font-bold text-[var(--ws-text-primary)]">{String(minutes).padStart(2, '0')}:{String(seconds).padStart(2, '0')}</p>
  1202→                <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{modeLabels[mode]}</p>
  1203→              </div>
  1204→            </div>
  1205→            <div className="mt-8 flex gap-4">
  1206→              <button onClick={() => switchMode(mode)} className="rounded-ws-button p-3 text-[var(--ws-text-tertiary)] transition-colors hover:bg-[color-mix(in_srgb,var(--ws-ink)_5%,transparent)] hover:text-[var(--ws-text-primary)]" title="Reiniciar"><RotateCcw size={18} /></button>
  1207→              <ZenButton variant="primary" size="lg" onClick={() => setIsRunning(!isRunning)} className="px-12">{isRunning ? 'Pausar' : 'Iniciar'}</ZenButton>
  1208→            </div>
  1209→          </div>
  1210→        </WabiSabiCard>
  1211→        <WabiSabiCard hover={false}>
  1212→          <h3 className="mb-3 text-sm font-semibold text-[var(--ws-text-secondary)]"><Clock size={14} className="mr-1.5 inline" /> Hoje</h3>
  1213→          <p className="font-serif-jp text-3xl font-bold text-[var(--ws-text-primary)]">{Math.round(totalToday / 60)} min</p>
  1214→          <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">de tempo de foco</p>
  1215→          {sessions.length > 0 && (
  1216→            <div className="mt-4 max-h-48 space-y-2 overflow-y-auto border-t border-[var(--ws-glass-border)] pt-4">

  1221→                </div>
  1222→              ))}
