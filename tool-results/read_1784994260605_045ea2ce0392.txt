   799→// ========== FLASHCARD REVIEWER ==========
   800→function FlashcardReviewer({ onBack }: { onBack: () => void }) {
   801→  const [cards, setCards] = useState<FlashcardItem[]>([]);
   802→  const [loading, setLoading] = useState(true);
   803→  const [currentIndex, setCurrentIndex] = useState(0);
   804→  const [flipped, setFlipped] = useState(false);
   805→  const [done, setDone] = useState(false);
   806→  const [reviewed, setReviewed] = useState(0);
   807→  const [ratings, setRatings] = useState<number[]>([]);
   808→  const fetchedRef = useRef(false);
   809→
   810→  useEffect(() => {
   811→    if (fetchedRef.current) return;
   812→    fetchedRef.current = true;
   813→    fetch('/api/flashcards?due=true').then(r => r.json()).then(d => {
   814→      if (d.flashcards) setCards(d.flashcards);
   815→    }).catch(() => {}).finally(() => setLoading(false));
   816→  }, []);
   817→
   818→  const handleReview = useCallback(async (quality: number) => {
   819→    const card = cards[currentIndex];
   820→    if (!card) return;
   821→    try {
   822→      await fetch(`/api/flashcards/${card.id}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ review: quality }) });
   823→    } catch {}
   824→    setReviewed(prev => prev + 1);
   825→    setRatings(prev => [...prev, quality]);
   826→    setFlipped(false);
   827→    if (currentIndex + 1 >= cards.length) setDone(true);
   828→    else setCurrentIndex(prev => prev + 1);
   829→  }, [cards, currentIndex]);
   830→
   831→  // Keyboard shortcuts
   832→  useEffect(() => {
   833→    if (done || !flipped || cards.length === 0) return;
   834→    const handleKey = (e: KeyboardEvent) => {
   835→      if (e.key === '1') handleReview(1);
   836→      else if (e.key === '2') handleReview(2);
   837→      else if (e.key === '3') handleReview(3);
   838→      else if (e.key === '4') handleReview(5);
   839→    };
   840→    window.addEventListener('keydown', handleKey);
   841→    return () => window.removeEventListener('keydown', handleKey);
   842→  }, [flipped, done, cards.length, handleReview]);
   843→
   844→  // Completion stats
   845→  const avgRating = ratings.length > 0 ? (ratings.reduce((a, b) => a + b, 0) / ratings.length).toFixed(1) : '0';
   846→  const hardCount = ratings.filter(r => r <= 2).length;
   847→  const easyCount = ratings.filter(r => r >= 4).length;
   848→
   849→  if (loading) return <div className="flex items-center justify-center py-16"><Loader2 size={24} className="animate-spin text-[var(--ws-text-tertiary)]" /></div>;
   850→  if (cards.length === 0) return (
   851→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="py-16 text-center">
   852→      <Check size={48} className="mx-auto mb-4 text-[var(--ws-verdigris)]" strokeWidth={1} />
   853→      <p className="text-sm text-[var(--ws-text-tertiary)]">Nenhuma carta para revisar agora</p>

   951→function PomodoroTimer() {
   952→  const [mode, setMode] = useState<'focus' | 'short' | 'long'>('focus');
   953→  const [timeLeft, setTimeLeft] = useState(25 * 60);
   954→  const [isRunning, setIsRunning] = useState(false);
   955→  const [totalToday, setTotalToday] = useState(0);
   956→  const [sessions, setSessions] = useState<{ duration: number; createdAt: string }[]>([]);
   957→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   958→  const durations = { focus: 25 * 60, short: 5 * 60, long: 15 * 60 };
   959→  const modeLabels = { focus: 'Foco', short: 'Pausa Curta', long: 'Pausa Longa' };
   960→  const fetchedRef = useRef(false);
   961→
   962→  useEffect(() => {
   963→    if (fetchedRef.current) return;
   964→    fetchedRef.current = true;
   965→    fetch('/api/sessions').then(r => r.json()).then(d => {
   966→      if (d.sessions) {
   967→        const today = new Date().toDateString();
   968→        setTotalToday(d.sessions.filter((s: any) => new Date(s.createdAt).toDateString() === today).reduce((sum: number, s: any) => sum + s.duration, 0));
   969→        setSessions(d.sessions.slice(0, 10));
   970→      }
   971→    });
   972→  }, []);
   973→
   974→  useEffect(() => {
   975→    if (isRunning) {
   976→      intervalRef.current = setInterval(() => {
   977→        setTimeLeft(prev => {
   978→          if (prev <= 1) {
   979→            setIsRunning(false);
   980→            if (mode === 'focus') {
   981→              const dur = durations.focus - prev;
   982→              fetch('/api/sessions', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ duration: dur, type: 'pomodoro' }) });
   983→              setTotalToday(t => t + dur);
   984→            }
   985→            return 0;
   986→          }
   987→          return prev - 1;
   988→        });
   989→      }, 1000);
   990→    } else if (intervalRef.current) clearInterval(intervalRef.current);
   991→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
   992→  }, [isRunning, mode, durations.focus]);
   993→
   994→  const switchMode = (m: 'focus' | 'short' | 'long') => { setMode(m); setTimeLeft(durations[m]); setIsRunning(false); };
   995→  const minutes = Math.floor(timeLeft / 60);
   996→  const seconds = timeLeft % 60;
   997→  const progress = 1 - timeLeft / durations[mode];
   998→
   999→  return (
  1000→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }}>
  1001→      <div className="mb-8">
  1002→        <h1 className="font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)] lg:text-3xl"><Timer size={24} className="mr-2 inline text-[var(--ws-accent)]" strokeWidth={1.5} />Pomodoro Timer</h1>
  1003→        <p className="mt-1 text-sm text-[var(--ws-text-tertiary)]">Estude com foco usando a tecnica Pomodoro</p>
  1004→      </div>
  1005→      <div className="mx-auto max-w-md">
  1006→        <div className="mb-8 flex justify-center gap-2">
  1007→          {(['focus', 'short', 'long'] as const).map(m => (
  1008→            <button key={m} onClick={() => switchMode(m)} className={`rounded-ws-button px-4 py-2 text-sm font-medium transition-colors ${mode === m ? 'bg-[var(--ws-accent)] text-[var(--ws-text-on-dark)]' : 'border border-[var(--ws-glass-border)] text-[var(--ws-text-tertiary)] hover:text-[var(--ws-text-secondary)]'}`}>
  1009→              {modeLabels[m]}
  1010→            </button>
  1011→          ))}
  1012→        </div>
  1013→        <WabiSabiCard className="mb-8" hover={false}>
  1014→          <div className="flex flex-col items-center py-8">
  1015→            <div className="relative flex h-56 w-56 items-center justify-center">
  1016→              <svg className="absolute inset-0 -rotate-90" viewBox="0 0 200 200">
  1017→                <circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-glass-border)" strokeWidth="4" />
  1018→                <motion.circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-accent)" strokeWidth="4" strokeLinecap="round" strokeDasharray={2 * Math.PI * 90} animate={{ strokeDashoffset: 2 * Math.PI * 90 * (1 - progress) }} transition={{ duration: 0.5 }} />
  1019→              </svg>
  1020→              <div className="text-center">
  1021→                <p className="font-serif-jp text-5xl font-bold text-[var(--ws-text-primary)]">{String(minutes).padStart(2, '0')}:{String(seconds).padStart(2, '0')}</p>
  1022→                <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{modeLabels[mode]}</p>
  1023→              </div>
  1024→            </div>
  1025→            <div className="mt-8 flex gap-4">
  1026→              <button onClick={() => switchMode(mode)} className="rounded-ws-button p-3 text-[var(--ws-text-tertiary)] transition-colors hover:bg-[color-mix(in_srgb,var(--ws-ink)_5%,transparent)] hover:text-[var(--ws-text-primary)]" title="Reiniciar"><RotateCcw size={18} /></button>
  1027→              <ZenButton variant="primary" size="lg" onClick={() => setIsRunning(!isRunning)} className="px-12">{isRunning ? 'Pausar' : 'Iniciar'}</ZenButton>
  1028→            </div>
  1029→          </div>
  1030→        </WabiSabiCard>
  1031→        <WabiSabiCard hover={false}>
  1032→          <h3 className="mb-3 text-sm font-semibold text-[var(--ws-text-secondary)]"><Clock size={14} className="mr-1.5 inline" /> Hoje</h3>
  1033→          <p className="font-serif-jp text-3xl font-bold text-[var(--ws-text-primary)]">{Math.round(totalToday / 60)} min</p>
