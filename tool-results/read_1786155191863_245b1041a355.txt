    74→// ===== CIRCULAR TIMER =====
    75→function CircularTimer({ seconds, maxSeconds }: { seconds: number; maxSeconds: number }) {
    76→  const radius = 42;
    77→  const circumference = 2 * Math.PI * radius;
    78→  const progress = seconds / maxSeconds;
    79→  const offset = circumference * (1 - progress);
    80→  const isLow = seconds <= 4;
    81→
    82→  return (
    83→    <div className="relative flex items-center justify-center">
    84→      <svg width="100" height="100" className="-rotate-90">
    85→        <circle
    86→          cx="50" cy="50" r={radius}
    87→          fill="none"
    88→          stroke="var(--ws-glass-border)"
    89→          strokeWidth="5"
    90→        />
    91→        <circle
    92→          cx="50" cy="50" r={radius}
    93→          fill="none"
    94→          stroke={isLow ? '#DC2626' : 'var(--ws-accent)'}
    95→          strokeWidth="5"
    96→          strokeLinecap="round"
    97→          strokeDasharray={circumference}
    98→          strokeDashoffset={offset}
    99→          className="transition-all duration-1000 ease-linear"
   100→        />
   101→      </svg>
   102→      <div className="absolute flex flex-col items-center">
   103→        <span
   104→          className="text-2xl font-bold tabular-nums"
   105→          style={{
   106→            color: isLow ? '#DC2626' : 'var(--ws-text-primary)',
   107→          }}
   108→        >
   109→          {seconds}
   110→        </span>
   111→        <span className="text-[10px]" style={{ color: 'var(--ws-text-tertiary)' }}>
   112→          seg
   113→        </span>
   114→      </div>
   115→    </div>
   116→  );
   117→}

   276→// ===== BATTLE SCREEN =====
   277→function BattleScreen({
   278→  battle,
   279→  onFinish,
   280→}: {
   281→  battle: BattleState;
   282→  onFinish: (answers: { questionId: string; selectedIndex: number; confidence: number; correct: boolean }[]) => void;
   283→}) {
   284→  const [currentIndex, setCurrentIndex] = useState(0);
   285→  const [timer, setTimer] = useState(TIMER_DURATION);
   286→  const [selectedOption, setSelectedOption] = useState<number | null>(null);
   287→  const [showConfidence, setShowConfidence] = useState(false);
   288→  const [answers, setAnswers] = useState<(boolean | null)[]>(
   289→    Array(battle.questions.length).fill(null)
   290→  );
   291→  const [answeredOptions, setAnsweredOptions] = useState<Record<number, number>>({});
   292→  const [results, setResults] = useState<
   293→    { questionId: string; selectedIndex: number; confidence: number; correct: boolean }[]
   294→  >([]);
   295→  const [showResult, setShowResult] = useState(false);
   296→  const [correctFlash, setCorrectFlash] = useState<boolean | null>(null);
   297→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   298→
   299→  const currentQuestion = battle.questions[currentIndex];
   300→
   301→  // Timer
   302→  useEffect(() => {
   303→    if (showConfidence || showResult) return;
   304→    timerRef.current = setInterval(() => {
   305→      setTimer((prev) => {
   306→        if (prev <= 1) {
   307→          // Time's up - auto wrong answer
   308→          handleTimeout();
   309→          return 0;
   310→        }
   311→        return prev - 1;
   312→      });
   313→    }, 1000);
   314→    return () => {
   315→      if (timerRef.current) clearInterval(timerRef.current);
   316→    };
   317→  }, [currentIndex, showConfidence, showResult]);
   318→
   319→  const handleTimeout = useCallback(() => {
   320→    if (timerRef.current) clearInterval(timerRef.current);
   321→    setAnsweredOptions((prev) => ({ ...prev, [currentIndex]: -1 }));
   322→    setAnswers((prev) => {
   323→      const next = [...prev];
   324→      next[currentIndex] = false;
   325→      return next;
   326→    });
   327→    setCorrectFlash(false);
   328→    setTimeout(() => {
   329→      setCorrectFlash(null);
   330→      setShowConfidence(true);
   331→    }, 600);
   332→  }, [currentIndex]);
   333→
   334→  const handleSelectOption = (optionIndex: number) => {
   335→    if (selectedOption !== null || showConfidence) return;
   336→    if (timerRef.current) clearInterval(timerRef.current);
   337→
   338→    const isCorrect = optionIndex === currentQuestion.correctIndex;
   339→    setSelectedOption(optionIndex);
   340→    setAnsweredOptions((prev) => ({ ...prev, [currentIndex]: optionIndex }));
   341→    setAnswers((prev) => {
   342→      const next = [...prev];
   343→      next[currentIndex] = isCorrect;
   344→      return next;
   345→    });
   346→    setCorrectFlash(isCorrect);
   347→
   348→    setTimeout(() => {
   349→      setCorrectFlash(null);
   350→      setShowConfidence(true);
   351→    }, 600);
   352→  };
   353→
   354→  const handleConfidence = (confidence: number) => {
   355→    const optionIdx = selectedOption ?? -1;
   356→    const isCorrect = optionIdx === currentQuestion.correctIndex;
   357→
   358→    setResults((prev) => [
   359→      ...prev,
   360→      {
   361→        questionId: currentQuestion.id,
   362→        selectedIndex: optionIdx,
   363→        confidence,
   364→        correct: isCorrect,
   365→      },
   366→    ]);
   367→
   368→    setShowConfidence(false);
   369→    setSelectedOption(null);
   370→
   371→    if (currentIndex < battle.questions.length - 1) {
   372→      setCurrentIndex((prev) => prev + 1);
   373→      setTimer(TIMER_DURATION);
   374→    } else {
   375→      setShowResult(true);
   376→    }
   377→  };
   378→
   379→  // Compute result and submit
   380→  useEffect(() => {
   381→    if (showResult && results.length === battle.questions.length) {
   382→      const timer = setTimeout(() => onFinish(results), 500);
   383→      return () => clearTimeout(timer);
   384→    }
   385→  }, [showResult, results, battle.questions.length, onFinish]);
   386→
   387→  return (
   388→    <motion.div
   389→      initial={{ opacity: 0, x: 30 }}
   390→      animate={{ opacity: 1, x: 0 }}
   391→      exit={{ opacity: 0, x: -30 }}
   392→      className="space-y-5"
   393→    >
   394→      {/* Top bar */}
   395→      <div className="flex items-center justify-between">
   396→        <div className="flex items-center gap-2">
   397→          <span className="text-sm font-medium" style={{ color: 'var(--ws-text-secondary)' }}>
   398→            {battle.subject}
   399→          </span>
   400→        </div>
   401→        <div className="flex items-center gap-2">
   402→          <span className="text-xs font-medium px-2 py-1 rounded-full" style={{
   403→            background: 'color-mix(in srgb, var(--ws-accent) 10%, transparent)',
   404→            color: 'var(--ws-accent)',
   405→          }}>
   406→            {currentIndex + 1}/{battle.questions.length}
   407→          </span>
   408→        </div>
   409→      </div>
   410→
   411→      {/* Timer & Progress */}
   412→      <div className="flex flex-col items-center gap-4">
   413→        <CircularTimer seconds={timer} maxSeconds={TIMER_DURATION} />
   414→        <ProgressDots
   415→          current={currentIndex}
   416→          total={battle.questions.length}
   417→          answers={answers}
   418→        />
   419→      </div>
   420→
   421→      {/* Question */}
   422→      <AnimatePresence mode="wait">
   423→        <motion.div
   424→          key={currentIndex}
   425→          initial={{ opacity: 0, y: 10 }}
   426→          animate={{ opacity: 1, y: 0 }}
   427→          exit={{ opacity: 0, y: -10 }}
   428→          transition={{ duration: 0.25 }}
   429→          className="space-y-4"
   430→        >
   431→          {/* Flash overlay */}
   432→          {correctFlash !== null && (
   433→            <motion.div
   434→              initial={{ opacity: 0.3 }}
   435→              animate={{ opacity: 0 }}
   436→              transition={{ duration: 0.6 }}
   437→              className="fixed inset-0 z-50 pointer-events-none"
   438→              style={{
   439→                backgroundColor: correctFlash ? 'rgba(22, 163, 74, 0.15)' : 'rgba(220, 38, 38, 0.15)',
   440→              }}
   441→            />
   442→          )}
   443→
   444→          <div
   445→            className="glass-enhanced p-5"
   446→            style={{ borderRadius: 'var(--ws-radius-card)' }}
   447→          >
   448→            <h2
   449→              className="font-serif-jp text-lg font-semibold leading-relaxed"
   450→              style={{ color: 'var(--ws-text-primary)' }}
   451→            >
   452→              {currentQuestion.question}
   453→            </h2>
   454→          </div>
   455→
   456→          {/* Options */}
   457→          <div className="space-y-2.5">
   458→            {currentQuestion.options.map((option, idx) => {
   459→              const answeredIdx = answeredOptions[currentIndex];
   460→              const isAnswered = answeredIdx !== undefined;
   461→              const isSelected = answeredIdx === idx;
   462→              const isCorrectAnswer = idx === currentQuestion.correctIndex;
   463→              const isTimeout = answeredIdx === -1;
   464→
   465→              let optionBg = 'var(--ws-glass)';
   466→              let optionBorder = 'var(--ws-glass-border)';
   467→              let optionColor = 'var(--ws-text-primary)';
   468→
   469→              if (isAnswered) {
   470→                if (isCorrectAnswer) {
   471→                  optionBg = 'rgba(22, 163, 74, 0.1)';
   472→                  optionBorder = 'rgba(22, 163, 74, 0.3)';
   473→                  optionColor = '#16A34A';
   474→                } else if (isSelected && !isCorrectAnswer) {
   475→                  optionBg = 'rgba(220, 38, 38, 0.1)';
   476→                  optionBorder = 'rgba(220, 38, 38, 0.3)';
   477→                  optionColor = '#DC2626';
   478→                } else {
   479→                  optionBg = 'color-mix(in srgb, var(--ws-glass) 50%, transparent)';
   480→                  optionBorder = 'var(--ws-glass-border)';
   481→                  optionColor = 'var(--ws-text-tertiary)';
   482→                }
   483→              }
   484→
   485→              return (
   486→                <motion.button
   487→                  key={idx}
   488→                  whileTap={!isAnswered ? { scale: 0.98 } : undefined}
   489→                  onClick={() => handleSelectOption(idx)}
   490→                  disabled={isAnswered || showConfidence}
   491→                  className="w-full flex items-center gap-3 p-4 text-left transition-ws"
   492→                  style={{
   493→                    borderRadius: 'var(--ws-radius-button)',
   494→                    background: optionBg,
   495→                    border: `1.5px solid ${optionBorder}`,
   496→                    color: optionColor,
   497→                    opacity: isAnswered && !isCorrectAnswer && !isSelected ? 0.5 : 1,
   498→                  }}
   499→                >
   500→                  <div
   501→                    className="flex-shrink-0 w-8 h-8 rounded-full flex items-center justify-center text-sm font-bold"
   502→                    style={{
   503→                      background: isAnswered && isCorrectAnswer
   504→                        ? 'rgba(22, 163, 74, 0.15)'
   505→                        : isAnswered && isSelected && !isCorrectAnswer
   506→                          ? 'rgba(220, 38, 38, 0.15)'
   507→                          : 'color-mix(in srgb, var(--ws-ink) 6%, transparent)',
   508→                    }}
   509→                  >
   510→                    {isAnswered && isCorrectAnswer ? (
   511→                      <Check className="w-4 h-4" />
   512→                    ) : isAnswered && isSelected && !isCorrectAnswer ? (
   513→                      <X className="w-4 h-4" />
   514→                    ) : (
   515→                      String.fromCharCode(65 + idx)
   516→                    )}
   517→                  </div>
   518→                  <span className="text-sm font-medium leading-snug">
   519→                    {option}
   520→                  </span>
   521→                </motion.button>
   522→              );
   523→            })}
   524→          </div>
   525→        </motion.div>
   526→      </AnimatePresence>
   527→
   528→      {/* Confidence Selector */}
   529→      <AnimatePresence>
   530→        {showConfidence && (
   531→          <motion.div
   532→            initial={{ opacity: 0, y: 20 }}
   533→            animate={{ opacity: 1, y: 0 }}
   534→            exit={{ opacity: 0, y: 20 }}
   535→            className="space-y-3 pt-2"
   536→          >
   537→            <p className="text-center text-sm font-medium" style={{ color: 'var(--ws-text-secondary)' }}>
   538→              Quao confiante voce esta?
   539→            </p>
   540→            <div className="grid grid-cols-4 gap-2">
   541→              {CONFIDENCE_LEVELS.map((level) => (
   542→                <motion.button
   543→                  key={level.value}
   544→                  whileTap={{ scale: 0.95 }}
   545→                  onClick={() => handleConfidence(level.value)}
   546→                  className="flex flex-col items-center gap-1.5 p-3 transition-ws hover-lift"
   547→                  style={{
   548→                    borderRadius: 'var(--ws-radius-button)',
   549→                    background: 'var(--ws-glass)',
   550→                    border: '1px solid var(--ws-glass-border)',
   551→                  }}
   552→                >
   553→                  <span className="text-2xl">{level.emoji}</span>
   554→                  <span className="text-[10px] font-medium text-center leading-tight" style={{ color: 'var(--ws-text-secondary)' }}>
   555→                    {level.label}
   556→                  </span>
   557→                </motion.button>
   558→              ))}
   559→            </div>
   560→          </motion.div>
   561→        )}
   562→      </AnimatePresence>
   563→    </motion.div>
   564→  );
   565→}

   567→// ===== RESULTS SCREEN =====
   568→function ResultsScreen({
   569→  result,
   570→  history,
   571→  onPlayAgain,
   572→  onBack,
   573→}: {
   574→  result: BattleResult;
   575→  history: BattleHistoryItem[];
   576→  onPlayAgain: () => void;
   577→  onBack: () => void;
   578→}) {

   778→type Screen = 'subject' | 'battle' | 'results';
   779→
   780→export function BattleView() {
   781→  const [screen, setScreen] = useState<Screen>('subject');
   782→  const [subject, setSubject] = useState('');
   783→  const [battle, setBattle] = useState<BattleState | null>(null);
   784→  const [result, setResult] = useState<BattleResult | null>(null);
   785→  const [history, setHistory] = useState<BattleHistoryItem[]>([]);
   786→  const [loading, setLoading] = useState(false);
   787→  const [loadingHistory, setLoadingHistory] = useState(true);
   788→
   789→  // Fetch history on mount
   790→  useEffect(() => {
   791→    const fetchHistory = async () => {
   792→      try {
   793→        const res = await fetch('/api/battle');
   794→        if (res.ok) {
   795→          const data = await res.json();
   796→          setHistory(data.battles || data || []);
   797→        }
   798→      } catch {
   799→        // silent
   800→      } finally {
   801→        setLoadingHistory(false);
   802→      }
   803→    };
   804→    fetchHistory();
   805→  }, []);
   806→
   807→  const handleStartBattle = async (selectedSubject: string) => {
   808→    setSubject(selectedSubject);
   809→    setLoading(true);
   810→    try {
   811→      const res = await fetch('/api/battle', {
   812→        method: 'POST',
   813→        headers: { 'Content-Type': 'application/json' },
   814→        body: JSON.stringify({ subject: selectedSubject }),
   815→      });
   816→      if (!res.ok) {
   817→        const err = await res.json().catch(() => ({}));
   818→        throw new Error(err.error || 'Failed to start battle');
   819→      }
   820→      const data = await res.json();
   821→      setBattle(data);
   822→      setScreen('battle');
   823→    } catch (err: any) {
   824→      toast({
   825→        title: 'Erro ao iniciar duelo',
   826→        description: err.message || 'Tente novamente.',
   827→      });
   828→    } finally {
   829→      setLoading(false);
   830→    }
   831→  };
   832→
   833→  const handleFinishBattle = async (
   834→    answers: { questionId: string; selectedIndex: number; confidence: number; correct: boolean }[]
   835→  ) => {
   836→    if (!battle) return;
   837→    try {
   838→      const res = await fetch('/api/battle/finish', {
   839→        method: 'POST',
   840→        headers: { 'Content-Type': 'application/json' },
   841→        body: JSON.stringify({
   842→          battleId: battle.id,
   843→          answers,
   844→        }),
   845→      });
   846→      if (res.ok) {
   847→        const data = await res.json();
   848→        setResult(data);
   849→        // Refresh history
   850→        const histRes = await fetch('/api/battle');
   851→        if (histRes.ok) {
   852→          const histData = await histRes.json();
   853→          setHistory(histData.battles || histData || []);
   854→        }
   855→      }
   856→    } catch {
   857→      // If API fails, compute locally
   858→      const score = answers.filter((a) => a.correct).length;
   859→      const localResult: BattleResult = {
   860→        id: battle.id,
   861→        subject,
   862→        score,
   863→        total: battle.questions.length,
   864→        xpEarned: score * 20,
   865→        accuracy: Math.round((score / battle.questions.length) * 100),
   866→        avgConfidence:
   867→          answers.length > 0
   868→            ? Math.round(answers.reduce((sum, a) => sum + a.confidence, 0) / answers.length * 10) / 10
   869→            : 0,
   870→        completedAt: new Date().toISOString(),
   871→      };
   872→      setResult(localResult);
   873→    }
   874→    setScreen('results');
   875→  };
   876→
   877→  const handlePlayAgain = () => {
   878→    setBattle(null);
   879→    setResult(null);
   880→    setScreen('subject');
   881→  };
   882→
   883→  const handleBackToStart = () => {
   884→    setBattle(null);
   885→    setResult(null);
   886→    setSubject('');
   887→    setScreen('subject');
   888→  };
   889→
   890→  return (
   891→    <div className="relative min-h-full pb-24">
   892→      {/* Header */}
   893→      <div className="mb-6">
   894→        <h1 className="font-serif-jp text-2xl font-bold mb-1" style={{ color: 'var(--ws-text-primary)' }}>
   895→          Duelo de Conhecimento
   896→        </h1>
   897→        <p className="text-sm" style={{ color: 'var(--ws-text-tertiary)' }}>
   898→          Teste seus conhecimentos em 1 minuto
   899→        </p>
   900→      </div>
   901→
   902→      <AnimatePresence mode="wait">
   903→        {screen === 'subject' && (
   904→          <SubjectSelection key="subject" onStart={handleStartBattle} />
   905→        )}
   906→        {screen === 'battle' && battle && (
   907→          <BattleScreen key="battle" battle={battle} onFinish={handleFinishBattle} />
   908→        )}
   909→        {screen === 'results' && result && (
   910→          <ResultsScreen
   911→            key="results"
   912→            result={result}
   913→            history={history}
   914→            onPlayAgain={handlePlayAgain}
   915→            onBack={handleBackToStart}
   916→          />
   917→        )}
   918→      </AnimatePresence>
   919→    </div>
   920→  );
   921→}
