   211→   211→function NotebooksList({ onOpen }: { onOpen: (id: string) => void }) {
   212→   212→  const [notebooks, setNotebooks] = useState<NotebookItem[]>([]);
   213→   213→  const [loading, setLoading] = useState(true);
   214→   214→  const [creating, setCreating] = useState(false);
   215→   215→  const [newTitle, setNewTitle] = useState('');
   216→   216→  const [newColor, setNewColor] = useState(notebookColors[0]);
   217→   217→  const fetchedRef = useRef(false);
   218→   218→
   219→   219→  useEffect(() => {
   220→   220→    if (fetchedRef.current) return; fetchedRef.current = true;
   221→   221→    fetch('/api/notebooks').then(r => r.json()).then(d => { if (d.notebooks) setNotebooks(d.notebooks); }).catch(() => {}).finally(() => setLoading(false));
   222→   222→  }, []);
   223→   223→
   224→   224→  const handleCreate = async (e: React.FormEvent) => {
   225→   225→    e.preventDefault(); if (!newTitle.trim()) return; setCreating(true);
   226→   226→    try {
   227→   227→      const res = await fetch('/api/notebooks', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ title: newTitle, color: newColor }) });
   228→   228→      const data = await res.json();
   229→   229→      if (data.notebook) { setNotebooks(prev => [data.notebook, ...prev]); setNewTitle(''); setCreating(false); onOpen(data.notebook.id); }
   230→   230→    } catch {} setCreating(false);
   231→   231→  };
   232→   232→
   233→   233→  const handleDelete = async (id: string, title: string) => {
   234→   234→    if (!confirm(`Deletar "${title}"?`)) return;
   235→   235→    await fetch(`/api/notebooks/${id}`, { method: 'DELETE' });
   236→   236→    setNotebooks(prev => prev.filter(n => n.id !== id));
   237→   237→  };
   238→   238→
   239→   239→  const stripHtml = (h: string) => h.replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim();
   240→   240→
   241→   241→  return (
   242→   242→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }}>
   243→   243→      <div className="mb-8"><h1 className="font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)] lg:text-3xl">Meus Cadernos</h1><p className="mt-1 text-sm text-[var(--ws-text-tertiary)]">Organize seus estudos por assunto</p></div>
   244→   244→      <WabiSabiCard className="mb-8" hover={false}>
   245→   245→        <form onSubmit={handleCreate} className="flex flex-col gap-4 sm:flex-row sm:items-end">
   246→   246→          <div className="flex-1">
   247→   247→            <label className="mb-1.5 block text-sm font-medium text-[var(--ws-text-secondary)]">Novo Caderno</label>
   248→   248→            <input type="text" value={newTitle} onChange={e => setNewTitle(e.target.value)} placeholder="Ex: Matematica, Historia, Biologia..." className="w-full rounded-ws-button border border-[var(--ws-glass-border)] bg-[var(--ws-bg)] px-4 py-3 text-sm text-[var(--ws-text-primary)] placeholder-[var(--ws-text-tertiary)] outline-none transition-colors focus:border-[var(--ws-accent)]/30" />
   249→   249→          </div>
   250→   250→          <div>
   251→   251→            <label className="mb-1.5 block text-sm font-medium text-[var(--ws-text-secondary)]">Cor</label>
   252→   252→            <div className="flex gap-2">{notebookColors.map(c => (
   253→   253→              <button key={c} type="button" onClick={() => setNewColor(c)} className={`h-8 w-8 rounded-full border-2 transition-transform ${newColor === c ? 'scale-110 border-[var(--ws-text-primary)]' : 'border-transparent'}`} style={{ backgroundColor: c }} />
   254→   254→            ))}</div>
   255→   255→          </div>
   256→   256→          <ZenButton type="submit" variant="primary" size="md" disabled={creating || !newTitle.trim()}>{creating ? <Loader2 size={14} className="animate-spin" /> : <Plus size={14} />} Criar</ZenButton>
   257→   257→        </form>
   258→   258→      </WabiSabiCard>
   259→   259→      {loading ? <div className="flex items-center justify-center py-16"><Loader2 size={24} className="animate-spin text-[var(--ws-text-tertiary)]" /></div> : notebooks.length === 0 ? (
   260→   260→        <div className="py-16 text-center"><FolderOpen size={48} className="mx-auto mb-4 text-[var(--ws-text-tertiary)]" strokeWidth={1} /><p className="text-sm text-[var(--ws-text-tertiary)]">Nenhum caderno criado ainda</p></div>
   261→   261→      ) : (
   262→   262→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   263→   263→          {notebooks.map((nb, i) => (
   264→   264→            <motion.div key={nb.id} initial={{ opacity: 0, y: 15 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: i * 0.05 }}>
   265→   265→              <WabiSabiCard>
   266→   266→                <button onClick={() => onOpen(nb.id)} className="w-full text-left">
   267→   267→                  <div className="flex items-start justify-between">
   268→   268→                    <div className="flex items-center gap-3"><div className="flex h-10 w-10 items-center justify-center rounded-ws-button" style={{ backgroundColor: nb.color + '18' }}><BookOpen size={18} style={{ color: nb.color }} strokeWidth={1.5} /></div><div className="min-w-0"><p className="truncate text-sm font-semibold text-[var(--ws-text-primary)]">{nb.title}</p><p className="text-xs text-[var(--ws-text-tertiary)]">{nb._count?.flashcards || 0} flashcards</p></div></div>
   269→   269→                    <ChevronRight size={16} className="shrink-0 text-[var(--ws-text-tertiary)]" />
   270→   270→                  </div>
   271→   271→                  {nb.content && <p className="mt-3 line-clamp-2 text-xs leading-relaxed text-[var(--ws-text-tertiary)]">{stripHtml(nb.content).substring(0, 120)}</p>}

   521→   521→// ========== POMODORO TIMER ==========
   522→   522→function PomodoroTimer() {
   523→   523→  const [mode, setMode] = useState<'focus' | 'short' | 'long'>('focus');
   524→   524→  const [timeLeft, setTimeLeft] = useState(25 * 60);
   525→   525→  const [isRunning, setIsRunning] = useState(false);
   526→   526→  const [totalToday, setTotalToday] = useState(0);
   527→   527→  const [sessions, setSessions] = useState<{ duration: number; createdAt: string }[]>([]);
   528→   528→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   529→   529→  const durations = { focus: 25 * 60, short: 5 * 60, long: 15 * 60 };
   530→   530→  const modeLabels = { focus: 'Foco', short: 'Pausa Curta', long: 'Pausa Longa' };
   531→   531→  const fetchedRef = useRef(false);
   532→   532→
   533→   533→  useEffect(() => {
   534→   534→    if (fetchedRef.current) return; fetchedRef.current = true;
   535→   535→    fetch('/api/sessions').then(r => r.json()).then(d => {
   536→   536→      if (d.sessions) {
   537→   537→        const today = new Date().toDateString();
   538→   538→        setTotalToday(d.sessions.filter((s: any) => new Date(s.createdAt).toDateString() === today).reduce((sum: number, s: any) => sum + s.duration, 0));
   539→   539→        setSessions(d.sessions.slice(0, 10));
   540→   540→      }
   541→   541→    });
   542→   542→  }, []);
   543→   543→
   544→   544→  useEffect(() => {
   545→   545→    if (isRunning) {
   546→   546→      intervalRef.current = setInterval(() => {
   547→   547→        setTimeLeft(prev => {
   548→   548→          if (prev <= 1) { setIsRunning(false); if (mode === 'focus') { const dur = durations.focus - prev; fetch('/api/sessions', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ duration: dur, type: 'pomodoro' }) }); setTotalToday(t => t + dur); } return 0; }
   549→   549→          return prev - 1;
   550→   550→        });
   551→   551→      }, 1000);
   552→   552→    } else if (intervalRef.current) clearInterval(intervalRef.current);
   553→   553→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
   554→   554→  }, [isRunning, mode, durations.focus]);
   555→   555→
   556→   556→  const switchMode = (m: 'focus' | 'short' | 'long') => { setMode(m); setTimeLeft(durations[m]); setIsRunning(false); };
   557→   557→  const minutes = Math.floor(timeLeft / 60);
   558→   558→  const seconds = timeLeft % 60;
   559→   559→  const progress = 1 - timeLeft / durations[mode];
   560→   560→
   561→   561→  return (
   562→   562→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }}>
   563→   563→      <div className="mb-8"><h1 className="font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)] lg:text-3xl"><Timer size={24} className="mr-2 inline text-[var(--ws-accent)]" strokeWidth={1.5} />Pomodoro Timer</h1><p className="mt-1 text-sm text-[var(--ws-text-tertiary)]">Estude com foco usando a tecnica Pomodoro</p></div>
   564→   564→      <div className="mx-auto max-w-md">
   565→   565→        <div className="mb-8 flex justify-center gap-2">{(['focus', 'short', 'long'] as const).map(m => (<button key={m} onClick={() => switchMode(m)} className={`rounded-ws-button px-4 py-2 text-sm font-medium transition-colors ${mode === m ? 'bg-[var(--ws-accent)] text-[var(--ws-text-on-dark)]' : 'border border-[var(--ws-glass-border)] text-[var(--ws-text-tertiary)] hover:text-[var(--ws-text-secondary)]'}`}>{modeLabels[m]}</button>))}</div>
   566→   566→        <WabiSabiCard className="mb-8" hover={false}>
   567→   567→          <div className="flex flex-col items-center py-8">
   568→   568→            <div className="relative flex h-56 w-56 items-center justify-center">
   569→   569→              <svg className="absolute inset-0 -rotate-90" viewBox="0 0 200 200"><circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-glass-border)" strokeWidth="4" /><motion.circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-accent)" strokeWidth="4" strokeLinecap="round" strokeDasharray={2 * Math.PI * 90} animate={{ strokeDashoffset: 2 * Math.PI * 90 * (1 - progress) }} transition={{ duration: 0.5 }} /></svg>
   570→   570→              <div className="text-center"><p className="font-serif-jp text-5xl font-bold text-[var(--ws-text-primary)]">{String(minutes).padStart(2, '0')}:{String(seconds).padStart(2, '0')}</p><p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{modeLabels[mode]}</p></div>
   571→   571→            </div>
   572→   572→            <div className="mt-8 flex gap-4">
   573→   573→              <button onClick={() => switchMode(mode)} className="rounded-ws-button p-3 text-[var(--ws-text-tertiary)] transition-colors hover:bg-[color-mix(in_srgb,var(--ws-ink)_5%,transparent)] hover:text-[var(--ws-text-primary)]" title="Reiniciar"><RotateCcw size={18} /></button>
   574→   574→              <ZenButton variant="primary" size="lg" onClick={() => setIsRunning(!isRunning)} className="px-12">{isRunning ? 'Pausar' : 'Iniciar'}</ZenButton>
   575→   575→            </div>
   576→   576→          </div>
   577→   577→        </WabiSabiCard>
   578→   578→        <WabiSabiCard hover={false}>
   579→   579→          <h3 className="mb-3 text-sm font-semibold text-[var(--ws-text-secondary)]"><Clock size={14} className="mr-1.5 inline" /> Hoje</h3>
   580→   580→          <p className="font-serif-jp text-3xl font-bold text-[var(--ws-text-primary)]">{Math.round(totalToday / 60)} min</p>
   581→   581→          <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">de tempo de foco</p>
   582→   582→          {sessions.length > 0 && <div className="mt-4 max-h-48 space-y-2 overflow-y-auto border-t border-[var(--ws-glass-border)] pt-4">{sessions.map((s, i) => (<div key={i} className="flex items-center justify-between text-xs"><span className="text-[var(--ws-text-tertiary)]">{new Date(s.createdAt).toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' })}</span><span className="font-medium text-[var(--ws-text-primary)]">{Math.round(s.duration / 60)} min</span></div>))}</div>}
   583→   583→        </WabiSabiCard>
   584→   584→      </div>
   585→   585→    </motion.div>
   586→   586→  );
   587→   587→}
