   116→// ========== HELPER: Time-based greeting ==========
   117→function getGreeting(): string {
   118→  const h = new Date().getHours();
   119→  if (h < 12) return 'Bom dia';
   120→  if (h < 18) return 'Boa tarde';
   121→  return 'Boa noite';
   122→}
   123→
   124→function getGreetingSubtext(): string {
   125→  const h = new Date().getHours();
   126→  if (h < 12) return 'Comece o dia com foco e determinacao. Cada estudo e um passo adiante.';
   127→  if (h < 18) return 'A tarde e perfeita para revisar e consolidar o conhecimento.';
   128→  return 'Boas sessoes noturnas fixam melhor o que voce aprendeu.';
   129→}

   131→function formatReviewDate(dateStr: string): string {
   132→  const d = new Date(dateStr);
   133→  const now = new Date();
   134→  const diffMs = d.getTime() - now.getTime();
   135→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   136→
   137→  if (diffDays < 0) return 'Atrasado';
   138→  if (diffDays === 0) return 'Hoje';
   139→  if (diffDays === 1) return 'Amanha';
   140→  if (diffDays <= 7) return `em ${diffDays} dias`;
   141→  return d.toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit' });
   142→}

   144→function getStatusBadge(repetitions: number): { label: string; color: string; bg: string } {
   145→  if (repetitions === 0) return { label: 'Novo', color: 'var(--ws-accent)', bg: 'color-mix(in srgb, var(--ws-accent) 12%, transparent)' };
   146→  if (repetitions < 5) return { label: 'Revisando', color: 'var(--ws-gold)', bg: 'color-mix(in srgb, var(--ws-gold) 12%, transparent)' };
   147→  return { label: 'Dominado', color: 'var(--ws-verdigris)', bg: 'color-mix(in srgb, var(--ws-verdigris) 12%, transparent)' };
   148→}

   332→// ========== NOTEBOOKS LIST ==========
   333→function NotebooksList({ onOpen }: { onOpen: (id: string) => void }) {
   334→  const [notebooks, setNotebooks] = useState<NotebookItem[]>([]);
   335→  const [loading, setLoading] = useState(true);
   336→  const [creating, setCreating] = useState(false);
   337→  const [newTitle, setNewTitle] = useState('');
   338→  const [newColor, setNewColor] = useState(notebookColors[0]);
   339→  const fetchedRef = useRef(false);
   340→
   341→  useEffect(() => {
   342→    if (fetchedRef.current) return;
   343→    fetchedRef.current = true;
   344→    fetch('/api/notebooks').then(r => r.json()).then(d => {
   345→      if (d.notebooks) setNotebooks(d.notebooks);
   346→    }).catch(() => {}).finally(() => setLoading(false));
   347→  }, []);
   348→
   349→  const handleCreate = async (e: React.FormEvent) => {
   350→    e.preventDefault();
   351→    if (!newTitle.trim()) return;
   352→    setCreating(true);
   353→    try {
   354→      const res = await fetch('/api/notebooks', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ title: newTitle, color: newColor }) });
   355→      const data = await res.json();
   356→      if (data.notebook) {
   357→        setNotebooks(prev => [data.notebook, ...prev]);
   358→        setNewTitle('');
   359→        setCreating(false);
   360→        onOpen(data.notebook.id);
   361→      }
   362→    } catch {} 
   363→    setCreating(false);
   364→  };
   365→
   366→  const handleDelete = async (id: string, title: string) => {
   367→    if (!confirm(`Deletar "${title}"?`)) return;
   368→    await fetch(`/api/notebooks/${id}`, { method: 'DELETE' });
   369→    setNotebooks(prev => prev.filter(n => n.id !== id));
   370→  };
   371→
   372→  const stripHtml = (h: string) => h.replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim();
   373→
   374→  return (
   375→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }}>
   376→      <div className="mb-8">
   377→        <h1 className="font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)] lg:text-3xl">Meus Cadernos</h1>
   378→        <p className="mt-1 text-sm text-[var(--ws-text-tertiary)]">Organize seus estudos por assunto</p>
   379→      </div>
