    19→// Safe rich text editor with fallback to plain textarea
    20→function SafeEditor({ content, onChange, placeholder }: { content: string; onChange: (html: string) => void; placeholder: string }) {
    21→  const [editorReady, setEditorReady] = useState(false);
    22→  const [editorFailed, setEditorFailed] = useState(false);
    23→  const [EditorComp, setEditorComp] = useState<React.ComponentType<{ content: string; onChange: (html: string) => void; placeholder: string }> | null>(null);
    24→  const saveTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    25→
    26→  useEffect(() => {
    27→    import('./RichTextEditor')
    28→      .then(m => { setEditorComp(() => m.RichTextEditor); setEditorReady(true); })
    29→      .catch(() => setEditorFailed(true));
    30→  }, []);
    31→
    32→  const handleChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => {
    33→    const val = e.target.value;
    34→    if (saveTimer.current) clearTimeout(saveTimer.current);
    35→    saveTimer.current = setTimeout(() => onChange(val), 500);
    36→  }, [onChange]);
    37→
    38→  if (editorFailed) {
    39→    return (
    40→      <textarea
    41→        value={content}
    42→        onChange={handleChange}
    43→        placeholder={placeholder}
    44→        className="min-h-[400px] w-full resize-none bg-transparent p-5 text-sm leading-relaxed text-[var(--ws-text-primary)] placeholder-[var(--ws-text-tertiary)] outline-none"
    45→      />
    46→    );
    47→  }
    48→
    49→  if (!editorReady) {
    50→    return (
    51→      <div className="flex min-h-[400px] items-center justify-center">
    52→        <div className="text-center">
    53→          <div className="mx-auto mb-3 h-6 w-6 animate-spin rounded-full border-2 border-[var(--ws-glass-border)] border-t-[var(--ws-accent)]" />
    54→          <p className="text-xs text-[var(--ws-text-tertiary)]">Carregando editor...</p>
    55→        </div>
    56→      </div>
    57→    );
    58→  }
    59→
    60→  if (EditorComp) return <EditorComp content={content} onChange={onChange} placeholder={placeholder} />;
    61→  return null;
    62→}

   160→// ========== HELPER: Time-based greeting ==========
   161→function getGreeting(): string {
   162→  const h = new Date().getHours();
   163→  if (h < 12) return 'Bom dia';
   164→  if (h < 18) return 'Boa tarde';
   165→  return 'Boa noite';
   166→}
   167→
   168→function getGreetingSubtext(): string {
   169→  const h = new Date().getHours();
   170→  if (h < 12) return 'Comece o dia com foco e determinacao. Cada estudo e um passo adiante.';
   171→  if (h < 18) return 'A tarde e perfeita para revisar e consolidar o conhecimento.';
   172→  return 'Boas sessoes noturnas fixam melhor o que voce aprendeu.';
   173→}

   175→function formatReviewDate(dateStr: string): string {
   176→  const d = new Date(dateStr);
   177→  const now = new Date();
   178→  const diffMs = d.getTime() - now.getTime();
   179→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   180→
   181→  if (diffDays < 0) return 'Atrasado';
   182→  if (diffDays === 0) return 'Hoje';
   183→  if (diffDays === 1) return 'Amanha';
   184→  if (diffDays <= 7) return `em ${diffDays} dias`;
   185→  return d.toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit' });
   186→}

   188→function getStatusBadge(repetitions: number): { label: string; color: string; bg: string } {
   189→  if (repetitions === 0) return { label: 'Novo', color: 'var(--ws-accent)', bg: 'color-mix(in srgb, var(--ws-accent) 12%, transparent)' };
   190→  if (repetitions < 5) return { label: 'Revisando', color: 'var(--ws-gold)', bg: 'color-mix(in srgb, var(--ws-gold) 12%, transparent)' };
   191→  return { label: 'Dominado', color: 'var(--ws-verdigris)', bg: 'color-mix(in srgb, var(--ws-verdigris) 12%, transparent)' };
   192→}

   486→  const [loading, setLoading] = useState(true);

   489→  const [title, setTitle] = useState('');
   490→  const [editTitle, setEditTitle] = useState(false);
   491→  const [showFcForm, setShowFcForm] = useState(false);
   492→  const [fcFront, setFcFront] = useState('');
   493→  const [fcBack, setFcBack] = useState('');
   494→  const [creatingFc, setCreatingFc] = useState(false);
   495→  const [generating, setGenerating] = useState(false);
   496→  const saveTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   497→  const fetchedRef = useRef(false);
   498→
   499→  useEffect(() => {
   500→    if (fetchedRef.current) return;
   501→    fetchedRef.current = true;

   503→      if (d.notebook) { setNotebook(d.notebook); setTitle(d.notebook.title); setContent(d.notebook.content); }

   505→  }, [notebookId]);
   506→

   508→    setSaving(true);

   511→      if (newTitle !== undefined) data.title = newTitle;
   512→      const res = await fetch(`/api/notebooks/${notebookId}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) });

   515→    } catch {}
   516→    setSaving(false);
   517→  }, [notebookId]);

   844→function FlashcardReviewer({ onBack }: { onBack: () => void }) {
   845→  const [cards, setCards] = useState<FlashcardItem[]>([]);
   846→  const [loading, setLoading] = useState(true);
   847→  const [currentIndex, setCurrentIndex] = useState(0);
   848→  const [flipped, setFlipped] = useState(false);
   849→  const [done, setDone] = useState(false);
   850→  const [reviewed, setReviewed] = useState(0);
   851→  const [ratings, setRatings] = useState<number[]>([]);
   852→  const fetchedRef = useRef(false);
   853→
   854→  useEffect(() => {
   855→    if (fetchedRef.current) return;
   856→    fetchedRef.current = true;
   857→    fetch('/api/flashcards?due=true').then(r => r.json()).then(d => {
   858→      if (d.flashcards) setCards(d.flashcards);
   859→    }).catch(() => {}).finally(() => setLoading(false));
   860→  }, []);
   861→
   862→  const handleReview = useCallback(async (quality: number) => {
   863→    const card = cards[currentIndex];
   864→    if (!card) return;
   865→    try {
   866→      await fetch(`/api/flashcards/${card.id}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ review: quality }) });
   867→    } catch {}
   868→    setReviewed(prev => prev + 1);
   869→    setRatings(prev => [...prev, quality]);
   870→    setFlipped(false);
   871→    if (currentIndex + 1 >= cards.length) setDone(true);
   872→    else setCurrentIndex(prev => prev + 1);

   880→      else if (e.key === '2') handleReview(2);

   882→      else if (e.key === '4') handleReview(5);

   912→        <WabiSabiCard hover={false}>
   913→          <p className="font-serif-jp text-xl font-bold text-[var(--ws-gold)]">{hardCount}</p>

   994→// ========== POMODORO TIMER ==========
   995→function PomodoroTimer() {
   996→  const [mode, setMode] = useState<'focus' | 'short' | 'long'>('focus');
   997→  const [timeLeft, setTimeLeft] = useState(25 * 60);
   998→  const [isRunning, setIsRunning] = useState(false);
   999→  const [totalToday, setTotalToday] = useState(0);
  1000→  const [sessions, setSessions] = useState<{ duration: number; createdAt: string }[]>([]);
  1001→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
  1002→  const durations = { focus: 25 * 60, short: 5 * 60, long: 15 * 60 };
  1003→  const modeLabels = { focus: 'Foco', short: 'Pausa Curta', long: 'Pausa Longa' };
  1004→  const fetchedRef = useRef(false);
  1005→
  1006→  useEffect(() => {
  1007→    if (fetchedRef.current) return;
  1008→    fetchedRef.current = true;
  1009→    fetch('/api/sessions').then(r => r.json()).then(d => {
  1010→      if (d.sessions) {
  1011→        const today = new Date().toDateString();
  1012→        setTotalToday(d.sessions.filter((s: any) => new Date(s.createdAt).toDateString() === today).reduce((sum: number, s: any) => sum + s.duration, 0));
  1013→        setSessions(d.sessions.slice(0, 10));
  1014→      }
  1015→    });
  1016→  }, []);
  1017→
  1018→  useEffect(() => {
  1019→    if (isRunning) {
  1020→      intervalRef.current = setInterval(() => {
  1021→        setTimeLeft(prev => {
  1022→          if (prev <= 1) {
  1023→            setIsRunning(false);
  1024→            if (mode === 'focus') {
  1025→              const dur = durations.focus - prev;
  1026→              fetch('/api/sessions', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ duration: dur, type: 'pomodoro' }) });
  1027→              setTotalToday(t => t + dur);
  1028→            }
  1029→            return 0;
  1030→          }
  1031→          return prev - 1;
  1032→        });
  1033→      }, 1000);
  1034→    } else if (intervalRef.current) clearInterval(intervalRef.current);
  1035→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
  1036→  }, [isRunning, mode, durations.focus]);
  1037→
  1038→  const switchMode = (m: 'focus' | 'short' | 'long') => { setMode(m); setTimeLeft(durations[m]); setIsRunning(false); };
  1039→  const minutes = Math.floor(timeLeft / 60);
  1040→  const seconds = timeLeft % 60;
  1041→  const progress = 1 - timeLeft / durations[mode];
  1042→
  1043→  return (
  1044→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }}>
  1045→      <div className="mb-8">
  1046→        <h1 className="font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)] lg:text-3xl"><Timer size={24} className="mr-2 inline text-[var(--ws-accent)]" strokeWidth={1.5} />Pomodoro Timer</h1>
  1047→        <p className="mt-1 text-sm text-[var(--ws-text-tertiary)]">Estude com foco usando a tecnica Pomodoro</p>
  1048→      </div>
  1049→      <div className="mx-auto max-w-md">
  1050→        <div className="mb-8 flex justify-center gap-2">
  1051→          {(['focus', 'short', 'long'] as const).map(m => (
  1052→            <button key={m} onClick={() => switchMode(m)} className={`rounded-ws-button px-4 py-2 text-sm font-medium transition-colors ${mode === m ? 'bg-[var(--ws-accent)] text-[var(--ws-text-on-dark)]' : 'border border-[var(--ws-glass-border)] text-[var(--ws-text-tertiary)] hover:text-[var(--ws-text-secondary)]'}`}>
  1053→              {modeLabels[m]}
  1054→            </button>
  1055→          ))}
  1056→        </div>
  1057→        <WabiSabiCard className="mb-8" hover={false}>
  1058→          <div className="flex flex-col items-center py-8">
  1059→            <div className="relative flex h-56 w-56 items-center justify-center">
  1060→              <svg className="absolute inset-0 -rotate-90" viewBox="0 0 200 200">
  1061→                <circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-glass-border)" strokeWidth="4" />
  1062→                <motion.circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-accent)" strokeWidth="4" strokeLinecap="round" strokeDasharray={2 * Math.PI * 90} animate={{ strokeDashoffset: 2 * Math.PI * 90 * (1 - progress) }} transition={{ duration: 0.5 }} />
  1063→              </svg>
  1064→              <div className="text-center">
  1065→                <p className="font-serif-jp text-5xl font-bold text-[var(--ws-text-primary)]">{String(minutes).padStart(2, '0')}:{String(seconds).padStart(2, '0')}</p>
  1066→                <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{modeLabels[mode]}</p>
  1067→              </div>
  1068→            </div>
  1069→            <div className="mt-8 flex gap-4">
  1070→              <button onClick={() => switchMode(mode)} className="rounded-ws-button p-3 text-[var(--ws-text-tertiary)] transition-colors hover:bg-[color-mix(in_srgb,var(--ws-ink)_5%,transparent)] hover:text-[var(--ws-text-primary)]" title="Reiniciar"><RotateCcw size={18} /></button>
  1071→              <ZenButton variant="primary" size="lg" onClick={() => setIsRunning(!isRunning)} className="px-12">{isRunning ? 'Pausar' : 'Iniciar'}</ZenButton>
  1072→            </div>
  1073→          </div>
