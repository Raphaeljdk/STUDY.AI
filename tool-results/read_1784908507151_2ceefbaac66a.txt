   322→tsx
   323→// components/hero/EnsoCircle.tsx
   324→'use client';
   325→
   326→import { motion } from 'framer-motion';
   327→
   328→interface EnsoCircleProps {
   329→  size: number;
   330→  strokeWidth: number;
   331→  color: string;
   332→  imperfection: number; // 0-1, quanto mais, mais imperfeito
   333→  className?: string;
   334→}
   335→
   336→export const EnsoCircle: React.FC<EnsoCircleProps> = ({
   337→  size,
   338→  strokeWidth,
   339→  color,
   340→  imperfection = 0.1,
   341→  className,
   342→}) => {
   343→  // Gera pontos de controle imperfeitos para o círculo
   344→  const generateImperfectPath = (radius: number, imperfection: number) => {
   345→    const points = 8;
   346→    let path = '';
   347→    
   348→    for (let i = 0; i <= points; i++) {
   349→      const angle = (i / points) * Math.PI * 2;
   350→      const imperfectRadius = radius + (Math.random() - 0.5) * radius * imperfection * 2;
   351→      const x = Math.cos(angle) * imperfectRadius;
   352→      const y = Math.sin(angle) * imperfectRadius;
   353→      
   354→      if (i === 0) {
   355→        path += `M ${x} ${y} `;
   356→      } else {
   357→        const cpRadius = imperfectRadius * 1.2;
   358→        const cpAngle = angle - (Math.PI / points);
   359→        const cpX = Math.cos(cpAngle) * cpRadius;
   360→        const cpY = Math.sin(cpAngle) * cpRadius;
   361→        path += `Q ${cpX} ${cpY} ${x} ${y} `;
   362→      }
   363→    }
   364→    
   365→    path += 'Z';
   366→    return path;
   367→  };
   368→
   369→  const radius = size / 2;
   370→  const path = generateImperfectPath(radius, imperfection);
   371→
   372→  return (
   373→    <motion.svg
   374→      width={size}
   375→      height={size}
   376→      viewBox={`-${size/2} -${size/2} ${size} ${size}`}
   377→      className={className}
   378→      initial={{ pathLength: 0, opacity: 0 }}
   379→      animate={{ pathLength: 1, opacity: 1 }}
   380→      transition={{ duration: 2, ease: "easeInOut" }}
   381→    >
   382→      {/* Círculo principal */}
   383→      <motion.path
   384→        d={path}
   385→        stroke={color}
   386→        strokeWidth={strokeWidth}
   387→        fill="none"
   388→        strokeLinecap="round"
   389→        strokeLinejoin="round"
   390→        initial={{ pathLength: 0 }}
   391→        animate={{ pathLength: 1 }}
   392→        transition={{ duration: 2, delay: 0.5 }}
   393→      />
   394→      
   395→      {/* Pincelada adicional (como tinta escorrendo) */}
   396→      <motion.path
   397→        d={path}
   398→        stroke={color}
   399→        strokeWidth={strokeWidth * 0.5}
   400→        fill="none"
   401→        strokeLinecap="round"
   402→        opacity={0.3}
   403→        initial={{ pathLength: 0 }}
   404→        animate={{ pathLength: 1 }}
   405→        transition={{ duration: 1.5, delay: 0.8 }}
   406→        transform="translate(2, 2)"
   407→      />
   408→      
   409→      {/* Brilho interno */}
   410→      <motion.circle
   411→        cx={0}
   412→        cy={0}
   413→        r={radius * 0.95}
   414→        fill="none"
   415→        stroke={`url(#glowGradient)`}
   416→        strokeWidth={1}
   417→        initial={{ opacity: 0 }}
   418→        animate={{ opacity: 0.5 }}
   419→        transition={{ duration: 1, delay: 2 }}
   420→      />
   421→      
   422→      <defs>
   423→        <radialGradient id="glowGradient">
   424→          <stop offset="0%" stopColor={color} stopOpacity="0.1" />
   425→          <stop offset="100%" stopColor={color} stopOpacity="0" />
   426→        </radialGradient>
   427→      </defs>
   428→    </motion.svg>
   429→  );
   430→};
