    26→/* ========================================================================
    27→   SmoothBrush - Caneta estabilizada para letra bonita
    28→   ======================================================================== */
    29→
    30→class SmoothBrush extends PencilBrush {
    31→  private buffer: Array<{ x: number; y: number }> = [];
    32→  private bufferSize = 4;
    33→
    34→  override onPointerDown(pointer: any) {
    35→    this.buffer = [];
    36→    super.onPointerDown(pointer);
    37→  }
    38→
    39→  override onPointerMove(pointer: any) {
    40→    this.buffer.push({ x: pointer.x, y: pointer.y });
    41→    if (this.buffer.length > this.bufferSize) {
    42→      this.buffer.shift();
    43→    }
    44→    if (this.buffer.length > 1) {
    45→      const avg = this.buffer.reduce(
    46→        (acc, p) => ({ x: acc.x + p.x / this.buffer.length, y: acc.y + p.y / this.buffer.length }),
    47→        { x: 0, y: 0 },
    48→      );
    49→      super.onPointerMove(avg);
    50→    } else {
    51→      super.onPointerMove(pointer);
    52→    }
    53→  }
    54→}

    99→/* ========================================================================
   100→   Componente principal
   101→   ======================================================================== */
   102→
   103→export default function CanvasEditor({
   104→  initialData,
   105→  paperStyle: initPaperStyle = 'blank',
   106→  paperColor: initPaperColor = '#ffffff',
   107→  lineColor: initLineColor = '#e0e0e0',
   108→  onChange,
   109→}: CanvasEditorProps) {
   110→  /* ----- refs ----- */
   111→  const canvasElRef = useRef<HTMLCanvasElement>(null);
   112→  const fcRef = useRef<Canvas | null>(null);
   113→  const containerRef = useRef<HTMLDivElement>(null);
   114→  const fileInputRef = useRef<HTMLInputElement>(null);
   115→  const activeToolRef = useRef('select');
   116→
   117→  /* ----- estado da ferramenta ----- */
   118→  const [activeTool, setActiveTool] = useState('select');
   119→  const [strokeColor, setStrokeColor] = useState('#000000');
   120→  const [strokeWidth, setStrokeWidth] = useState(2);
   121→  const [zoom, setZoom] = useState(1);
   122→  const [paperStyle, setPaperStyle] = useState(initPaperStyle);
   123→  const [paperColor, setPaperColor] = useState(initPaperColor);
   124→  const [lineColor] = useState(initLineColor);
   125→  const [tapeMode, setTapeMode] = useState(false);
   126→  const [tapeColor, setTapeColor] = useState(getTapeColors()[0]);
   127→
   128→  /* ----- texto format (Canva-like) ----- */
   129→  const [textFormat, setTextFormat] = useState<TextFormat>({ ...DEFAULT_TEXT_FORMAT, fill: strokeColor });
   130→  const [showTextBar, setShowTextBar] = useState(false);
   131→  const [textBarPos, setTextBarPos] = useState<{ top: number; left: number }>({ top: 0, left: 0 });
   132→
   133→  /* ----- undo/redo ----- */
   134→  const undoStackRef = useRef<string[]>([]);
   135→  const redoStackRef = useRef<string[]>([]);
   136→  const [canUndo, setCanUndo] = useState(false);
   137→  const [canRedo, setCanRedo] = useState(false);
   138→  const skipUndoRef = useRef(false);
   139→
   140→  /* ----- paginas ----- */
   141→  const [pages, setPages] = useState<PageData[]>([
   142→    { id: 'page-1', pageNumber: 1, paperStyle: initPaperStyle, paperColor: initPaperColor },
   143→  ]);
   144→  const [activePageId, setActivePageId] = useState('page-1');
   145→
   146→  /* ----- desenho de formas (temporarios) ----- */
   147→  const isDrawingShape = useRef(false);
   148→  const shapeOriginRef = useRef({ x: 0, y: 0 });
   149→  const tempShapeRef = useRef<Rect | Circle | Line | null>(null);
   150→
   151→  /* ----- auto-save debounced ----- */
   152→  const debounceTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   153→  const lastSavedJSON = useRef<string>('');
   154→
   155→  /* ----- space pan ----- */
   156→  const spaceHeld = useRef(false);
   157→  const isPanning = useRef(false);
   158→  const panStart = useRef({ x: 0, y: 0 });
   159→  const viewportOffset = useRef({ left: 0, top: 0 });
   160→
   161→  /* ===== propriedades especiais para serializacao ===== */
   162→  const SERIALIZE_PROPS = ['__isPaperBg', '__isTape', '__tapeColor'];
   163→
   164→  /* ========================================================================
   165→     Gerador de fundo de papel
   166→   ======================================================================== */
   167→
   168→  const buildPaperBackground = useCallback(
   169→    (fc: Canvas): void => {
   170→      const existing = fc.getObjects().find((o) => (o as any).__isPaperBg);
   171→      if (existing) fc.remove(existing);
   172→
   173→      if (paperStyle === 'blank') return;
   174→
   175→      const objs: any[] = [];
   176→
   177→      if (paperStyle === 'lined') {
   178→        for (let y = LINE_SPACING; y < CANVAS_H; y += LINE_SPACING) {
   179→          objs.push(
   180→            new Line([0, y, CANVAS_W, y], {
   181→              stroke: lineColor,
   182→              strokeWidth: 1,
   183→              selectable: false,
   184→              evented: false,
   185→            } as any),
   186→          );
   187→        }
   188→      } else if (paperStyle === 'grid') {
   189→        for (let x = LINE_SPACING; x < CANVAS_W; x += LINE_SPACING) {
   190→          objs.push(
   191→            new Line([x, 0, x, CANVAS_H], {
   192→              stroke: lineColor,
   193→              strokeWidth: 0.5,
   194→              selectable: false,
   195→              evented: false,
   196→            } as any),
   197→          );
   198→        }
   199→        for (let y = LINE_SPACING; y < CANVAS_H; y += LINE_SPACING) {
   200→          objs.push(
   201→            new Line([0, y, CANVAS_W, y], {
   202→              stroke: lineColor,
   203→              strokeWidth: 0.5,
   204→              selectable: false,
   205→              evented: false,
   206→            } as any),
   207→          );
   208→        }
   209→      } else if (paperStyle === 'dotted') {
   210→        for (let x = LINE_SPACING; x < CANVAS_W; x += LINE_SPACING) {
   211→          for (let y = LINE_SPACING; y < CANVAS_H; y += LINE_SPACING) {
   212→            objs.push(
   213→              new Circle(
   214→                { left: x, top: y, radius: 1.2, fill: lineColor, originX: 'center', originY: 'center' } as any,
   215→              ),
   216→            );
   217→          }
   218→        }
   219→      }
   220→
   221→      if (objs.length === 0) return;
   222→
   223→      const group = new Group(objs, {
   224→        selectable: false,
   225→        evented: false,
   226→      } as any);
   227→      (group as any).__isPaperBg = true;
   228→
   229→      fc.add(group);
   230→      fc.sendObjectToBack(group);
   231→    },
   232→    [paperStyle, lineColor],
   233→  );
   234→
   235→  /* ========================================================================
   236→     Undo / Redo
   237→   ======================================================================== */
   238→
   239→  const saveUndoState = useCallback((fc: Canvas) => {
   240→    if (skipUndoRef.current) {
   241→      skipUndoRef.current = false;
   242→      return;
   243→    }
   244→    const json = JSON.stringify(fc.toJSON(SERIALIZE_PROPS));
   245→    undoStackRef.current.push(json);
   246→    if (undoStackRef.current.length > MAX_UNDO) undoStackRef.current.shift();
   247→    redoStackRef.current = [];
   248→    setCanUndo(undoStackRef.current.length > 0);
   249→    setCanRedo(false);
   250→  }, []);
   251→
   252→  const handleUndo = useCallback(() => {
   253→    const fc = fcRef.current;
   254→    if (!fc || undoStackRef.current.length === 0) return;
   255→    const currentJSON = JSON.stringify(fc.toJSON(SERIALIZE_PROPS));
   256→    redoStackRef.current.push(currentJSON);
   257→    const prevJSON = undoStackRef.current.pop()!;
   258→    skipUndoRef.current = true;
   259→    fc.loadFromJSON(prevJSON).then(() => {
   260→      fc.renderAll();
   261→      setCanUndo(undoStackRef.current.length > 0);
   262→      setCanRedo(true);
   263→    });
   264→  }, []);
   265→
   266→  const handleRedo = useCallback(() => {
   267→    const fc = fcRef.current;
   268→    if (!fc || redoStackRef.current.length === 0) return;
   269→    const currentJSON = JSON.stringify(fc.toJSON(SERIALIZE_PROPS));
   270→    undoStackRef.current.push(currentJSON);
   271→    const nextJSON = redoStackRef.current.pop()!;
   272→    skipUndoRef.current = true;
   273→    fc.loadFromJSON(nextJSON).then(() => {
   274→      fc.renderAll();
   275→      setCanUndo(true);
   276→      setCanRedo(redoStackRef.current.length > 0);
   277→    });
   278→  }, []);
   279→
   280→  /* ========================================================================
   281→     Auto-save debounced
   282→   ======================================================================== */
   283→
   284→  const scheduleAutoSave = useCallback((fc: Canvas) => {
   285→    if (debounceTimerRef.current) clearTimeout(debounceTimerRef.current);
   286→    debounceTimerRef.current = setTimeout(() => {
   287→      const json = JSON.stringify(fc.toJSON(SERIALIZE_PROPS));
   288→      if (json !== lastSavedJSON.current) {
   289→        lastSavedJSON.current = json;
   290→        onChange?.(json);
   291→      }
   292→    }, SAVE_DEBOUNCE_MS);
   293→  }, [onChange]);
   294→
   295→  /* ========================================================================
   296→     Atualizar brush/ferramenta
   297→   ======================================================================== */
   298→
   299→  const applyTool = useCallback(
   300→    (fc: Canvas, tool: string) => {
   301→      fc.isDrawingMode = false;
   302→      fc.selection = tool === 'select';
   303→      activeToolRef.current = tool;
   304→
   305→      switch (tool) {
   306→        case 'pen': {
   307→          fc.isDrawingMode = true;
   308→          const brush = new SmoothBrush(fc);
   309→          brush.color = strokeColor;
   310→          brush.width = 1.5;
   311→          fc.freeDrawingBrush = brush;
   312→          break;
   313→        }
   314→        case 'pencil': {
   315→          fc.isDrawingMode = true;
   316→          const brush = new SmoothBrush(fc);
   317→          brush.color = strokeColor;
   318→          brush.width = 3;
   319→          fc.freeDrawingBrush = brush;
   320→          break;
   321→        }
   322→        case 'highlighter': {
   323→          fc.isDrawingMode = true;
   324→          const brush = new PencilBrush(fc);
   325→          brush.color = strokeColor;
   326→          brush.width = 20;
   327→          (brush as any).globalCompositeOperation = 'multiply';
   328→          fc.freeDrawingBrush = brush;
   329→          break;
   330→        }
   331→        case 'eraser': {
   332→          fc.isDrawingMode = true;
   333→          const brush = new PencilBrush(fc);
   334→          brush.color = paperColor;
   335→          brush.width = 20;
   336→          fc.freeDrawingBrush = brush;
   337→          break;
   338→        }
   339→        case 'text':
   340→        case 'rectangle':
   341→        case 'circle':
   342→        case 'line':
   343→        case 'select':
   344→          break;
   345→      }
   346→
   347→      // Cursor
   348→      if (tool === 'text') {
   349→        fc.defaultCursor = 'text';
   350→        fc.hoverCursor = 'text';
   351→      } else if (tool === 'select') {
   352→        fc.defaultCursor = 'default';
   353→        fc.hoverCursor = 'move';
   354→      } else {
   355→        fc.defaultCursor = 'crosshair';
   356→        fc.hoverCursor = 'crosshair';
   357→      }
   358→    },
   359→    [strokeColor, paperColor],
   360→  );
   361→
   362→  /* ========================================================================
   363→     Text helpers
   364→   ======================================================================== */
   365→
   366→  const syncFormatFromText = useCallback((obj: any) => {
   367→    if (!obj || obj.type !== 'textbox') return;
   368→    setTextFormat({
   369→      fontFamily: obj.fontFamily || DEFAULT_TEXT_FORMAT.fontFamily,
   370→      fontSize: obj.fontSize || DEFAULT_TEXT_FORMAT.fontSize,
   371→      fontWeight: obj.fontWeight || 'normal',
   372→      fontStyle: obj.fontStyle || 'normal',
   373→      underline: !!obj.underline,
   374→      linethrough: !!obj.linethrough,
   375→      textAlign: obj.textAlign || 'left',
   376→      fill: obj.fill || '#000000',
   377→      lineHeight: obj.lineHeight || 1.3,
   378→      charSpacing: obj.charSpacing || 0,
   379→    });
   380→  }, []);
   381→
   382→  const applyFormatToActiveText = useCallback(
   383→    (patch: Partial<TextFormat>) => {
   384→      const fc = fcRef.current;
   385→      if (!fc) return;
   386→      const obj = fc.getActiveObject() as any;
   387→      if (!obj || obj.type !== 'textbox') return;
   388→
   389→      setTextFormat((prev) => {
   390→        const next = { ...prev, ...patch };
   391→        obj.set({
   392→          fontFamily: next.fontFamily,
   393→          fontSize: next.fontSize,
   394→          fontWeight: next.fontWeight,
   395→          fontStyle: next.fontStyle,
   396→          underline: next.underline,
   397→          linethrough: next.linethrough,
   398→          textAlign: next.textAlign,
   399→          fill: next.fill,
   400→          lineHeight: next.lineHeight,
   401→          charSpacing: next.charSpacing,
   402→        });
   403→        fc.renderAll();
   404→        scheduleAutoSave(fc);
   405→        return next;
   406→      });
   407→    },
   408→    [scheduleAutoSave],
   409→  );
   410→
   411→  const updateTextBarPosition = useCallback(() => {
   412→    const fc = fcRef.current;
   413→    if (!fc) return;
   414→    const obj = fc.getActiveObject() as any;
   415→    if (!obj || obj.type !== 'textbox') {
   416→      setShowTextBar(false);
   417→      return;
   418→    }
   419→
   420→    const canvasRect = containerRef.current?.querySelector('canvas')?.getBoundingClientRect();
   421→    if (!canvasRect) return;
   422→
   423→    // Get the bounding rect of the text object in viewport coordinates
   424→    const objBound = obj.getBoundingRect(true, true);
   425→    const vpt = fc.viewportTransform!;
   426→    const scaleX = vpt[0];
   427→    const scaleY = vpt[3];
   428→
   429→    const barLeft = canvasRect.left + objBound.left * scaleX;
   430→    const barTop = canvasRect.top + objBound.top * scaleY - 44;
   431→
   432→    setTextBarPos({ top: Math.max(4, barTop), left: barLeft });
   433→    setShowTextBar(true);
   434→  }, []);
   435→
   436→  /* ========================================================================
   437→     Inicializacao do canvas
   438→   ======================================================================== */
   439→
   440→  useEffect(() => {
   441→    const el = canvasElRef.current;
   442→    if (!el) return;
   443→
   444→    const fc = new Canvas(el, {
   445→      width: CANVAS_W,
   446→      height: CANVAS_H,
   447→      backgroundColor: paperColor,
   448→      selection: true,
   449→      preserveObjectStacking: true,
   450→    });
   451→    fcRef.current = fc;
   452→
   453→    // Fundo do papel
   454→    buildPaperBackground(fc);
   455→    fc.renderAll();
   456→
   457→    // Carregar estado inicial se existir
   458→    if (initialData) {
   459→      try {
   460→        fc.loadFromJSON(initialData).then(() => {
   461→          fc.renderAll();
   462→          saveUndoState(fc);
   463→        });
   464→      } catch {
   465→        saveUndoState(fc);
   466→      }
   467→    } else {
   468→      saveUndoState(fc);
   469→    }
   470→
   471→    /* ---------- eventos de ferramenta ---------- */
   472→
   473→    fc.on('mouse:down', (opt) => {
   474→      const e = opt.e as MouseEvent;
   475→      const pointer = fc.getScenePoint(e);
   476→      const currentTool = activeToolRef.current;
   477→
   478→      // Fita adesiva
   479→      if (tapeMode) {
   480→        addTape(fc, pointer.x, pointer.y, tapeColor);
   481→        saveUndoState(fc);
   482→        scheduleAutoSave(fc);
   483→        return;
   484→      }
   485→
   486→      // Ferramenta de texto - criar Textbox Canva-style
   487→      if (currentTool === 'text') {
   488→        // If clicking on existing text, let fabric handle it (double-click to edit)
   489→        const target = opt.target as any;
   490→        if (target && target.type === 'textbox') {
   491→          return; // Let fabric handle selection/editing of existing text
   492→        }
   493→
   494→        const textbox = new Textbox('', {
   495→          left: pointer.x,
   496→          top: pointer.y,
   497→          width: 300,
   498→          fontSize: textFormat.fontSize,
   499→          fill: strokeColor,
   500→          fontFamily: textFormat.fontFamily,
   501→          fontWeight: textFormat.fontWeight,
   502→          fontStyle: textFormat.fontStyle,
   503→          underline: textFormat.underline,
   504→          linethrough: textFormat.linethrough,
   505→          textAlign: textFormat.textAlign,
   506→          lineHeight: textFormat.lineHeight,
   507→          charSpacing: textFormat.charSpacing,
   508→          editable: true,
   509→          cursorColor: '#000000',
   510→          cursorWidth: 2,
   511→          editingBorderColor: '#f59e0b',
   512→          padding: 8,
   513→        } as any);
   514→
   515→        fc.add(textbox);
   516→        fc.setActiveObject(textbox);
   517→        fc.renderAll();
   518→
   519→        // Enter editing mode with a small delay for reliable focus
   520→        setTimeout(() => {
   521→          (textbox as any).enterEditing();
   522→          (textbox as any).selectAll();
   523→          // Focus the hidden textarea that fabric creates
   524→          const hiddenTextarea = el.parentElement?.querySelector('textarea.fabric-textbox') as HTMLTextAreaElement;
   525→          if (hiddenTextarea) {
   526→            hiddenTextarea.focus();
   527→          }
   528→        }, 50);
   529→
   530→        // Switch back to select after creating text (Canva behavior)
   531→        setActiveTool('select');
   532→        activeToolRef.current = 'select';
   533→        applyTool(fc, 'select');
   534→
   535→        saveUndoState(fc);
   536→        scheduleAutoSave(fc);
   537→        return;
   538→      }
   539→
   540→      // Pan com space ou botao do meio
   541→      if (spaceHeld.current || e.button === 1) {
   542→        isPanning.current = true;
   543→        panStart.current = { x: e.clientX, y: e.clientY };
   544→        viewportOffset.current = { left: fc.viewportLeft, top: fc.viewportTop };
   545→        fc.selection = false;
   546→        return;
   547→      }
   548→
   549→      // Formas (retangulo, circulo, linha)
   550→      if (['rectangle', 'circle', 'line'].includes(currentTool)) {
   551→        isDrawingShape.current = true;
   552→        shapeOriginRef.current = { x: pointer.x, y: pointer.y };
   553→
   554→        if (currentTool === 'rectangle') {
   555→          const rect = new Rect({
   556→            left: pointer.x,
   557→            top: pointer.y,
   558→            width: 0,
   559→            height: 0,
   560→            fill: 'transparent',
   561→            stroke: strokeColor,
   562→            strokeWidth,
   563→          } as any);
   564→          fc.add(rect);
   565→          fc.setActiveObject(rect);
   566→          tempShapeRef.current = rect;
   567→        } else if (currentTool === 'circle') {
   568→          const circ = new Circle({
   569→            left: pointer.x,
   570→            top: pointer.y,
   571→            radius: 1,
   572→            fill: 'transparent',
   573→            stroke: strokeColor,
   574→            strokeWidth,
   575→            originX: 'center',
   576→            originY: 'center',
   577→          } as any);
   578→          fc.add(circ);
   579→          fc.setActiveObject(circ);
   580→          tempShapeRef.current = circ;
   581→        } else if (currentTool === 'line') {
   582→          const line = new Line([pointer.x, pointer.y, pointer.x, pointer.y], {
   583→            stroke: strokeColor,
   584→            strokeWidth,
   585→          } as any);
   586→          fc.add(line);
   587→          fc.setActiveObject(line);
   588→          tempShapeRef.current = line;
   589→        }
   590→      }
   591→    });
   592→
   593→    fc.on('mouse:move', (opt) => {
   594→      const e = opt.e as MouseEvent;
   595→
   596→      if (isPanning.current) {
   597→        const dx = e.clientX - panStart.current.x;
   598→        const dy = e.clientY - panStart.current.y;
   599→        fc.setViewportTransform([
   600→          zoom, 0, 0, zoom,
   601→          viewportOffset.current.left + dx,
   602→          viewportOffset.current.top + dy,
   603→        ]);
   604→        return;
   605→      }
   606→
   607→      if (isDrawingShape.current && tempShapeRef.current) {
   608→        const pointer = fc.getScenePoint(e);
   609→        const origin = shapeOriginRef.current;
   610→        const shape = tempShapeRef.current;
   611→
   612→        if (activeTool === 'rectangle') {
   613→          const left = Math.min(origin.x, pointer.x);
   614→          const top = Math.min(origin.y, pointer.y);
   615→          const w = Math.abs(pointer.x - origin.x);
   616→          const h = Math.abs(pointer.y - origin.y);
   617→          shape.set({ left, top, width: w, height: h });
   618→        } else if (activeTool === 'circle') {
   619→          const dx = pointer.x - origin.x;
   620→          const dy = pointer.y - origin.y;
   621→          const r = Math.sqrt(dx * dx + dy * dy) / 2;
   622→          const cx = (origin.x + pointer.x) / 2;
   623→          const cy = (origin.y + pointer.y) / 2;
   624→          shape.set({ left: cx, top: cy, radius: Math.max(1, r) } as any);
   625→        } else if (activeTool === 'line') {
   626→          (shape as any).set({ x2: pointer.x, y2: pointer.y });
   627→        }
   628→        fc.renderAll();
   629→      }
   630→    });
   631→
   632→    fc.on('mouse:up', () => {
   633→      if (isPanning.current) {
   634→        isPanning.current = false;
   635→        fc.selection = activeToolRef.current === 'select';
   636→        return;
   637→      }
   638→
   639→      if (isDrawingShape.current) {
   640→        isDrawingShape.current = false;
   641→        tempShapeRef.current = null;
   642→        saveUndoState(fc);
   643→        scheduleAutoSave(fc);
   644→      }
   645→    });
   646→
   647→    fc.on('path:created', () => {
   648→      saveUndoState(fc);
   649→      scheduleAutoSave(fc);
   650→    });
   651→
   652→    fc.on('object:modified', () => {
   653→      saveUndoState(fc);
   654→      scheduleAutoSave(fc);
   655→    });
   656→
   657→    /* ---------- Text object events ---------- */
   658→
   659→    // When text starts/ends editing, update the toolbar
   660→    fc.on('text:editing:entered', () => {
   661→      const obj = fc.getActiveObject() as any;
   662→      if (obj) syncFormatFromText(obj);
   663→      updateTextBarPosition();
   664→    });
   665→
   666→    fc.on('text:editing:exited', () => {
   667→ const obj = fc.getActiveObject() as any;
   668→      if (obj && obj.type === 'textbox') {
   669→        // If text is empty after editing, remove it
   670→        if (!obj.text || obj.text.trim() === '') {
   671→          fc.remove(obj);
   672→          fc.renderAll();
   673→          setShowTextBar(false);
   674→          saveUndoState(fc);
   675→          scheduleAutoSave(fc);
   676→          return;
   677→        }
   678→        saveUndoState(fc);
   679→        scheduleAutoSave(fc);
   680→      }
   681→    });
   682→
   683→    // When selection changes, show/hide text bar
   684→    fc.on('selection:created', (opt) => {
   685→      const obj = opt.selected?.[0] as any;
   686→      if (obj && obj.type === 'textbox') {
   687→        syncFormatFromText(obj);
   688→        setTimeout(updateTextBarPosition, 10);
   689→      } else {
   690→        setShowTextBar(false);
   691→      }
   692→    });
   693→
   694→    fc.on('selection:updated', (opt) => {
   695→      const obj = opt.selected?.[0] as any;
   696→      if (obj && obj.type === 'textbox') {
   697→        syncFormatFromText(obj);
   698→        setTimeout(updateTextBarPosition, 10);
   699→      } else {
   700→        setShowTextBar(false);
   701→      }
   702→    });
   703→
   704→    fc.on('selection:cleared', () => {
   705→      setShowTextBar(false);
   706→    });
   707→
   708→    /* ---------- zoom com rolagem ---------- */
   709→
   710→    fc.on('mouse:wheel', (opt) => {
   711→      const e = opt.e as WheelEvent;
   712→      e.preventDefault();
   713→      e.stopPropagation();
   714→
   715→      const delta = e.deltaY;
   716→      let newZoom = zoom + (delta > 0 ? -0.05 : 0.05);
   717→      newZoom = Math.min(5, Math.max(0.1, newZoom));
   718→
   719→      const vpt = fc.viewportTransform;
   720→      const newVpt = [...vpt];
   721→      newVpt[0] = newZoom;
   722→      newVpt[3] = newZoom;
   723→      newVpt[4] = vpt[4] + (vpt[0] - newZoom) * (CANVAS_W / 2);
   724→      newVpt[5] = vpt[5] + (vpt[3] - newZoom) * (CANVAS_H / 2);
   725→
   726→      fc.setViewportTransform(newVpt as [number, number, number, number, number, number]);
   727→      setZoom(newZoom);
   728→    });
   729→
   730→    // Ferramenta inicial
   731→    applyTool(fc, activeTool);
   732→
   733→    return () => {
   734→      fc.dispose();
   735→      fcRef.current = null;
   736→    };
   737→    }, []);
   738→
   739→  /* ========================================================================
   740→     Sincronizar estado de ferramenta com canvas
   741→   ======================================================================== */
   742→
   743→  useEffect(() => {
   744→    const fc = fcRef.current;
   745→    if (!fc) return;
   746→    applyTool(fc, activeTool);
   747→  }, [activeTool, strokeColor, strokeWidth, paperColor, applyTool]);
   748→
   749→  /* ========================================================================
   750→     Mudanca de cor de papel / estilo - reconstruir fundo
   751→   ======================================================================== */
   752→
   753→  useEffect(() => {
   754→    const fc = fcRef.current;
   755→    if (!fc) return;
   756→    fc.backgroundColor = paperColor;
   757→    buildPaperBackground(fc);
   758→    fc.renderAll();
   759→    scheduleAutoSave(fc);
   760→  }, [paperColor, paperStyle, buildPaperBackground, scheduleAutoSave]);
   761→
   762→  /* ========================================================================
   763→     Atalhos de teclado
   764→   ======================================================================== */
   765→
   766→  useEffect(() => {
   767→    const handleKeyDown = (e: KeyboardEvent) => {
   768→      const fc = fcRef.current;
   769→      const activeObj = fc?.getActiveObject() as any;
   770→
   771→      // Don't intercept if editing text
   772→      if (activeObj && activeObj.isEditing) {
   773→        // But handle formatting shortcuts inside text
   774→        if ((e.ctrlKey || e.metaKey)) {
   775→          if (e.key === 'b' || e.key === 'B') {
   776→            e.preventDefault();
   777→            const newWeight = activeObj.fontWeight === 'bold' ? 'normal' : 'bold';
   778→            activeObj.set('fontWeight', newWeight);
   779→            fc?.renderAll();
   780→            syncFormatFromText(activeObj);
   781→          } else if (e.key === 'i' || e.key === 'I') {
   782→            e.preventDefault();
   783→            const newStyle = activeObj.fontStyle === 'italic' ? 'normal' : 'italic';
   784→            activeObj.set('fontStyle', newStyle);
   785→            fc?.renderAll();
   786→            syncFormatFromText(activeObj);
   787→          } else if (e.key === 'u' || e.key === 'U') {
   788→            e.preventDefault();
   789→            activeObj.set('underline', !activeObj.underline);
   790→            fc?.renderAll();
   791→            syncFormatFromText(activeObj);
   792→          }
   793→        }
   794→        return;
   795→      }
   796→
   797→      if (e.code === 'Space' && !e.repeat) {
   798→        e.preventDefault();
   799→        spaceHeld.current = true;
   800→      }
   801→
   802→      if ((e.ctrlKey || e.metaKey) && e.key === 'z') {
   803→        e.preventDefault();
   804→        if (e.shiftKey) {
   805→          handleRedo();
   806→        } else {
   807→          handleUndo();
   808→        }
   809→      }
   810→
   811→      // Delete selected object
   812→      if ((e.key === 'Delete' || e.key === 'Backspace') && activeObj && !activeObj.isEditing) {
   813→        fc?.remove(activeObj);
   814→        fc?.renderAll();
   815→        saveUndoState(fc);
   816→        scheduleAutoSave(fc);
   817→      }
   818→
   819→      // Double-tap T shortcut for text
   820→      if (e.key === 't' && !e.ctrlKey && !e.metaKey && !e.altKey) {
   821→        const target = e.target as HTMLElement;
   822→        if (target.tagName !== 'INPUT' && target.tagName !== 'TEXTAREA') {
   823→          setActiveTool('text');
   824→        }
   825→      }
   826→
   827→      // Escape to deselect / exit text mode
   828→      if (e.key === 'Escape') {
   829→        if (activeObj && activeObj.isEditing) {
   830→          (activeObj as any).exitEditing();
   831→        } else {
   832→          fc?.discardActiveObject();
   833→          fc?.renderAll();
   834→          setShowTextBar(false);
   835→        }
   836→      }
   837→
   838→      // Enter on selected text to start editing
   839→      if (e.key === 'Enter' && activeObj && activeObj.type === 'textbox' && !activeObj.isEditing) {
   840→        e.preventDefault();
   841→        (activeObj as any).enterEditing();
   842→      }
   843→    };
   844→    const handleKeyUp = (e: KeyboardEvent) => {
   845→      if (e.code === 'Space') {
   846→        spaceHeld.current = false;
   847→      }
   848→    };
   849→    window.addEventListener('keydown', handleKeyDown);
   850→    window.addEventListener('keyup', handleKeyUp);
   851→    return () => {
   852→      window.removeEventListener('keydown', handleKeyDown);
   853→      window.removeEventListener('keyup', handleKeyUp);
   854→    };
   855→  }, [handleUndo, handleRedo, saveUndoState, scheduleAutoSave, syncFormatFromText]);
   856→
   857→  /* ========================================================================
   858→     Paginas
   859→   ======================================================================== */
   860→
   861→  const handlePageSelect = useCallback(
   862→    (id: string) => {
   863→      const fc = fcRef.current;
   864→      if (!fc) return;
   865→
   866→      setPages((prev) =>
   867→        prev.map((p) =>
   868→          p.id === activePageId
   869→            ? { ...p, canvasJSON: JSON.stringify(fc.toJSON(SERIALIZE_PROPS)) }
   870→            : p,
   871→        ),
   872→      );
   873→
   874→      const page = pages.find((p) => p.id === id);
   875→      if (!page) return;
   876→
   877→      setActivePageId(id);
   878→      setPaperStyle(page.paperStyle);
   879→      setPaperColor(page.paperColor);
   880→      setShowTextBar(false);
   881→
   882→      const objects = fc.getObjects();
   883→      for (let i = objects.length - 1; i >= 0; i--) {
   884→        fc.remove(objects[i]);
   885→      }
   886→
   887→      if (page.canvasJSON) {
   888→        try {
   889→          fc.loadFromJSON(page.canvasJSON).then(() => {
   890→            fc.renderAll();
   891→            saveUndoState(fc);
   892→          });
   893→        } catch {
   894→          buildPaperBackground(fc);
   895→          fc.renderAll();
   896→          saveUndoState(fc);
   897→        }
   898→      } else {
   899→        buildPaperBackground(fc);
   900→        fc.renderAll();
   901→        saveUndoState(fc);
   902→      }
   903→    },
   904→    [activePageId, pages, buildPaperBackground, saveUndoState],
   905→  );
   906→
   907→  const handleAddPage = useCallback(() => {
   908→    const fc = fcRef.current;
   909→    if (!fc) return;
   910→
   911→    setPages((prev) =>
   912→      prev.map((p) =>
   913→        p.id === activePageId
   914→          ? { ...p, canvasJSON: JSON.stringify(fc.toJSON(SERIALIZE_PROPS)) }
   915→          : p,
   916→      ),
   917→    );
   918→
   919→    const newId = `page-${Date.now()}`;
   920→    const newPage: PageData = {
   921→      id: newId,
   922→      pageNumber: pages.length + 1,
   923→      paperStyle,
   924→      paperColor,
   925→    };
   926→
   927→    setPages((prev) => [...prev, newPage]);
   928→    setActivePageId(newId);
   929→    setShowTextBar(false);
   930→
   931→    const objects = fc.getObjects();
   932→    for (let i = objects.length - 1; i >= 0; i--) {
   933→      fc.remove(objects[i]);
   934→    }
   935→    buildPaperBackground(fc);
   936→    fc.renderAll();
   937→    saveUndoState(fc);
   938→  }, [activePageId, pages.length, paperStyle, paperColor, buildPaperBackground, saveUndoState]);
   939→
   940→  const handleDeletePage = useCallback(
   941→    (id: string) => {
   942→      if (pages.length <= 1) return;
   943→
   944→      setPages((prev) => {
   945→        const filtered = prev.filter((p) => p.id !== id);
   946→        return filtered.map((p, i) => ({ ...p, pageNumber: i + 1 }));
   947→      });
   948→
   949→      if (id === activePageId) {
   950→        const remaining = pages.filter((p) => p.id !== id);
   951→        if (remaining.length > 0) {
   952→          handlePageSelect(remaining[0].id);
   953→        }
   954→      }
   955→    },
   956→    [pages, activePageId, handlePageSelect],
   957→  );
   958→
   959→  /* ========================================================================
   960→     Adicionar texto / imagem
   961→   ======================================================================== */
   962→
   963→  const handleAddText = useCallback(() => {
   964→    const fc = fcRef.current;
   965→    if (!fc) return;
   966→    setActiveTool('text');
   967→    applyTool(fc, 'text');
   968→  }, [applyTool]);
   969→
   970→  const handleAddImage = useCallback(() => {
   971→    fileInputRef.current?.click();
   972→  }, []);
   973→
   974→  const handleFileChange = useCallback(
   975→    (e: React.ChangeEvent<HTMLInputElement>) => {
   976→      const fc = fcRef.current;
   977→      if (!fc) return;
   978→      const file = e.target.files?.[0];
   979→      if (!file) return;
   980→
   981→      const reader = new FileReader();
   982→      reader.onload = () => {
   983→        const dataUrl = reader.result as string;
   984→        FabricImage.fromURL(dataUrl).then((img) => {
   985→          const scale = Math.min(
   986→            (CANVAS_W * 0.6) / (img.width || 1),
   987→            (CANVAS_H * 0.6) / (img.height || 1),
   988→            1,
   989→          );
   990→          img.set({
   991→            scaleX: scale,
   992→            scaleY: scale,
   993→            left: CANVAS_W / 2 - ((img.width || 0) * scale) / 2,
   994→            top: CANVAS_H / 2 - ((img.height || 0) * scale) / 2,
   995→          } as any);
   996→          fc.add(img);
   997→          fc.setActiveObject(img);
   998→          fc.renderAll();
   999→          saveUndoState(fc);
  1000→          scheduleAutoSave(fc);
  1001→        });
  1002→      };
  1003→      reader.readAsDataURL(file);
  1004→      e.target.value = '';
  1005→    },
  1006→    [saveUndoState, scheduleAutoSave],
  1007→  );
  1008→
  1009→  /* ========================================================================
  1010→     Render
  1011→   ======================================================================== */
  1012→
  1013→  return (
  1014→    <div
  1015→      ref={containerRef}
  1016→      className="relative flex h-full w-full flex-col bg-gray-100"
  1017→      style={{ cursor: activeTool === 'select' ? 'default' : activeTool === 'text' ? 'text' : 'crosshair' }}
  1018→    >
  1019→      {/* Toolbar fixa no topo */}
  1020→      <EditorToolbar
  1021→        activeTool={activeTool}
  1022→        onToolChange={setActiveTool}
  1023→        strokeColor={strokeColor}
  1024→        onColorChange={setStrokeColor}
  1025→        strokeWidth={strokeWidth}
  1026→        onWidthChange={setStrokeWidth}
  1027→        onUndo={handleUndo}
  1028→        onRedo={handleRedo}
  1029→        canUndo={canUndo}
  1030→        canRedo={canRedo}
  1031→        zoom={zoom}
  1032→        onZoomChange={setZoom}
  1033→        paperStyle={paperStyle}
  1034→        onPaperStyleChange={setPaperStyle}
  1035→        paperColor={paperColor}
  1036→        onPaperColorChange={setPaperColor}
  1037→        onAddText={handleAddText}
  1038→        onAddImage={handleAddImage}
  1039→        onToggleTape={() => setTapeMode((prev) => !prev)}
  1040→        tapeMode={tapeMode}
  1041→      />
  1042→
  1043→      {/* Floating text formatting bar (Canva-style) */}
  1044→      <TextFormattingBar
  1045→        format={textFormat}
  1046→        onFormatChange={applyFormatToActiveText}
  1047→        visible={showTextBar}
  1048→        position={textBarPos}
  1049→      />
  1050→
  1051→      {/* Fita color picker (aparece quando tapeMode ativo) */}
  1052→      {tapeMode && (
  1053→        <div className="flex items-center gap-3 rounded-b-xl border border-t-0 border-black/5 bg-white/80 px-3 py-2 backdrop-blur-md">
  1054→          <span className="text-xs font-medium text-gray-500">Cor da fita:</span>
  1055→          <StickyTapePicker activeTapeColor={tapeColor} onColorSelect={setTapeColor} />
  1056→          <span className="ml-auto text-xs text-gray-400">Clique no canvas para adicionar fita</span>
  1057→        </div>
  1058→      )}
  1059→
  1060→      {/* Area principal */}
  1061→      <div className="flex flex-1 overflow-hidden">
  1062→        {/* Painel de paginas (desktop) */}
  1063→        <div className="hidden md:block">
  1064→          <PagePanel
  1065→            pages={pages}
  1066→            activePageId={activePageId}
  1067→            onPageSelect={handlePageSelect}
  1068→            onAddPage={handleAddPage}
  1069→            onDeletePage={handleDeletePage}
  1070→          />
  1071→        </div>
  1072→
  1073→        {/* Canvas wrapper */}
  1074→        <div className="relative flex-1 overflow-auto flex items-start justify-center p-4">
  1075→          <div
  1076→            className="shadow-xl"
  1077→            style={{
  1078→              width: CANVAS_W * zoom,
  1079→              height: CANVAS_H * zoom,
  1080→              overflow: 'hidden',
  1081→              borderRadius: '4px',
  1082→            }}
  1083→          >
  1084→            <canvas ref={canvasElRef} />
  1085→          </div>
  1086→
  1087→          {/* Hidden file input */}
  1088→          <input
  1089→            ref={fileInputRef}
  1090→            type="file"
  1091→            accept="image/*"
  1092→            className="hidden"
  1093→            onChange={handleFileChange}
  1094→          />
  1095→
  1096→          {/* Floating pen toolbar */}
  1097→          <FloatingPenToolbar
  1098→            activeTool={activeTool}
  1099→            onToolChange={setActiveTool}
  1100→            strokeColor={strokeColor}
  1101→            onStrokeColorChange={setStrokeColor}
  1102→            strokeWidth={strokeWidth}
  1103→            onStrokeWidthChange={setStrokeWidth}
  1104→            onUndo={handleUndo}
  1105→            onRedo={handleRedo}
  1106→            onImageImport={handleAddImage}
  1107→            smoothing="media"
  1108→            onSmoothingChange={() => {}}
  1109→            stabilizer={50}
  1110→            onStabilizerChange={() => {}}
  1111→          />
  1112→        </div>
  1113→      </div>
  1114→    </div>
  1115→  );
  1116→}
