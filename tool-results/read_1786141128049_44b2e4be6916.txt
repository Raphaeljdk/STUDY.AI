   518→    setFormEstimatedMinutes(task.estimatedMinutes?.toString() || '');
   519→    setFormStatus(task.status);
   520→    setFormTitleError('');

   524→  const formatDueDate = (dateStr: string) => {
   525→    try {
   526→      const date = parseISO(dateStr);
   527→      if (isToday(date)) return 'Hoje';
   528→      if (isPast(date)) return 'Atrasada';
   529→      return formatDistanceToNow(date, { addSuffix: true, locale: ptBR });
   530→    } catch {
   531→      return '';
   532→    }
   533→  };
   534→
   535→  const isOverdue = (dateStr: string) => {
   536→    try {
   537→      const date = parseISO(dateStr);
   538→      return isPast(date) && !isToday(date);
   539→    } catch {
   540→      return false;
   541→    }
   542→  };
   543→
   544→  const formatMinutes = (mins: number) => {
   545→    if (mins < 60) return `${mins}min`;
   546→    const h = Math.floor(mins / 60);
   547→    const m = mins % 60;
   548→    return m > 0 ? `${h}h${m}min` : `${h}h`;
   549→  };
   550→
   551→  // ===== SORT & FILTER =====
   552→  const filteredAndSorted = tasks
   553→    .filter((t) => {
   554→      if (activeFilter === 'ALL') return true;
   555→      return t.status === activeFilter;
   556→    })
   557→    .sort((a, b) => {
   558→      let cmp = 0;
   559→      switch (sortKey) {
   560→        case 'dueDate': {
   561→          const aD = a.dueDate ? parseISO(a.dueDate).getTime() : Infinity;
   562→          const bD = b.dueDate ? parseISO(b.dueDate).getTime() : Infinity;
   563→          cmp = aD - bD;
   564→          break;
   565→        }
   566→        case 'priority':
   567→          cmp = (PRIORITY_CONFIG[a.priority]?.weight || 0) - (PRIORITY_CONFIG[b.priority]?.weight || 0);
   568→          break;
   569→        case 'createdAt':
   570→          cmp = new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
   571→          break;
   572→        case 'title':
   573→          cmp = a.title.localeCompare(b.title, 'pt-BR');
   574→          break;
   575→      }
   576→      return sortDir === 'asc' ? cmp : -cmp;
   577→    });
