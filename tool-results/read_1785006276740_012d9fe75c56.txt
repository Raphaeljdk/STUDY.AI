    20→// Lightweight error boundary for individual sections
    21→class SectionErrorBoundary extends Component<{ children: ReactNode; fallback?: ReactNode; name?: string }, { hasError: boolean; error: Error | null }> {
    22→  constructor(props: { children: ReactNode; fallback?: ReactNode; name?: string }) {
    23→    super(props);
    24→    this.state = { hasError: false, error: null };
    25→  }
    26→  static getDerivedStateFromError(error: Error) {
    27→    return { hasError: true, error };
    28→  }
    29→  componentDidCatch(error: Error, info: React.ErrorInfo) {
    30→    console.error(`[SectionErrorBoundary${this.props.name ? ` (${this.props.name})` : ''}]`, error, info.componentStack);
    31→  }
    32→  render() {
    33→    if (this.state.hasError) {
    34→      if (this.props.fallback) return this.props.fallback;
    35→      return (
    36→        <div className="flex min-h-[300px] items-center justify-center rounded-lg border border-dashed border-[var(--ws-glass-border)] p-6">
    37→          <div className="text-center">
    38→            <AlertCircle size={24} className="mx-auto mb-3 text-[var(--ws-accent)]" />
    39→            <p className="text-sm font-medium text-[var(--ws-text-primary)]">Erro ao carregar esta secao</p>
    40→            <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{this.state.error?.message || 'Erro desconhecido'}</p>
    41→            <button
    42→              onClick={() => { this.setState({ hasError: false, error: null }); }}
    43→              className="mt-3 text-xs font-medium text-[var(--ws-accent)] hover:underline"
    44→            >
    45→              Tentar novamente
    46→            </button>
    47→          </div>
    48→        </div>
    49→      );
    50→    }
    51→    return this.props.children;
    52→  }
    53→}

    55→// Safe rich text editor with fallback to plain textarea
    56→function SafeEditor({ content, onChange, placeholder, onError }: { content: string; onChange: (html: string) => void; placeholder: string; onError?: (msg: string) => void }) {
    57→  const [editorReady, setEditorReady] = useState(false);
    58→  const [editorFailed, setEditorFailed] = useState(false);
    59→  const [EditorComp, setEditorComp] = useState<React.ComponentType<{ content: string; onChange: (html: string) => void; placeholder: string }> | null>(null);
    60→  const saveTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    61→
    62→  useEffect(() => {
    63→    import('./RichTextEditor')
    64→      .then(m => { setEditorComp(() => m.RichTextEditor); setEditorReady(true); })
    65→      .catch((err) => {
    66→        console.warn('[SafeEditor] RichTextEditor failed to load, using fallback:', err);
    67→        setEditorFailed(true);
    68→        onError?.(err?.message || 'Editor falhou ao carregar');
    69→      });
    70→  }, []);
    71→
    72→  const handleChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => {
    73→    const val = e.target.value;
    74→    if (saveTimer.current) clearTimeout(saveTimer.current);
    75→    saveTimer.current = setTimeout(() => onChange(val), 500);
    76→  }, [onChange]);
    77→
    78→  if (editorFailed) {
    79→    return (
    80→      <div>
    81→        <div className="mb-3 flex items-center gap-2 rounded-lg bg-[var(--ws-accent)]/10 px-3 py-2 text-xs text-[var(--ws-accent)]">
    82→          <AlertCircle size={14} /> Editor avancado indisponivel. Usando editor simples.
    83→        </div>
    84→        <textarea
    85→          value={content}
    86→          onChange={handleChange}
    87→          placeholder={placeholder}
    88→          className="min-h-[400px] w-full resize-none bg-transparent p-5 text-sm leading-relaxed text-[var(--ws-text-primary)] placeholder-[var(--ws-text-tertiary)] outline-none"
    89→        />
    90→      </div>
    91→    );
    92→  }
    93→
    94→  if (!editorReady) {
    95→    return (
    96→      <div className="flex min-h-[400px] items-center justify-center">
    97→        <div className="text-center">
    98→          <div className="mx-auto mb-3 h-6 w-6 animate-spin rounded-full border-2 border-[var(--ws-glass-border)] border-t-[var(--ws-accent)]" />
    99→          <p className="text-xs text-[var(--ws-text-tertiary)]">Carregando editor...</p>
   100→        </div>
   101→      </div>
   102→    );
   103→  }
   104→
   105→  if (EditorComp) return <EditorComp content={content} onChange={onChange} placeholder={placeholder} />;
   106→  return null;
   107→}

   203→// ========== HELPER: Time-based greeting ==========
   204→function getGreeting(): string {
   205→  const h = new Date().getHours();
   206→  if (h < 12) return 'Bom dia';
   207→  if (h < 18) return 'Boa tarde';
   208→  return 'Boa noite';
   209→}

   211→function getGreetingSubtext(): string {
   212→  const h = new Date().getHours();
   213→  if (h < 12) return 'Comece o dia com foco e determinacao. Cada estudo e um passo adiante.';
   214→  if (h < 18) return 'A tarde e perfeita para revisar e consolidar o conhecimento.';
   215→  return 'Boas sessoes noturnas fixam melhor o que voce aprendeu.';
   216→}

   218→function formatReviewDate(dateStr: string): string {
   219→  const d = new Date(dateStr);
   220→  const now = new Date();
   221→  const diffMs = d.getTime() - now.getTime();
   222→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   223→
   224→  if (diffDays < 0) return 'Atrasado';
   225→  if (diffDays === 0) return 'Hoje';
   226→  if (diffDays === 1) return 'Amanha';
   227→  if (diffDays <= 7) return `em ${diffDays} dias`;
   228→  return d.toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit' });
   229→}

   231→function getStatusBadge(repetitions: number): { label: string; color: string; bg: string } {
   232→  if (repetitions === 0) return { label: 'Novo', color: 'var(--ws-accent)', bg: 'color-mix(in srgb, var(--ws-accent) 12%, transparent)' };

   235→}

   943→// ========== FLASHCARD REVIEWER ==========
   944→function FlashcardReviewer({ onBack }: { onBack: () => void }) {
   945→  const [cards, setCards] = useState<FlashcardItem[]>([]);
   946→  const [loading, setLoading] = useState(true);
   947→  const [currentIndex, setCurrentIndex] = useState(0);
   948→  const [flipped, setFlipped] = useState(false);
   949→  const [done, setDone] = useState(false);
   950→  const [reviewed, setReviewed] = useState(0);
   951→  const [ratings, setRatings] = useState<number[]>([]);
   952→  const fetchedRef = useRef(false);
   953→
   954→  useEffect(() => {
   955→    if (fetchedRef.current) return;
   956→    fetchedRef.current = true;
   957→    fetch('/api/flashcards?due=true').then(r => r.json()).then(d => {
   958→      if (d.flashcards) setCards(d.flashcards);
   959→    }).catch(() => {}).finally(() => setLoading(false));
   960→  }, []);
   961→
   962→  const handleReview = useCallback(async (quality: number) => {
   963→    const card = cards[currentIndex];
   964→    if (!card) return;
   965→    try {
   966→      await fetch(`/api/flashcards/${card.id}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ review: quality }) });
   967→    } catch {}
   968→    setReviewed(prev => prev + 1);
   969→    setRatings(prev => [...prev, quality]);
   970→    setFlipped(false);
   971→    if (currentIndex + 1 >= cards.length) setDone(true);
   972→    else setCurrentIndex(prev => prev + 1);
   973→  }, [cards, currentIndex]);
   974→
   975→  // Keyboard shortcuts
   976→  useEffect(() => {
   977→    if (done || !flipped || cards.length === 0) return;
   978→    const handleKey = (e: KeyboardEvent) => {
   979→      if (e.key === '1') handleReview(1);
   980→      else if (e.key === '2') handleReview(2);
   981→      else if (e.key === '3') handleReview(3);
   982→      else if (e.key === '4') handleReview(5);
   983→    };
   984→    window.addEventListener('keydown', handleKey);
   985→    return () => window.removeEventListener('keydown', handleKey);
   986→  }, [flipped, done, cards.length, handleReview]);
   987→
   988→  // Completion stats
   989→  const avgRating = ratings.length > 0 ? (ratings.reduce((a, b) => a + b, 0) / ratings.length).toFixed(1) : '0';
   990→  const hardCount = ratings.filter(r => r <= 2).length;
   991→  const easyCount = ratings.filter(r => r >= 4).length;
   992→
   993→  if (loading) return <div className="flex items-center justify-center py-16"><Loader2 size={24} className="animate-spin text-[var(--ws-text-tertiary)]" /></div>;

   995→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="py-16 text-center">
   996→      <Check size={48} className="mx-auto mb-4 text-[var(--ws-verdigris)]" strokeWidth={1} />
   997→      <p className="text-sm text-[var(--ws-text-tertiary)]">Nenhuma carta para revisar agora</p>
   998→      <button onClick={onBack} className="mt-4 text-sm text-[var(--ws-accent)] hover:underline">Voltar</button>
   999→    </motion.div>
  1000→  );
  1001→  if (done) return (
  1002→    <motion.div initial={{ opacity: 0, scale: 0.95 }} animate={{ opacity: 1, scale: 1 }} className="mx-auto max-w-md py-16 text-center">
  1003→      <EnsoCircle size={80} strokeWidth={2} color="var(--ws-verdigris)" imperfection={0.1} animate={false} />
  1004→      <h2 className="mt-6 font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)]">Sessao Completa!</h2>
  1005→      <p className="mt-2 text-sm text-[var(--ws-text-tertiary)]">Voce revisou {reviewed} cartas</p>
  1006→
  1007→      <div className="mt-8 grid grid-cols-3 gap-4">
  1008→        <WabiSabiCard hover={false}>
  1009→          <p className="font-serif-jp text-xl font-bold text-[var(--ws-accent)]">{avgRating}</p>
  1010→          <p className="mt-1 text-[10px] text-[var(--ws-text-tertiary)]">Media</p>
  1011→        </WabiSabiCard>
  1012→        <WabiSabiCard hover={false}>
  1013→          <p className="font-serif-jp text-xl font-bold text-[var(--ws-gold)]">{hardCount}</p>
  1014→          <p className="mt-1 text-[10px] text-[var(--ws-text-tertiary)]">Dificeis</p>
  1015→        </WabiSabiCard>
  1016→        <WabiSabiCard hover={false}>
  1017→          <p className="font-serif-jp text-xl font-bold text-[var(--ws-verdigris)]">{easyCount}</p>
  1018→          <p className="mt-1 text-[10px] text-[var(--ws-text-tertiary)]">Faceis</p>

  1022→      <ZenButton variant="primary" size="md" className="mt-8" onClick={onBack}>Voltar aos Flashcards</ZenButton>

  1026→  const card = cards[currentIndex];
  1027→  return (
  1028→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
  1029→      {/* Progress header */}
  1030→      <div className="mb-6">
  1031→        <div className="mb-4 flex items-center justify-between">
  1032→          <button onClick={onBack} className="flex items-center gap-2 text-sm text-[var(--ws-text-tertiary)] hover:text-[var(--ws-text-primary)]">

  1041→          <motion.div className="h-1.5 rounded-full bg-[var(--ws-accent)]" initial={{ width: 0 }} animate={{ width: `${((currentIndex + 1) / cards.length) * 100}%` }} transition={{ duration: 0.3 }} />
