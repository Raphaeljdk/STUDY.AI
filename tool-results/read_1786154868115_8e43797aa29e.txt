   277→// ===== BATTLE SCREEN =====
   278→function BattleScreen({
   279→  battle,
   280→  onFinish,
   281→}: {
   282→  battle: BattleState;
   283→  onFinish: (answers: { questionId: string; selectedIndex: number; confidence: number; correct: boolean }[]) => void;
   284→}) {
   285→  const [currentIndex, setCurrentIndex] = useState(0);
   286→  const [timer, setTimer] = useState(TIMER_DURATION);
   287→  const [selectedOption, setSelectedOption] = useState<number | null>(null);
   288→  const [showConfidence, setShowConfidence] = useState(false);
   289→  const [answers, setAnswers] = useState<(boolean | null)[]>(
   290→    Array(battle.questions.length).fill(null)
   291→  );
   292→  const [answeredOptions, setAnsweredOptions] = useState<Record<number, number>>({});
   293→  const [results, setResults] = useState<
   294→    { questionId: string; selectedIndex: number; confidence: number; correct: boolean }[]
   295→  >([]);
   296→  const [showResult, setShowResult] = useState(false);
   297→  const [correctFlash, setCorrectFlash] = useState<boolean | null>(null);
   298→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   299→
   300→  const currentQuestion = battle.questions[currentIndex];
   301→
   302→  // Timer
   303→  useEffect(() => {
   304→    if (showConfidence || showResult) return;
   305→    timerRef.current = setInterval(() => {
   306→      setTimer((prev) => {
   307→        if (prev <= 1) {
   308→          // Time's up - auto wrong answer
   309→          handleTimeout();
   310→          return 0;
   311→        }
   312→        return prev - 1;
   313→      });
   314→    }, 1000);
   315→    return () => {
   316→      if (timerRef.current) clearInterval(timerRef.current);
   317→    };
   318→  }, [currentIndex, showConfidence, showResult]);
   319→
   320→  const handleTimeout = useCallback(() => {
   321→    if (timerRef.current) clearInterval(timerRef.current);
   322→    setAnsweredOptions((prev) => ({ ...prev, [currentIndex]: -1 }));
   323→    setAnswers((prev) => {
   324→      const next = [...prev];
   325→      next[currentIndex] = false;
   326→      return next;
   327→    });
   328→    setCorrectFlash(false);
   329→    setTimeout(() => {
   330→      setCorrectFlash(null);
   331→      setShowConfidence(true);
   332→    }, 600);
   333→  }, [currentIndex]);
   334→
   335→  const handleSelectOption = (optionIndex: number) => {
   336→    if (selectedOption !== null || showConfidence) return;
   337→    if (timerRef.current) clearInterval(timerRef.current);
   338→
   339→    const isCorrect = optionIndex === currentQuestion.correctIndex;
   340→    setSelectedOption(optionIndex);
   341→    setAnsweredOptions((prev) => ({ ...prev, [currentIndex]: optionIndex }));
   342→    setAnswers((prev) => {
   343→      const next = [...prev];
   344→      next[currentIndex] = isCorrect;
   345→      return next;
   346→    });
   347→    setCorrectFlash(isCorrect);
   348→
   349→    setTimeout(() => {
   350→      setCorrectFlash(null);
   351→      setShowConfidence(true);
   352→    }, 600);
   353→  };
   354→
   355→  const handleConfidence = (confidence: number) => {
   356→    const optionIdx = selectedOption ?? -1;
   357→    const isCorrect = optionIdx === currentQuestion.correctIndex;
   358→
   359→    setResults((prev) => [
   360→      ...prev,
   361→      {
   362→        questionId: currentQuestion.id,
   363→        selectedIndex: optionIdx,
   364→        confidence,
   365→        correct: isCorrect,
   366→      },
   367→    ]);
   368→
   369→    setShowConfidence(false);
   370→    setSelectedOption(null);
   371→
   372→    if (currentIndex < battle.questions.length - 1) {
   373→      setCurrentIndex((prev) => prev + 1);
   374→      setTimer(TIMER_DURATION);
   375→    } else {
   376→      setShowResult(true);
   377→    }
   378→  };
   379→
   380→  // Compute result and submit
   381→  useEffect(() => {
   382→    if (showResult && results.length === battle.questions.length) {
   383→      const timer = setTimeout(() => onFinish(results), 500);
   384→      return () => clearTimeout(timer);
   385→    }
   386→  }, [showResult, results, battle.questions.length, onFinish]);
   387→
   388→  return (
   389→    <motion.div
   390→      initial={{ opacity: 0, x: 30 }}
   391→      animate={{ opacity: 1, x: 0 }}
   392→      exit={{ opacity: 0, x: -30 }}
   393→      className="space-y-5"
   394→    >
   395→      {/* Top bar */}
   396→      <div className="flex items-center justify-between">
   397→        <div className="flex items-center gap-2">
   398→          <span className="text-sm font-medium" style={{ color: 'var(--ws-text-secondary)' }}>
   399→            {battle.subject}
   400→          </span>
   401→        </div>
   402→        <div className="flex items-center gap-2">
   403→          <span className="text-xs font-medium px-2 py-1 rounded-full" style={{
   404→            background: 'color-mix(in srgb, var(--ws-accent) 10%, transparent)',
   405→            color: 'var(--ws-accent)',
   406→          }}>
   407→            {currentIndex + 1}/{battle.questions.length}
   408→          </span>
   409→        </div>
   410→      </div>
   411→
   412→      {/* Timer & Progress */}
   413→      <div className="flex flex-col items-center gap-4">
   414→        <CircularTimer seconds={timer} maxSeconds={TIMER_DURATION} />
   415→        <ProgressDots
   416→          current={currentIndex}
   417→          total={battle.questions.length}
   418→          answers={answers}
   419→        />
   420→      </div>
   421→
   422→      {/* Question */}
   423→      <AnimatePresence mode="wait">
   424→        <motion.div
   425→          key={currentIndex}
   426→          initial={{ opacity: 0, y: 10 }}
   427→          animate={{ opacity: 1, y: 0 }}
   428→          exit={{ opacity: 0, y: -10 }}
   429→          transition={{ duration: 0.25 }}
   430→          className="space-y-4"
   431→        >
   432→          {/* Flash overlay */}
   433→          {correctFlash !== null && (
   434→            <motion.div
   435→              initial={{ opacity: 0.3 }}
   436→              animate={{ opacity: 0 }}
   437→              transition={{ duration: 0.6 }}
   438→              className="fixed inset-0 z-50 pointer-events-none"
   439→              style={{
   440→                backgroundColor: correctFlash ? 'rgba(22, 163, 74, 0.15)' : 'rgba(220, 38, 38, 0.15)',
   441→              }}
   442→            />
   443→          )}
   444→
   445→          <div
   446→            className="glass-enhanced p-5"
   447→            style={{ borderRadius: 'var(--ws-radius-card)' }}
   448→          >
   449→            <h2
   450→              className="font-serif-jp text-lg font-semibold leading-relaxed"
   451→              style={{ color: 'var(--ws-text-primary)' }}
   452→            >
   453→              {currentQuestion.question}
   454→            </h2>
   455→          </div>
   456→
   457→          {/* Options */}
   458→          <div className="space-y-2.5">
   459→            {currentQuestion.options.map((option, idx) => {
   460→              const answeredIdx = answeredOptions[currentIndex];
   461→              const isAnswered = answeredIdx !== undefined;
   462→              const isSelected = answeredIdx === idx;
   463→              const isCorrectAnswer = idx === currentQuestion.correctIndex;
   464→              const isTimeout = answeredIdx === -1;
   465→
   466→              let optionBg = 'var(--ws-glass)';
   467→              let optionBorder = 'var(--ws-glass-border)';
   468→              let optionColor = 'var(--ws-text-primary)';

   472→                  optionBg = 'rgba(22, 163, 74, 0.1)';
   473→                  optionBorder = 'rgba(22, 163, 74, 0.3)';
   474→                  optionColor = '#16A34A';
   475→                } else if (isSelected && !isCorrectAnswer) {
   476→                  optionBg = 'rgba(220, 38, 38, 0.1)';
   477→                  optionBorder = 'rgba(220, 38, 38, 0.3)';
   478→                  optionColor = '#DC2626';
   479→                } else {

   778→// ===== MAIN COMPONENT =====
   779→type Screen = 'subject' | 'battle' | 'results';
   780→
   781→export function BattleView() {
   782→  const [screen, setScreen] = useState<Screen>('subject');
   783→  const [subject, setSubject] = useState('');
   784→  const [battle, setBattle] = useState<BattleState | null>(null);
   785→  const [result, setResult] = useState<BattleResult | null>(null);
   786→  const [history, setHistory] = useState<BattleHistoryItem[]>([]);
   787→  const [loading, setLoading] = useState(false);
   788→  const [loadingHistory, setLoadingHistory] = useState(true);
   789→
   790→  // Fetch history on mount
   791→  useEffect(() => {
   792→    const fetchHistory = async () => {
   793→      try {
   794→        const res = await fetch('/api/battle');
   795→        if (res.ok) {
   796→          const data = await res.json();
   797→          setHistory(data.battles || data || []);
   798→        }
   799→      } catch {
   800→        // silent
   801→      } finally {
   802→        setLoadingHistory(false);
   803→      }
   804→    };
   805→    fetchHistory();
   806→  }, []);
   807→
   808→  const handleStartBattle = async (selectedSubject: string) => {
   809→    setSubject(selectedSubject);
   810→    setLoading(true);
   811→    try {
   812→      const res = await fetch('/api/battle', {
   813→        method: 'POST',
   814→        headers: { 'Content-Type': 'application/json' },
   815→        body: JSON.stringify({ subject: selectedSubject }),
   816→      });
   817→      if (!res.ok) {
   818→        const err = await res.json().catch(() => ({}));
   819→        throw new Error(err.error || 'Failed to start battle');
   820→      }
   821→      const data = await res.json();
   822→      setBattle(data);
   823→      setScreen('battle');
   824→    } catch (err: any) {
   825→      toast({
   826→        title: 'Erro ao iniciar duelo',
   827→        description: err.message || 'Tente novamente.',
   828→      });
   829→    } finally {
   830→      setLoading(false);
   831→    }
   832→  };
   833→
   834→  const handleFinishBattle = async (
   835→    answers: { questionId: string; selectedIndex: number; confidence: number; correct: boolean }[]
   836→  ) => {
   837→    if (!battle) return;
   838→    try {
   839→      const res = await fetch('/api/battle/finish', {
   840→        method: 'POST',
   841→        headers: { 'Content-Type': 'application/json' },
   842→        body: JSON.stringify({
   843→          battleId: battle.id,
   844→          answers,
   845→        }),
   846→      });
   847→      if (res.ok) {
   848→        const data = await res.json();
   849→        setResult(data);
   850→        // Refresh history
   851→        const histRes = await fetch('/api/battle');
   852→        if (histRes.ok) {
   853→          const histData = await histRes.json();
   854→          setHistory(histData.battles || histData || []);
   855→        }
   856→      }
   857→    } catch {
   858→      // If API fails, compute locally
   859→      const score = answers.filter((a) => a.correct).length;
   860→      const localResult: BattleResult = {
   861→        id: battle.id,
   862→        subject,
   863→        score,
   864→        total: battle.questions.length,
   865→        xpEarned: score * 20,
   866→        accuracy: Math.round((score / battle.questions.length) * 100),
   867→        avgConfidence:
   868→          answers.length > 0
   869→            ? Math.round(answers.reduce((sum, a) => sum + a.confidence, 0) / answers.length * 10) / 10
   870→            : 0,
   871→        completedAt: new Date().toISOString(),
   872→      };
   873→      setResult(localResult);
   874→    }
   875→    setScreen('results');
   876→  };
   877→
   878→  const handlePlayAgain = () => {
   879→    setBattle(null);
   880→    setResult(null);
   881→    setScreen('subject');
   882→  };
   883→
   884→  const handleBackToStart = () => {
   885→    setBattle(null);
   886→    setResult(null);
   887→    setSubject('');
   888→    setScreen('subject');
   889→  };
   890→
   891→  return (
   892→    <div className="relative min-h-full pb-24">
   893→      {/* Header */}
   894→      <div className="mb-6">
   895→        <h1 className="font-serif-jp text-2xl font-bold mb-1" style={{ color: 'var(--ws-text-primary)' }}>
   896→          Duelo de Conhecimento
   897→        </h1>
   898→        <p className="text-sm" style={{ color: 'var(--ws-text-tertiary)' }}>
   899→          Teste seus conhecimentos em 1 minuto
   900→        </p>
   901→      </div>
   902→
   903→      <AnimatePresence mode="wait">
   904→        {screen === 'subject' && (
   905→          <SubjectSelection key="subject" onStart={handleStartBattle} />
   906→        )}
   907→        {screen === 'battle' && battle && (
   908→          <BattleScreen key="battle" battle={battle} onFinish={handleFinishBattle} />
   909→        )}
   910→        {screen === 'results' && result && (
   911→          <ResultsScreen
   912→            key="results"
   913→            result={result}
   914→            history={history}
   915→            onPlayAgain={handlePlayAgain}
   916→            onBack={handleBackToStart}
   917→          />
   918→        )}
   919→      </AnimatePresence>
   920→    </div>
   921→  );
   922→}
