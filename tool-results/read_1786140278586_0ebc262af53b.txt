   197→function getMasteryColor(mastery: number): string {
   198→  if (mastery >= 80) return '#22c55e';
   199→  if (mastery >= 60) return '#84cc16';
   200→  if (mastery >= 40) return '#f59e0b';
   201→  if (mastery >= 20) return '#f97316';
   202→  return '#ef4444';
   203→}
   204→
   205→function getMasteryLabel(mastery: number): string {
   206→  if (mastery >= 90) return 'Dominio total';
   207→  if (mastery >= 70) return 'Avancado';
   208→  if (mastery >= 50) return 'Intermediario';
   209→  if (mastery >= 25) return 'Iniciante';
   210→  return 'Nao iniciado';
   211→}
   212→
   213→function formatRelativeDate(dateStr: string | null | undefined): string {
   214→  if (!dateStr) return '—';
   215→  try {
   216→    return formatDistanceToNow(parseISO(dateStr), {
   217→      addSuffix: true,
   218→      locale: ptBR,
   219→    });
   220→  } catch {
   221→    return '—';
   222→  }
   223→}

   253→// Main export
   254→export function SubjectsView({ onNavigate }: SubjectsViewProps) {
   255→  // ===== STATE =====
   256→  const [subjects, setSubjects] = useState<SubjectWithMastery[]>([]);
   257→  const [loading, setLoading] = useState(true);
   258→  const [selectedSubjectId, setSelectedSubjectId] = useState<string | null>(null);
   259→  const [subjectDetail, setSubjectDetail] = useState<SubjectDetail | null>(null);
   260→  const [detailLoading, setDetailLoading] = useState(false);
   261→  const [detailTopics, setDetailTopics] = useState<Topic[]>([]);
   262→
   263→  // Add/Edit dialog state
   264→  const [showAddDialog, setShowAddDialog] = useState(false);
   265→  const [showEditDialog, setShowEditDialog] = useState(false);
   266→  const [formSubmitting, setFormSubmitting] = useState(false);
   267→  const [formName, setFormName] = useState('');
   268→  const [formDescription, setFormDescription] = useState('');
   269→  const [formColor, setFormColor] = useState('#f59e0b');
   270→  const [formIcon, setFormIcon] = useState('book');
   271→
   272→  // Delete confirmation
   273→  const [showDeleteDialog, setShowDeleteDialog] = useState(false);
   274→  const [deleteTarget, setDeleteTarget] = useState<string | null>(null);
   275→  const [deleting, setDeleting] = useState(false);
   276→
   277→  // Add topic form
   278→  const [showAddTopic, setShowAddTopic] = useState(false);
   279→  const [topicName, setTopicName] = useState('');
   280→  const [topicDescription, setTopicDescription] = useState('');
   281→  const [topicSubmitting, setTopicSubmitting] = useState(false);
   282→
   283→  // ===== FETCH SUBJECTS =====
   284→  const fetchSubjects = useCallback(async () => {
   285→    try {
   286→      setLoading(true);
   287→      const res = await fetch('/api/subjects');
   288→      if (!res.ok) throw new Error('Erro ao carregar materias');
   289→      const data = await res.json();
   290→
   291→      // Enrich subjects with mastery and pending task data
   292→      const enriched: SubjectWithMastery[] = await Promise.all(
   293→        (data.subjects || []).map(async (s: SubjectList) => {
   294→          try {
   295→            const detailRes = await fetch(`/api/subjects/${s.id}`);
   296→            if (!detailRes.ok) {
   297→              return {
   298→                ...s,
   299→                mastery: 0,
   300→                lastStudied: null,
   301→                pendingTasks: 0,
   302→              };
   303→            }
   304→            const detailData = await detailRes.json();
   305→            const detail: SubjectDetail = detailData.subject;
   306→            const topics = detail.topics || [];
   307→            const avgMastery =
   308→              topics.length > 0
   309→                ? Math.round(topics.reduce((sum: number, t: Topic) => sum + t.mastery, 0) / topics.length)
   310→                : 0;
   311→
   312→            // Get pending tasks count - we need to fetch tasks separately
   313→            let pendingTasks = 0;
   314→            let lastStudied: string | null = null;
   315→
   316→            // Use sessions count as a proxy - if sessions exist, it was studied
   317→            if (detail._count?.sessions > 0) {
   318→              lastStudied = detail.updatedAt || detail.createdAt;
   319→            }
   320→
   321→            return {
   322→              ...s,
   323→              mastery: avgMastery,
   324→              lastStudied,
   325→              pendingTasks,
   326→            };
   327→          } catch {
   328→            return {
   329→              ...s,
   330→              mastery: 0,
   331→              lastStudied: null,
   332→              pendingTasks: 0,
   333→            };
   334→          }
   335→        })
   336→      );
   337→
   338→      setSubjects(enriched);
   339→    } catch (err: any) {
   340→      toast({
   341→        title: 'Erro ao carregar',
   342→        description: err.message || 'Nao foi possivel carregar suas materias.',
   343→        variant: 'destructive',
   344→      });
   345→    } finally {
   346→      setLoading(false);
   347→    }
   348→  }, []);
   349→
   350→  useEffect(() => {
   351→    fetchSubjects();
   352→  }, [fetchSubjects]);
   353→
   354→  // ===== FETCH SUBJECT DETAIL =====
   355→  const fetchSubjectDetail = useCallback(async (subjectId: string) => {
   356→    try {
   357→      setDetailLoading(true);
   358→      const res = await fetch(`/api/subjects/${subjectId}`);
   359→      if (!res.ok) throw new Error('Materia nao encontrada');
   360→      const data = await res.json();
   361→      setSubjectDetail(data.subject);
   362→      setDetailTopics(data.subject.topics || []);
   363→    } catch (err: any) {
   364→      toast({
   365→        title: 'Erro',
   366→        description: err.message || 'Nao foi possivel carregar os detalhes.',
   367→        variant: 'destructive',
   368→      });
   369→    } finally {
   370→      setDetailLoading(false);
   371→    }
   372→  }, []);
   373→
   374→  useEffect(() => {
   375→    if (selectedSubjectId) {
   376→      fetchSubjectDetail(selectedSubjectId);
   377→      setShowAddTopic(false);
   378→      setTopicName('');
   379→      setTopicDescription('');
   380→    } else {
   381→      setSubjectDetail(null);
   382→      setDetailTopics([]);
   383→    }
   384→  }, [selectedSubjectId, fetchSubjectDetail]);
   385→
   386→  // ===== HANDLERS =====
   387→
   388→  // Open add dialog
   389→  const openAddDialog = () => {
   390→    setFormName('');
   391→    setFormDescription('');
   392→    setFormColor('#f59e0b');
   393→    setFormIcon('book');
   394→    setShowAddDialog(true);
   395→  };
   396→
   397→  // Open edit dialog
   398→  const openEditDialog = () => {
   399→    if (!subjectDetail) return;
   400→    setFormName(subjectDetail.name);
   401→    setFormDescription(subjectDetail.description || '');
   402→    setFormColor(subjectDetail.color);
   403→    setFormIcon(subjectDetail.icon);
   404→    setShowEditDialog(true);
   405→  };
   406→
   407→  // Submit add
   408→  const handleAddSubject = async () => {
   409→    if (!formName.trim()) {
   410→      toast({ title: 'Campo obrigatorio', description: 'Informe o nome da materia.', variant: 'destructive' });
   411→      return;
   412→    }
   413→    try {
   414→      setFormSubmitting(true);
   415→      const res = await fetch('/api/subjects', {
   416→        method: 'POST',
   417→        headers: { 'Content-Type': 'application/json' },
   418→        body: JSON.stringify({
   419→          name: formName.trim(),
   420→          description: formDescription.trim() || null,
   421→          color: formColor,
   422→          icon: formIcon,
   423→        }),
   424→      });
   425→      if (!res.ok) {
   426→        const err = await res.json();
   427→        throw new Error(err.error || 'Erro ao criar materia');
   428→      }
   429→      toast({ title: 'Materia criada!', description: `${formName.trim()} foi adicionada com sucesso.` });
   430→      setShowAddDialog(false);
   431→      fetchSubjects();
   432→    } catch (err: any) {
   433→      toast({ title: 'Erro ao criar', description: err.message, variant: 'destructive' });
   434→    } finally {
   435→      setFormSubmitting(false);
   436→    }
   437→  };
   438→
   439→  // Submit edit
   440→  const handleEditSubject = async () => {
   441→    if (!subjectDetail || !formName.trim()) {
   442→      toast({ title: 'Campo obrigatorio', description: 'Informe o nome da materia.', variant: 'destructive' });
   443→      return;
   444→    }
   445→    try {
   446→      setFormSubmitting(true);
   447→      const res = await fetch(`/api/subjects/${subjectDetail.id}`, {
   448→        method: 'PATCH',
   449→        headers: { 'Content-Type': 'application/json' },
   450→        body: JSON.stringify({
   451→          name: formName.trim(),
   452→          description: formDescription.trim() || null,
   453→          color: formColor,
   454→          icon: formIcon,
   455→        }),
   456→      });
   457→      if (!res.ok) {
   458→        const err = await res.json();
   459→        throw new Error(err.error || 'Erro ao atualizar materia');
   460→      }
   461→      toast({ title: 'Materia atualizada!', description: `${formName.trim()} foi salva com sucesso.` });
   462→      setShowEditDialog(false);
   463→      fetchSubjectDetail(subjectDetail.id);
   464→      fetchSubjects();
   465→    } catch (err: any) {
   466→      toast({ title: 'Erro ao atualizar', description: err.message, variant: 'destructive' });
   467→    } finally {
   468→      setFormSubmitting(false);
   469→    }
   470→  };
   471→
   472→  // Delete
   473→  const handleDelete = async () => {
   474→    if (!deleteTarget) return;
   475→    try {
   476→      setDeleting(true);
   477→      const res = await fetch(`/api/subjects/${deleteTarget}`, { method: 'DELETE' });
   478→      if (!res.ok) {
   479→        const err = await res.json();
   480→        throw new Error(err.error || 'Erro ao excluir materia');
   481→      }
   482→      toast({ title: 'Materia excluida', description: 'A materia foi removida com sucesso.' });
   483→      setShowDeleteDialog(false);
   484→      setDeleteTarget(null);
   485→      if (selectedSubjectId === deleteTarget) {
   486→        setSelectedSubjectId(null);
   487→      }
   488→      fetchSubjects();
   489→    } catch (err: any) {
   490→      toast({ title: 'Erro ao excluir', description: err.message, variant: 'destructive' });
   491→    } finally {
   492→      setDeleting(false);
   493→    }
   494→  };
   495→
   496→  // Add topic
   497→  const handleAddTopic = async () => {
   498→    if (!selectedSubjectId || !topicName.trim()) {
   499→      toast({ title: 'Campo obrigatorio', description: 'Informe o nome do topico.', variant: 'destructive' });
   500→      return;
   501→    }
   502→    try {
   503→      setTopicSubmitting(true);
   504→      const res = await fetch(`/api/subjects/${selectedSubjectId}/topics`, {
   505→        method: 'POST',
   506→        headers: { 'Content-Type': 'application/json' },
   507→        body: JSON.stringify({
   508→          name: topicName.trim(),
   509→          description: topicDescription.trim() || null,
   510→        }),
   511→      });
   512→      if (!res.ok) {
   513→        const err = await res.json();
   514→        throw new Error(err.error || 'Erro ao criar topico');
   515→      }
   516→      toast({ title: 'Topico adicionado!', description: `${topicName.trim()} foi criado com sucesso.` });
   517→      setTopicName('');
   518→      setTopicDescription('');
   519→      setShowAddTopic(false);
   520→      fetchSubjectDetail(selectedSubjectId);
   521→      fetchSubjects();
   522→    } catch (err: any) {
   523→      toast({ title: 'Erro ao criar topico', description: err.message, variant: 'destructive' });
   524→    } finally {
   525→      setTopicSubmitting(false);
   526→    }
   527→  };
   528→
   529→  // ===== RENDER: SUBJECT FORM DIALOG =====
   530→  const renderFormDialog = (
   531→    open: boolean,
   532→    onOpenChange: (v: boolean) => void,
   533→    title: string,
   534→    description: string,
   535→    onSubmit: () => void,
   536→    submitting: boolean,
   537→  ) => (
   538→    <Dialog open={open} onOpenChange={onOpenChange}>
   539→      <DialogContent className="sm:max-w-[480px] border-[var(--ws-glass-border)] bg-[var(--ws-glass)] backdrop-blur-xl"
   540→        style={{ borderRadius: 'var(--ws-radius-card)' }}
   541→      >
   542→        <DialogHeader>
   543→          <DialogTitle className="text-[var(--ws-text-primary)]">{title}</DialogTitle>
   544→          <DialogDescription className="text-[var(--ws-text-tertiary)]">{description}</DialogDescription>
   545→        </DialogHeader>
   546→
   547→        <div className="space-y-4 py-2">
   548→          {/* Name */}
   549→          <div className="space-y-2">
   550→            <Label htmlFor="subject-name" className="text-[var(--ws-text-secondary)]">Nome da materia *</Label>
   551→            <Input
   552→              id="subject-name"
   553→              value={formName}
   554→              onChange={(e) => setFormName(e.target.value)}
   555→              placeholder="Ex: Matematica, Portugues, Fisica..."
   556→              className="border-[var(--ws-glass-border)] bg-[var(--ws-bg)] text-[var(--ws-text-primary)] placeholder:text-[var(--ws-text-tertiary)]"
   557→              onKeyDown={(e) => e.key === 'Enter' && onSubmit()}
   558→            />
   559→          </div>
   560→
   561→          {/* Description */}
   562→          <div className="space-y-2">
   563→            <Label htmlFor="subject-desc" className="text-[var(--ws-text-secondary)]">Descricao</Label>
   564→            <Textarea
   565→              id="subject-desc"
   566→              value={formDescription}
   567→              onChange={(e) => setFormDescription(e.target.value)}
   568→              placeholder="Opcional: uma breve descricao da materia..."
   569→              rows={2}
   570→              className="border-[var(--ws-glass-border)] bg-[var(--ws-bg)] text-[var(--ws-text-primary)] placeholder:text-[var(--ws-text-tertiary)] resize-none"
   571→            />
   572→          </div>
   573→
   574→          {/* Color picker */}
   575→          <div className="space-y-2">
   576→            <Label className="text-[var(--ws-text-secondary)]">Cor</Label>
   577→            <div className="flex flex-wrap gap-2">
   578→              {COLOR_PRESETS.map((c) => (
   579→                <button
   580→                  key={c.value}
   581→                  type="button"
   582→                  onClick={() => setFormColor(c.value)}
   583→                  className="relative h-8 w-8 rounded-full transition-all duration-200 hover:scale-110 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-offset-[var(--ws-bg)]"
   584→                  style={{
   585→                    backgroundColor: c.value,
   586→                    focusRingColor: c.value,
   587→                    boxShadow: formColor === c.value ? `0 0 0 3px var(--ws-bg), 0 0 0 5px ${c.value}` : 'none',
   588→                  }}
   589→                  title={c.name}
   590→                  aria-label={`Cor ${c.name}`}
   591→                />
   592→              ))}
   593→            </div>
   594→          </div>
   595→
   596→          {/* Icon picker */}
