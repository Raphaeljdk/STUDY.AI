   431→  const [completingTaskId, setCompletingTaskId] = useState<string | null>(null);
   432→  const [xpCelebrationTaskId, setXpCelebrationTaskId] = useState<string | null>(null);

   437→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
   438→  const [taskToEdit, setTaskToEdit] = useState<Task | null>(null);
   439→  const [taskToDelete, setTaskToDelete] = useState<Task | null>(null);
   440→  const [submitting, setSubmitting] = useState(false);
   441→
   442→  // Form states
   443→  const [formTitle, setFormTitle] = useState('');
   444→  const [formDescription, setFormDescription] = useState('');
   445→  const [formSubjectId, setFormSubjectId] = useState('');
   446→  const [formPriority, setFormPriority] = useState('MEDIUM');
   447→  const [formDueDate, setFormDueDate] = useState('');
   448→  const [formEstimatedMinutes, setFormEstimatedMinutes] = useState('');
   449→  const [formStatus, setFormStatus] = useState('PENDING');
   450→  const [formTitleError, setFormTitleError] = useState('');
   451→
   452→  // ===== FETCH =====
   453→  const fetchTasks = useCallback(async () => {
   454→    try {
   455→      const params = new URLSearchParams();
   456→      if (activeFilter !== 'ALL') {
   457→        params.set('status', activeFilter);
   458→      }
   459→      const res = await fetch(`/api/tasks?${params.toString()}`);
   460→      if (!res.ok) throw new Error('Erro ao buscar tarefas');
   461→      const data = await res.json();
   462→      setTasks(data.tasks || []);
   463→    } catch {
   464→      toast({ title: 'Erro', description: 'Nao foi possivel carregar as tarefas.', variant: 'destructive' });
   465→    } finally {
   466→      setLoading(false);
   467→    }
   468→  }, [activeFilter]);
   469→
   470→  const fetchSubjects = useCallback(async () => {
   471→    try {
   472→      const res = await fetch('/api/subjects');
   473→      if (!res.ok) return;
   474→      const data = await res.json();
   475→      setSubjects(
   476→        (data.subjects || []).map((s: any) => ({
   477→          id: s.id,
   478→          name: s.name,
   479→          color: s.color,
   480→          icon: s.icon,
   481→        }))
   482→      );
   483→    } catch {
   484→      // subjects are optional
   485→    }
   486→  }, []);
   487→
   488→  useEffect(() => {
   489→    setLoading(true);
   490→    fetchTasks();
   491→  }, [fetchTasks]);
   492→
   493→  useEffect(() => {
   494→    fetchSubjects();
   495→  }, [fetchSubjects]);
   496→
   497→  // ===== HELPERS =====
   498→  const resetForm = useCallback(() => {
   499→    setFormTitle('');
   500→    setFormDescription('');
   501→    setFormSubjectId('');
   502→    setFormPriority('MEDIUM');
   503→    setFormDueDate('');
   504→    setFormEstimatedMinutes('');
   505→    setFormStatus('PENDING');
   506→    setFormTitleError('');
   507→  }, []);
   508→
   509→  const openEditDialog = useCallback((task: Task) => {
   510→    setTaskToEdit(task);
   511→    setFormTitle(task.title);
   512→    setFormDescription(task.description || '');
   513→    setFormSubjectId(task.subjectId || '');
   514→    setFormPriority(task.priority);
   515→    setFormDueDate(task.dueDate ? task.dueDate.split('T')[0] : '');
   516→    setFormEstimatedMinutes(task.estimatedMinutes?.toString() || '');
   517→    setFormStatus(task.status);
   518→    setFormTitleError('');
   519→    setEditDialogOpen(true);
   520→  }, []);
   521→
   522→  const formatDueDate = (dateStr: string) => {
   523→    try {
   524→      const date = parseISO(dateStr);
   525→      if (isToday(date)) return 'Hoje';
   526→      if (isPast(date)) return 'Atrasada';
   527→      return formatDistanceToNow(date, { addSuffix: true, locale: ptBR });
   528→    } catch {
   529→      return '';
   530→    }
   531→  };
   532→
   533→  const isOverdue = (dateStr: string) => {
   534→    try {
   535→      const date = parseISO(dateStr);
   536→      return isPast(date) && !isToday(date);
   537→    } catch {
   538→      return false;
   539→    }
   540→  };
   541→
   542→  const formatMinutes = (mins: number) => {
   543→    if (mins < 60) return `${mins}min`;
   544→    const h = Math.floor(mins / 60);
   545→    const m = mins % 60;
   546→    return m > 0 ? `${h}h${m}min` : `${h}h`;
   547→  };
   548→
   549→  // ===== SORT & FILTER =====
   550→  const filteredAndSorted = tasks
   551→    .filter((t) => {
   552→      if (activeFilter === 'ALL') return true;
   553→      return t.status === activeFilter;
   554→    })
   555→    .sort((a, b) => {
   556→      let cmp = 0;
   557→      switch (sortKey) {
   558→        case 'dueDate': {
   559→          const aD = a.dueDate ? parseISO(a.dueDate).getTime() : Infinity;
   560→          const bD = b.dueDate ? parseISO(b.dueDate).getTime() : Infinity;
   561→          cmp = aD - bD;
   562→          break;
   563→        }
   564→        case 'priority':
   565→          cmp = (PRIORITY_CONFIG[a.priority]?.weight || 0) - (PRIORITY_CONFIG[b.priority]?.weight || 0);
   566→          break;
   567→        case 'createdAt':
   568→          cmp = new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
   569→          break;
   570→        case 'title':
   571→          cmp = a.title.localeCompare(b.title, 'pt-BR');
   572→          break;
   573→      }
   574→      return sortDir === 'asc' ? cmp : -cmp;
   575→    });
   576→
   577→  const taskCounts = {
   578→    ALL: tasks.length,
   579→    PENDING: tasks.filter((t) => t.status === 'PENDING').length,

   581→    COMPLETED: tasks.filter((t) => t.status === 'COMPLETED').length,
   582→  };
