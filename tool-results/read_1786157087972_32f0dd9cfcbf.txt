   365→// ========== HELPER: Time-based greeting ==========
   366→function getGreeting(): string {
   367→  const h = new Date().getHours();
   368→  if (h < 12) return 'Bom dia';
   369→  if (h < 18) return 'Boa tarde';
   370→  return 'Boa noite';
   371→}
   372→
   373→function getGreetingSubtext(): string {
   374→  const h = new Date().getHours();
   375→  if (h < 12) return 'Comece o dia com foco e determinacao. Cada estudo e um passo adiante.';
   376→  if (h < 18) return 'A tarde e perfeita para revisar e consolidar o conhecimento.';
   377→  return 'Boas sessoes noturnas fixam melhor o que voce aprendeu.';
   378→}

   380→function formatReviewDate(dateStr: string): string {
   381→  const d = new Date(dateStr);
   382→  const now = new Date();
   383→  const diffMs = d.getTime() - now.getTime();
   384→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   385→
   386→  if (diffDays < 0) return 'Atrasado';
   387→  if (diffDays === 0) return 'Hoje';
   388→  if (diffDays === 1) return 'Amanha';
   389→  if (diffDays <= 7) return `em ${diffDays} dias`;
   390→  return d.toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit' });
   391→}

  1123→function PomodoroTimer() {
  1124→  const [mode, setMode] = useState<'focus' | 'short' | 'long'>('focus');
  1125→  const [timeLeft, setTimeLeft] = useState(25 * 60);
  1126→  const [isRunning, setIsRunning] = useState(false);
  1127→  const [totalToday, setTotalToday] = useState(0);
  1128→  const [sessions, setSessions] = useState<{ duration: number; createdAt: string }[]>([]);
  1129→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
  1130→  const durations = { focus: 25 * 60, short: 5 * 60, long: 15 * 60 };
  1131→  const modeLabels = { focus: 'Foco', short: 'Pausa Curta', long: 'Pausa Longa' };
  1132→  const fetchedRef = useRef(false);
  1133→
  1134→  useEffect(() => {
  1135→    if (fetchedRef.current) return;
  1136→    fetchedRef.current = true;
  1137→    fetch('/api/sessions').then(r => r.json()).then(d => {
  1138→      if (d.sessions) {
  1139→        const today = new Date().toDateString();
  1140→        setTotalToday(d.sessions.filter((s: any) => new Date(s.createdAt).toDateString() === today).reduce((sum: number, s: any) => sum + s.duration, 0));
  1141→        setSessions(d.sessions.slice(0, 10));
  1142→      }
  1143→    });
  1144→  }, []);
  1145→
  1146→  useEffect(() => {
  1147→    if (isRunning) {
  1148→      intervalRef.current = setInterval(() => {
  1149→        setTimeLeft(prev => {
  1150→          if (prev <= 1) {
  1151→            setIsRunning(false);
  1152→            if (mode === 'focus') {
  1153→              const dur = durations.focus - prev;
  1154→              fetch('/api/sessions', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ duration: dur, type: 'pomodoro' }) });
  1155→              setTotalToday(t => t + dur);
  1156→            }
  1157→            return 0;
  1158→          }
  1159→          return prev - 1;
  1160→        });
  1161→      }, 1000);
  1162→    } else if (intervalRef.current) clearInterval(intervalRef.current);
  1163→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
  1164→  }, [isRunning, mode, durations.focus]);
  1165→
  1166→  const switchMode = (m: 'focus' | 'short' | 'long') => { setMode(m); setTimeLeft(durations[m]); setIsRunning(false); };
  1167→  const minutes = Math.floor(timeLeft / 60);
  1168→  const seconds = timeLeft % 60;
  1169→  const progress = 1 - timeLeft / durations[mode];
  1170→
  1171→  return (
  1172→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
  1173→      <div className="mb-8">
  1174→        <h1 className="font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)] lg:text-3xl"><Timer size={24} className="mr-2 inline text-[var(--ws-accent)]" strokeWidth={1.5} />Pomodoro Timer</h1>
  1175→        <p className="mt-1 text-sm text-[var(--ws-text-tertiary)]">Estude com foco usando a tecnica Pomodoro</p>
  1176→      </div>
  1177→      <div className="mx-auto max-w-md">
  1178→        <div className="mb-8 flex justify-center gap-2">
  1179→          {(['focus', 'short', 'long'] as const).map(m => (
  1180→            <button key={m} onClick={() => switchMode(m)} className={`rounded-ws-button px-4 py-2 text-sm font-medium transition-colors ${mode === m ? 'bg-[var(--ws-accent)] text-[var(--ws-text-on-dark)]' : 'border border-[var(--ws-glass-border)] text-[var(--ws-text-tertiary)] hover:text-[var(--ws-text-secondary)]'}`}>
  1181→              {modeLabels[m]}
  1182→            </button>
  1183→          ))}
  1184→        </div>
  1185→        <WabiSabiCard className="mb-8" hover={false}>
  1186→          <div className="flex flex-col items-center py-8">
  1187→            <div className="relative flex h-56 w-56 items-center justify-center">

  1190→                <motion.circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-accent)" strokeWidth="4" strokeLinecap="round" strokeDasharray={2 * Math.PI * 90} initial={{ strokeDashoffset: 2 * Math.PI * 90 }} animate={{ strokeDashoffset: 2 * Math.PI * 90 * (1 - progress) }} transition={{ duration: 0.5 }} />
  1191→              </svg>

  1193→                <p className="font-serif-jp text-5xl font-bold text-[var(--ws-text-primary)]">{String(minutes).padStart(2, '0')}:{String(seconds).padStart(2, '0')}</p>
