    23→// Lightweight error boundary for individual sections
    24→class SectionErrorBoundary extends Component<{ children: ReactNode; fallback?: ReactNode; name?: string }, { hasError: boolean; error: Error | null }> {
    25→  constructor(props: { children: ReactNode; fallback?: ReactNode; name?: string }) {
    26→    super(props);
    27→    this.state = { hasError: false, error: null };
    28→  }
    29→  static getDerivedStateFromError(error: Error) {
    30→    return { hasError: true, error };
    31→  }
    32→  componentDidCatch(error: Error, info: React.ErrorInfo) {
    33→    console.error(`[SectionErrorBoundary${this.props.name ? ` (${this.props.name})` : ''}]`, error, info.componentStack);
    34→  }
    35→  render() {
    36→    if (this.state.hasError) {
    37→      if (this.props.fallback) return this.props.fallback;
    38→      return (
    39→        <div className="flex min-h-[300px] items-center justify-center rounded-lg border border-dashed border-[var(--ws-glass-border)] p-6">
    40→          <div className="text-center">
    41→            <AlertCircle size={24} className="mx-auto mb-3 text-[var(--ws-accent)]" />
    42→            <p className="text-sm font-medium text-[var(--ws-text-primary)]">Erro ao carregar esta secao</p>
    43→            <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{this.state.error?.message || 'Erro desconhecido'}</p>
    44→            <button
    45→              onClick={() => { this.setState({ hasError: false, error: null }); }}
    46→              className="mt-3 text-xs font-medium text-[var(--ws-accent)] hover:underline"
    47→            >
    48→              Tentar novamente
    49→            </button>
    50→          </div>
    51→        </div>
    52→      );
    53→    }
    54→    return this.props.children;
    55→  }
    56→}

    58→// Safe rich text editor with fallback to plain textarea
    59→function SafeEditor({ content, onChange, placeholder, onError }: { content: string; onChange: (html: string) => void; placeholder: string; onError?: (msg: string) => void }) {
    60→  const [editorReady, setEditorReady] = useState(false);
    61→  const [editorFailed, setEditorFailed] = useState(false);
    62→  const [EditorComp, setEditorComp] = useState<React.ComponentType<{ content: string; onChange: (html: string) => void; placeholder: string }> | null>(null);
    63→  const saveTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    64→
    65→  useEffect(() => {
    66→    import('./RichTextEditor')
    67→      .then(m => { setEditorComp(() => m.RichTextEditor); setEditorReady(true); })
    68→      .catch((err) => {
    69→        console.warn('[SafeEditor] RichTextEditor failed to load, using fallback:', err);
    70→        setEditorFailed(true);
    71→        onError?.(err?.message || 'Editor falhou ao carregar');
    72→      });
    73→  }, []);
    74→
    75→  const handleChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => {
    76→    const val = e.target.value;
    77→    if (saveTimer.current) clearTimeout(saveTimer.current);
    78→    saveTimer.current = setTimeout(() => onChange(val), 500);
    79→  }, [onChange]);
    80→
    81→  if (editorFailed) {
    82→    return (
    83→      <div>
    84→        <div className="mb-3 flex items-center gap-2 rounded-lg bg-[var(--ws-accent)]/10 px-3 py-2 text-xs text-[var(--ws-accent)]">
    85→          <AlertCircle size={14} /> Editor avancado indisponivel. Usando editor simples.
    86→        </div>
    87→        <textarea
    88→          value={content}
    89→          onChange={handleChange}
    90→          placeholder={placeholder}
    91→          className="min-h-[400px] w-full resize-none bg-transparent p-5 text-sm leading-relaxed text-[var(--ws-text-primary)] placeholder-[var(--ws-text-tertiary)] outline-none"
    92→        />
    93→      </div>
    94→    );
    95→  }
    96→
    97→  if (!editorReady) {
    98→    return (
    99→      <div className="flex min-h-[400px] items-center justify-center">
   100→        <div className="text-center">
   101→          <div className="mx-auto mb-3 h-6 w-6 animate-spin rounded-full border-2 border-[var(--ws-glass-border)] border-t-[var(--ws-accent)]" />
   102→          <p className="text-xs text-[var(--ws-text-tertiary)]">Carregando editor...</p>
   103→        </div>
   104→      </div>
   105→    );
   106→  }
   107→
   108→  if (EditorComp) return <EditorComp content={content} onChange={onChange} placeholder={placeholder} />;
   109→  return null;
   110→}

   134→function useActiveUsers() {
   135→  const [count, setCount] = useState(0);
   136→  useEffect(() => {
   137→    const base = 87;
   138→    const tick = () => setCount(base + Math.floor(Math.random() * 30));
   139→    tick();
   140→    const id = setInterval(tick, 5000);
   141→    return () => clearInterval(id);
   142→  }, []);
   143→  return count;

   146→// ========== SWIPE HOOK ==========
   147→function useSwipeGestures(onSwipeLeft: () => void, onSwipeRight: () => void) {
   148→  const touchStartX = useRef(0);
   149→  const touchEndX = useRef(0);
   150→
   151→  const onTouchStart = useCallback((e: React.TouchEvent) => {
   152→    touchStartX.current = e.changedTouches[0].screenX;
   153→  }, []);
   154→
   155→  const onTouchEnd = useCallback((e: React.TouchEvent) => {
   156→    touchEndX.current = e.changedTouches[0].screenX;
   157→    const diff = touchEndX.current - touchStartX.current;
   158→    if (Math.abs(diff) > 80) {
   159→      if (diff < 0) onSwipeLeft();
   160→      else onSwipeRight();
   161→    }
   162→  }, [onSwipeLeft, onSwipeRight]);
   163→
   164→  return { onTouchStart, onTouchEnd };
   165→}

   296→function formatReviewDate(dateStr: string): string {
   297→  const d = new Date(dateStr);
   298→  const now = new Date();
   299→  const diffMs = d.getTime() - now.getTime();
   300→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   301→
   302→  if (diffDays < 0) return 'Atrasado';
   303→  if (diffDays === 0) return 'Hoje';
   304→  if (diffDays === 1) return 'Amanha';
   305→  if (diffDays <= 7) return `em ${diffDays} dias`;
   306→  return d.toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit' });
   307→}

  1089→// ========== FLASHCARD REVIEWER ==========
  1090→function FlashcardReviewer({ onBack }: { onBack: () => void }) {
  1091→  const [cards, setCards] = useState<FlashcardItem[]>([]);
  1092→  const [loading, setLoading] = useState(true);
  1093→  const [currentIndex, setCurrentIndex] = useState(0);
  1094→  const [flipped, setFlipped] = useState(false);
  1095→  const [done, setDone] = useState(false);
  1096→  const [reviewed, setReviewed] = useState(0);
  1097→  const [ratings, setRatings] = useState<number[]>([]);
  1098→  const fetchedRef = useRef(false);
  1099→
  1100→  useEffect(() => {
  1101→    if (fetchedRef.current) return;
  1102→    fetchedRef.current = true;
  1103→    fetch('/api/flashcards?due=true').then(r => r.json()).then(d => {
  1104→      if (d.flashcards) setCards(d.flashcards);
  1105→    }).catch(() => {}).finally(() => setLoading(false));
  1106→  }, []);
  1107→
  1108→  const handleReview = useCallback(async (quality: number) => {
  1109→    const card = cards[currentIndex];
  1110→    if (!card) return;
  1111→    try {
  1112→      await fetch(`/api/flashcards/${card.id}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ review: quality }) });
  1113→    } catch {}
  1114→    setReviewed(prev => prev + 1);
  1115→    setRatings(prev => [...prev, quality]);
  1116→    setFlipped(false);
  1117→    if (currentIndex + 1 >= cards.length) setDone(true);
  1118→    else setCurrentIndex(prev => prev + 1);
  1119→  }, [cards, currentIndex]);
  1120→
  1121→  // Keyboard shortcuts
  1122→  useEffect(() => {
  1123→    if (done || !flipped || cards.length === 0) return;
  1124→    const handleKey = (e: KeyboardEvent) => {
  1125→      if (e.key === '1') handleReview(1);
  1126→      else if (e.key === '2') handleReview(2);
  1127→      else if (e.key === '3') handleReview(3);
  1128→      else if (e.key === '4') handleReview(5);
  1129→    };
  1130→    window.addEventListener('keydown', handleKey);
  1131→    return () => window.removeEventListener('keydown', handleKey);
  1132→  }, [flipped, done, cards.length, handleReview]);
  1133→
  1134→  // Completion stats
  1135→  const avgRating = ratings.length > 0 ? (ratings.reduce((a, b) => a + b, 0) / ratings.length).toFixed(1) : '0';
  1136→  const hardCount = ratings.filter(r => r <= 2).length;
  1137→  const easyCount = ratings.filter(r => r >= 4).length;
  1138→
  1139→  if (loading) return <div className="flex items-center justify-center py-16"><Loader2 size={24} className="animate-spin text-[var(--ws-text-tertiary)]" /></div>;

  1241→function PomodoroTimer() {
  1242→  const [mode, setMode] = useState<'focus' | 'short' | 'long'>('focus');
  1243→  const [timeLeft, setTimeLeft] = useState(25 * 60);
  1244→  const [isRunning, setIsRunning] = useState(false);
  1245→  const [totalToday, setTotalToday] = useState(0);
  1246→  const [sessions, setSessions] = useState<{ duration: number; createdAt: string }[]>([]);
  1247→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
  1248→  const durations = { focus: 25 * 60, short: 5 * 60, long: 15 * 60 };
  1249→  const modeLabels = { focus: 'Foco', short: 'Pausa Curta', long: 'Pausa Longa' };
  1250→  const fetchedRef = useRef(false);
  1251→
  1252→  useEffect(() => {
  1253→    if (fetchedRef.current) return;
  1254→    fetchedRef.current = true;
  1255→    fetch('/api/sessions').then(r => r.json()).then(d => {
  1256→      if (d.sessions) {
  1257→        const today = new Date().toDateString();
  1258→        setTotalToday(d.sessions.filter((s: any) => new Date(s.createdAt).toDateString() === today).reduce((sum: number, s: any) => sum + s.duration, 0));
  1259→        setSessions(d.sessions.slice(0, 10));
  1260→      }
  1261→    });
  1262→  }, []);
  1263→
  1264→  useEffect(() => {
  1265→    if (isRunning) {
  1266→      intervalRef.current = setInterval(() => {
  1267→        setTimeLeft(prev => {
  1268→          if (prev <= 1) {
  1269→            setIsRunning(false);
  1270→            if (mode === 'focus') {
  1271→              const dur = durations.focus - prev;
  1272→              fetch('/api/sessions', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ duration: dur, type: 'pomodoro' }) });
  1273→              setTotalToday(t => t + dur);
  1274→            }
  1275→            return 0;
  1276→          }
  1277→          return prev - 1;
  1278→        });
  1279→      }, 1000);
  1280→    } else if (intervalRef.current) clearInterval(intervalRef.current);
  1281→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
  1282→  }, [isRunning, mode, durations.focus]);
  1283→
  1284→  const switchMode = (m: 'focus' | 'short' | 'long') => { setMode(m); setTimeLeft(durations[m]); setIsRunning(false); };
  1285→  const minutes = Math.floor(timeLeft / 60);
  1286→  const seconds = timeLeft % 60;
  1287→  const progress = 1 - timeLeft / durations[mode];
  1288→
  1289→  return (
  1290→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
  1291→      <div className="mb-8">
  1292→        <h1 className="font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)] lg:text-3xl"><Timer size={24} className="mr-2 inline text-[var(--ws-accent)]" strokeWidth={1.5} />Pomodoro Timer</h1>
  1293→        <p className="mt-1 text-sm text-[var(--ws-text-tertiary)]">Estude com foco usando a tecnica Pomodoro</p>
  1294→      </div>
  1295→      <div className="mx-auto max-w-md">
  1296→        <div className="mb-8 flex justify-center gap-2">
  1297→          {(['focus', 'short', 'long'] as const).map(m => (
  1298→            <button key={m} onClick={() => switchMode(m)} className={`rounded-ws-button px-4 py-2 text-sm font-medium transition-colors ${mode === m ? 'bg-[var(--ws-accent)] text-[var(--ws-text-on-dark)]' : 'border border-[var(--ws-glass-border)] text-[var(--ws-text-tertiary)] hover:text-[var(--ws-text-secondary)]'}`}>
  1299→              {modeLabels[m]}
  1300→            </button>
  1301→          ))}
  1302→        </div>
  1303→        <WabiSabiCard className="mb-8" hover={false}>
  1304→          <div className="flex flex-col items-center py-8">
  1305→            <div className="relative flex h-56 w-56 items-center justify-center">
  1306→              <svg className="absolute inset-0 -rotate-90" viewBox="0 0 200 200">
  1307→                <circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-glass-border)" strokeWidth="4" />
  1308→                <motion.circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-accent)" strokeWidth="4" strokeLinecap="round" strokeDasharray={2 * Math.PI * 90} initial={{ strokeDashoffset: 2 * Math.PI * 90 }} animate={{ strokeDashoffset: 2 * Math.PI * 90 * (1 - progress) }} transition={{ duration: 0.5 }} />
  1309→              </svg>
  1310→              <div className="text-center">
  1311→                <p className="font-serif-jp text-5xl font-bold text-[var(--ws-text-primary)]">{String(minutes).padStart(2, '0')}:{String(seconds).padStart(2, '0')}</p>

  1321→        <WabiSabiCard hover={false}>
