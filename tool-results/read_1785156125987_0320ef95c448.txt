    21→// Lightweight error boundary for individual sections
    22→class SectionErrorBoundary extends Component<{ children: ReactNode; fallback?: ReactNode; name?: string }, { hasError: boolean; error: Error | null }> {
    23→  constructor(props: { children: ReactNode; fallback?: ReactNode; name?: string }) {
    24→    super(props);
    25→    this.state = { hasError: false, error: null };
    26→  }
    27→  static getDerivedStateFromError(error: Error) {
    28→    return { hasError: true, error };
    29→  }
    30→  componentDidCatch(error: Error, info: React.ErrorInfo) {
    31→    console.error(`[SectionErrorBoundary${this.props.name ? ` (${this.props.name})` : ''}]`, error, info.componentStack);
    32→  }
    33→  render() {
    34→    if (this.state.hasError) {
    35→      if (this.props.fallback) return this.props.fallback;
    36→      return (
    37→        <div className="flex min-h-[300px] items-center justify-center rounded-lg border border-dashed border-[var(--ws-glass-border)] p-6">
    38→          <div className="text-center">
    39→            <AlertCircle size={24} className="mx-auto mb-3 text-[var(--ws-accent)]" />
    40→            <p className="text-sm font-medium text-[var(--ws-text-primary)]">Erro ao carregar esta secao</p>
    41→            <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{this.state.error?.message || 'Erro desconhecido'}</p>
    42→            <button
    43→              onClick={() => { this.setState({ hasError: false, error: null }); }}
    44→              className="mt-3 text-xs font-medium text-[var(--ws-accent)] hover:underline"
    45→            >
    46→              Tentar novamente
    47→            </button>
    48→          </div>
    49→        </div>
    50→      );
    51→    }
    52→    return this.props.children;
    53→  }
    54→}

    56→// Safe rich text editor with fallback to plain textarea
    57→function SafeEditor({ content, onChange, placeholder, onError }: { content: string; onChange: (html: string) => void; placeholder: string; onError?: (msg: string) => void }) {
    58→  const [editorReady, setEditorReady] = useState(false);
    59→  const [editorFailed, setEditorFailed] = useState(false);
    60→  const [EditorComp, setEditorComp] = useState<React.ComponentType<{ content: string; onChange: (html: string) => void; placeholder: string }> | null>(null);
    61→  const saveTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    62→
    63→  useEffect(() => {
    64→    import('./RichTextEditor')
    65→      .then(m => { setEditorComp(() => m.RichTextEditor); setEditorReady(true); })
    66→      .catch((err) => {
    67→        console.warn('[SafeEditor] RichTextEditor failed to load, using fallback:', err);
    68→        setEditorFailed(true);
    69→        onError?.(err?.message || 'Editor falhou ao carregar');
    70→      });
    71→  }, []);
    72→
    73→  const handleChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => {
    74→    const val = e.target.value;
    75→    if (saveTimer.current) clearTimeout(saveTimer.current);
    76→    saveTimer.current = setTimeout(() => onChange(val), 500);
    77→  }, [onChange]);
    78→
    79→  if (editorFailed) {
    80→    return (
    81→      <div>
    82→        <div className="mb-3 flex items-center gap-2 rounded-lg bg-[var(--ws-accent)]/10 px-3 py-2 text-xs text-[var(--ws-accent)]">
    83→          <AlertCircle size={14} /> Editor avancado indisponivel. Usando editor simples.
    84→        </div>
    85→        <textarea
    86→          value={content}
    87→          onChange={handleChange}
    88→          placeholder={placeholder}
    89→          className="min-h-[400px] w-full resize-none bg-transparent p-5 text-sm leading-relaxed text-[var(--ws-text-primary)] placeholder-[var(--ws-text-tertiary)] outline-none"
    90→        />
    91→      </div>
    92→    );
    93→  }
    94→
    95→  if (!editorReady) {
    96→    return (
    97→      <div className="flex min-h-[400px] items-center justify-center">
    98→        <div className="text-center">
    99→          <div className="mx-auto mb-3 h-6 w-6 animate-spin rounded-full border-2 border-[var(--ws-glass-border)] border-t-[var(--ws-accent)]" />
   100→          <p className="text-xs text-[var(--ws-text-tertiary)]">Carregando editor...</p>
   101→        </div>
   102→      </div>
   103→    );
   104→  }
   105→
   106→  if (EditorComp) return <EditorComp content={content} onChange={onChange} placeholder={placeholder} />;
   107→  return null;
   108→}

   131→// ========== ACTIVE USERS HOOK ==========
   132→function useActiveUsers() {
   133→  const [count, setCount] = useState(0);
   134→  useEffect(() => {
   135→    const base = 87;
   136→    const tick = () => setCount(base + Math.floor(Math.random() * 30));
   137→    tick();
   138→    const id = setInterval(tick, 5000);
   139→    return () => clearInterval(id);
   140→  }, []);
   141→  return count;
   142→}

   144→// ========== SWIPE HOOK ==========
   145→function useSwipeGestures(onSwipeLeft: () => void, onSwipeRight: () => void) {
   146→  const touchStartX = useRef(0);
   147→  const touchEndX = useRef(0);
   148→
   149→  const onTouchStart = useCallback((e: React.TouchEvent) => {
   150→    touchStartX.current = e.changedTouches[0].screenX;
   151→  }, []);
   152→
   153→  const onTouchEnd = useCallback((e: React.TouchEvent) => {
   154→    touchEndX.current = e.changedTouches[0].screenX;
   155→    const diff = touchEndX.current - touchStartX.current;
   156→    if (Math.abs(diff) > 80) {
   157→      if (diff < 0) onSwipeLeft();
   158→      else onSwipeRight();
   159→    }
   160→  }, [onSwipeLeft, onSwipeRight]);
   161→
   162→  return { onTouchStart, onTouchEnd };
   163→}

   273→function formatReviewDate(dateStr: string): string {
   274→  const d = new Date(dateStr);
   275→  const now = new Date();
   276→  const diffMs = d.getTime() - now.getTime();
   277→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   278→
   279→  if (diffDays < 0) return 'Atrasado';
   280→  if (diffDays === 0) return 'Hoje';
   281→  if (diffDays === 1) return 'Amanha';
   282→  if (diffDays <= 7) return `em ${diffDays} dias`;
   283→  return d.toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit' });
   284→}

   650→function NotebookEditor({ notebookId, onBack }: { notebookId: string; onBack: () => void }) {
   651→  const [notebook, setNotebook] = useState<NotebookItem | null>(null);
   652→  const [loading, setLoading] = useState(true);
   653→  const [saving, setSaving] = useState(false);
   654→  const [content, setContent] = useState('');

   656→  const [editTitle, setEditTitle] = useState(false);
   657→  const [showFcForm, setShowFcForm] = useState(false);
   658→  const [fcFront, setFcFront] = useState('');
   659→  const [fcBack, setFcBack] = useState('');
   660→  const [creatingFc, setCreatingFc] = useState(false);
   661→  const [generating, setGenerating] = useState(false);
   662→  const [editorError, setEditorError] = useState<string | null>(null);
   663→  const saveTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   664→  const fetchedRef = useRef(false);
   665→
   666→  useEffect(() => {
   667→    if (fetchedRef.current) return;
   668→    fetchedRef.current = true;
   669→    fetch(`/api/notebooks/${notebookId}`)
   670→      .then(r => r.json())
   671→      .then(d => {
   672→        if (d.notebook) { setNotebook(d.notebook); setTitle(d.notebook.title); setContent(d.notebook.content || ''); }
   673→        else { toast({ title: 'Caderno nao encontrado', description: 'O caderno solicitado nao existe.', variant: 'destructive' }); }
   674→      })
   675→      .catch(err => {
   676→        console.error('[NotebookEditor] fetch error:', err);
   677→        toast({ title: 'Erro ao carregar', description: 'Nao foi possivel carregar o caderno.', variant: 'destructive' });
   678→      })
   679→      .finally(() => setLoading(false));
   680→  }, [notebookId]);
   681→
   682→  const saveContent = useCallback(async (newContent: string, newTitle?: string) => {
   683→    setSaving(true);
   684→    try {
   685→      const data: any = { content: newContent };
   686→      if (newTitle !== undefined) data.title = newTitle;
   687→      const res = await fetch(`/api/notebooks/${notebookId}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) });
   688→      const d = await res.json();
   689→      if (d.notebook) setNotebook(d.notebook);
   690→    } catch (err) {
   691→      console.error('[NotebookEditor] save error:', err);
   692→    }
   693→    setSaving(false);
   694→  }, [notebookId]);
   695→

   697→    setContent(html);
   698→    if (saveTimerRef.current) clearTimeout(saveTimerRef.current);
   699→    saveTimerRef.current = setTimeout(() => saveContent(html), 1500);
   700→  };
   701→
   702→  const handleTitleSave = () => {
   703→    if (title.trim()) { saveContent(content, title.trim()); setEditTitle(false); }
   704→  };

   707→    e.preventDefault();
   708→    if (!fcFront.trim() || !fcBack.trim()) return;
   709→    setCreatingFc(true);

   711→      const res = await fetch('/api/flashcards', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ front: fcFront, back: fcBack, notebookId }) });
   712→      const data = await res.json();

   715→        setFcFront(''); setFcBack(''); setShowFcForm(false);
   716→      } else {
   717→        toast({ title: 'Erro ao criar flashcard', description: data.error || 'Tente novamente.', variant: 'destructive' });
   718→      }
   719→    } catch (err) {
   720→      console.error('[NotebookEditor] create fc error:', err);
   721→      toast({ title: 'Erro de conexao', description: 'Nao foi possivel criar o flashcard.', variant: 'destructive' });
   722→    }
   723→    setCreatingFc(false);
   724→  };
   725→

   728→      await fetch(`/api/flashcards/${fcId}`, { method: 'DELETE' });
   729→      setNotebook(prev => prev ? { ...prev, flashcards: prev.flashcards?.filter(f => f.id !== fcId) || [] } : prev);
   730→    } catch (err) {
   731→      console.error('[NotebookEditor] delete fc error:', err);
   732→    }
   733→  };
   734→
   735→  const handleGenerateFc = async () => {

  1052→// ========== FLASHCARD REVIEWER ==========
  1053→function FlashcardReviewer({ onBack }: { onBack: () => void }) {
  1054→  const [cards, setCards] = useState<FlashcardItem[]>([]);
  1055→  const [loading, setLoading] = useState(true);
  1056→  const [currentIndex, setCurrentIndex] = useState(0);
  1057→  const [flipped, setFlipped] = useState(false);
  1058→  const [done, setDone] = useState(false);
  1059→  const [reviewed, setReviewed] = useState(0);
  1060→  const [ratings, setRatings] = useState<number[]>([]);
  1061→  const fetchedRef = useRef(false);
  1062→
  1063→  useEffect(() => {
  1064→    if (fetchedRef.current) return;
  1065→    fetchedRef.current = true;
  1066→    fetch('/api/flashcards?due=true').then(r => r.json()).then(d => {
  1067→      if (d.flashcards) setCards(d.flashcards);
  1068→    }).catch(() => {}).finally(() => setLoading(false));
  1069→  }, []);
  1070→
  1071→  const handleReview = useCallback(async (quality: number) => {
  1072→    const card = cards[currentIndex];
  1073→    if (!card) return;
  1074→    try {
  1075→      await fetch(`/api/flashcards/${card.id}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ review: quality }) });
  1076→    } catch {}
  1077→    setReviewed(prev => prev + 1);
  1078→    setRatings(prev => [...prev, quality]);
  1079→    setFlipped(false);
  1080→    if (currentIndex + 1 >= cards.length) setDone(true);
  1081→    else setCurrentIndex(prev => prev + 1);
  1082→  }, [cards, currentIndex]);
  1083→
  1084→  // Keyboard shortcuts
  1085→  useEffect(() => {
  1086→    if (done || !flipped || cards.length === 0) return;
  1087→    const handleKey = (e: KeyboardEvent) => {
  1088→      if (e.key === '1') handleReview(1);
  1089→      else if (e.key === '2') handleReview(2);
  1090→      else if (e.key === '3') handleReview(3);
  1091→      else if (e.key === '4') handleReview(5);
  1092→    };
  1093→    window.addEventListener('keydown', handleKey);
  1094→    return () => window.removeEventListener('keydown', handleKey);
  1095→  }, [flipped, done, cards.length, handleReview]);
  1096→
  1097→  // Completion stats
  1098→  const avgRating = ratings.length > 0 ? (ratings.reduce((a, b) => a + b, 0) / ratings.length).toFixed(1) : '0';
  1099→  const hardCount = ratings.filter(r => r <= 2).length;
  1100→  const easyCount = ratings.filter(r => r >= 4).length;
  1101→
  1102→  if (loading) return <div className="flex items-center justify-center py-16"><Loader2 size={24} className="animate-spin text-[var(--ws-text-tertiary)]" /></div>;
  1103→  if (cards.length === 0) return (
  1104→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="py-16 text-center">
  1105→      <Check size={48} className="mx-auto mb-4 text-[var(--ws-verdigris)]" strokeWidth={1} />
  1106→      <p className="text-sm text-[var(--ws-text-tertiary)]">Nenhuma carta para revisar agora</p>
  1107→      <button onClick={onBack} className="mt-4 text-sm text-[var(--ws-accent)] hover:underline">Voltar</button>
  1108→    </motion.div>
  1109→  );
  1110→  if (done) return (
  1111→    <motion.div initial={{ opacity: 0, scale: 0.95 }} animate={{ opacity: 1, scale: 1 }} className="mx-auto max-w-md py-16 text-center">
  1112→      <EnsoCircle size={80} strokeWidth={2} color="var(--ws-verdigris)" imperfection={0.1} animate={false} />
  1113→      <h2 className="mt-6 font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)]">Sessao Completa!</h2>
  1114→      <p className="mt-2 text-sm text-[var(--ws-text-tertiary)]">Voce revisou {reviewed} cartas</p>
  1115→
  1116→      <div className="mt-8 grid grid-cols-3 gap-4">
  1117→        <WabiSabiCard hover={false}>
  1118→          <p className="font-serif-jp text-xl font-bold text-[var(--ws-accent)]">{avgRating}</p>
  1119→          <p className="mt-1 text-[10px] text-[var(--ws-text-tertiary)]">Media</p>
  1120→        </WabiSabiCard>
  1121→        <WabiSabiCard hover={false}>
  1122→          <p className="font-serif-jp text-xl font-bold text-[var(--ws-gold)]">{hardCount}</p>
  1123→          <p className="mt-1 text-[10px] text-[var(--ws-text-tertiary)]">Dificeis</p>
  1124→        </WabiSabiCard>
  1125→        <WabiSabiCard hover={false}>
  1126→          <p className="font-serif-jp text-xl font-bold text-[var(--ws-verdigris)]">{easyCount}</p>

  1204→function PomodoroTimer() {
  1205→  const [mode, setMode] = useState<'focus' | 'short' | 'long'>('focus');
  1206→  const [timeLeft, setTimeLeft] = useState(25 * 60);
  1207→  const [isRunning, setIsRunning] = useState(false);
  1208→  const [totalToday, setTotalToday] = useState(0);
  1209→  const [sessions, setSessions] = useState<{ duration: number; createdAt: string }[]>([]);
  1210→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
  1211→  const durations = { focus: 25 * 60, short: 5 * 60, long: 15 * 60 };
  1212→  const modeLabels = { focus: 'Foco', short: 'Pausa Curta', long: 'Pausa Longa' };
  1213→  const fetchedRef = useRef(false);
  1214→
  1215→  useEffect(() => {
  1216→    if (fetchedRef.current) return;
  1217→    fetchedRef.current = true;
  1218→    fetch('/api/sessions').then(r => r.json()).then(d => {
  1219→      if (d.sessions) {
  1220→        const today = new Date().toDateString();
  1221→        setTotalToday(d.sessions.filter((s: any) => new Date(s.createdAt).toDateString() === today).reduce((sum: number, s: any) => sum + s.duration, 0));
  1222→        setSessions(d.sessions.slice(0, 10));
  1223→      }
  1224→    });
  1225→  }, []);
  1226→
  1227→  useEffect(() => {
  1228→    if (isRunning) {
  1229→      intervalRef.current = setInterval(() => {
  1230→        setTimeLeft(prev => {
  1231→          if (prev <= 1) {
  1232→            setIsRunning(false);
  1233→            if (mode === 'focus') {
  1234→              const dur = durations.focus - prev;
  1235→              fetch('/api/sessions', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ duration: dur, type: 'pomodoro' }) });
  1236→              setTotalToday(t => t + dur);
  1237→            }
  1238→            return 0;
  1239→          }
  1240→          return prev - 1;
  1241→        });
  1242→      }, 1000);
  1243→    } else if (intervalRef.current) clearInterval(intervalRef.current);
  1244→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
  1245→  }, [isRunning, mode, durations.focus]);
  1246→
  1247→  const switchMode = (m: 'focus' | 'short' | 'long') => { setMode(m); setTimeLeft(durations[m]); setIsRunning(false); };
  1248→  const minutes = Math.floor(timeLeft / 60);
  1249→  const seconds = timeLeft % 60;
  1250→  const progress = 1 - timeLeft / durations[mode];
  1251→
  1252→  return (
  1253→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
  1254→      <div className="mb-8">
  1255→        <h1 className="font-serif-jp text-2xl font-bold text-[var(--ws-text-primary)] lg:text-3xl"><Timer size={24} className="mr-2 inline text-[var(--ws-accent)]" strokeWidth={1.5} />Pomodoro Timer</h1>
  1256→        <p className="mt-1 text-sm text-[var(--ws-text-tertiary)]">Estude com foco usando a tecnica Pomodoro</p>
  1257→      </div>
  1258→      <div className="mx-auto max-w-md">
  1259→        <div className="mb-8 flex justify-center gap-2">
  1260→          {(['focus', 'short', 'long'] as const).map(m => (
  1261→            <button key={m} onClick={() => switchMode(m)} className={`rounded-ws-button px-4 py-2 text-sm font-medium transition-colors ${mode === m ? 'bg-[var(--ws-accent)] text-[var(--ws-text-on-dark)]' : 'border border-[var(--ws-glass-border)] text-[var(--ws-text-tertiary)] hover:text-[var(--ws-text-secondary)]'}`}>
  1262→              {modeLabels[m]}
  1263→            </button>
  1264→          ))}
  1265→        </div>
  1266→        <WabiSabiCard className="mb-8" hover={false}>
  1267→          <div className="flex flex-col items-center py-8">
  1268→            <div className="relative flex h-56 w-56 items-center justify-center">
  1269→              <svg className="absolute inset-0 -rotate-90" viewBox="0 0 200 200">
  1270→                <circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-glass-border)" strokeWidth="4" />
  1271→                <motion.circle cx="100" cy="100" r="90" fill="none" stroke="var(--ws-accent)" strokeWidth="4" strokeLinecap="round" strokeDasharray={2 * Math.PI * 90} initial={{ strokeDashoffset: 2 * Math.PI * 90 }} animate={{ strokeDashoffset: 2 * Math.PI * 90 * (1 - progress) }} transition={{ duration: 0.5 }} />
  1272→              </svg>
  1273→              <div className="text-center">
  1274→                <p className="font-serif-jp text-5xl font-bold text-[var(--ws-text-primary)]">{String(minutes).padStart(2, '0')}:{String(seconds).padStart(2, '0')}</p>
  1275→                <p className="mt-1 text-xs text-[var(--ws-text-tertiary)]">{modeLabels[mode]}</p>

  1277→            </div>
  1278→            <div className="mt-8 flex gap-4">
  1279→              <button onClick={() => switchMode(mode)} className="rounded-ws-button p-3 text-[var(--ws-text-tertiary)] transition-colors hover:bg-[color-mix(in_srgb,var(--ws-ink)_5%,transparent)] hover:text-[var(--ws-text-primary)]" title="Reiniciar"><RotateCcw size={18} /></button>

  1282→          </div>
  1283→        </WabiSabiCard>
  1284→        <WabiSabiCard hover={false}>
  1285→          <h3 className="mb-3 text-sm font-semibold text-[var(--ws-text-secondary)]"><Clock size={14} className="mr-1.5 inline" /> Hoje</h3>
