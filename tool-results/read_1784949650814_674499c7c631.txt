   116→   116→// ========== HELPER: Time-based greeting ==========
   117→   117→function getGreeting(): string {
   118→   118→  const h = new Date().getHours();
   119→   119→  if (h < 12) return 'Bom dia';
   120→   120→  if (h < 18) return 'Boa tarde';
   121→   121→  return 'Boa noite';
   122→   122→}
   123→   123→
   124→   124→function getGreetingSubtext(): string {
   125→   125→  const h = new Date().getHours();
   126→   126→  if (h < 12) return 'Comece o dia com foco e determinacao. Cada estudo e um passo adiante.';
   127→   127→  if (h < 18) return 'A tarde e perfeita para revisar e consolidar o conhecimento.';
   128→   128→  return 'Boas sessoes noturnas fixam melhor o que voce aprendeu.';
   129→   129→}
   130→   130→
   131→   131→function formatReviewDate(dateStr: string): string {
   132→   132→  const d = new Date(dateStr);
   133→   133→  const now = new Date();
   134→   134→  const diffMs = d.getTime() - now.getTime();
   135→   135→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   136→   136→
   137→   137→  if (diffDays < 0) return 'Atrasado';
   138→   138→  if (diffDays === 0) return 'Hoje';
   139→   139→  if (diffDays === 1) return 'Amanha';
   140→   140→  if (diffDays <= 7) return `em ${diffDays} dias`;
   141→   141→  return d.toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit' });
   142→   142→}
   143→   143→
   144→   144→function getStatusBadge(repetitions: number): { label: string; color: string; bg: string } {
   145→   145→  if (repetitions === 0) return { label: 'Novo', color: 'var(--ws-accent)', bg: 'color-mix(in srgb, var(--ws-accent) 12%, transparent)' };
   146→   146→  if (repetitions < 5) return { label: 'Revisando', color: 'var(--ws-gold)', bg: 'color-mix(in srgb, var(--ws-gold) 12%, transparent)' };
   147→   147→  return { label: 'Dominado', color: 'var(--ws-verdigris)', bg: 'color-mix(in srgb, var(--ws-verdigris) 12%, transparent)' };
   148→   148→}
   149→   149→
   150→   150→function formatTimestamp(iso: string): string {
   151→   151→  const d = new Date(iso);
   152→   152→  return d.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' });
   153→   153→}
