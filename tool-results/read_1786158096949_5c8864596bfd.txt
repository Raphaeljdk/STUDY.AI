   100→function ShapeIcon({ shape }: { shape: string }) {
   101→  const s = 16;
   102→  switch (shape) {
   103→    case 'line': return <Minus size={s} />;

   407→
   408→export function FloatingPenToolbar({
   409→  activeTool,
   410→  onToolChange,
   411→  strokeColor,
   412→  onStrokeColorChange,
   413→  strokeWidth,
   414→  onStrokeWidthChange,
   415→  onUndo,
   416→  onRedo,
   417→  onImageImport,
   418→  smoothing,
   419→  onSmoothingChange,
   420→  stabilizer,
   421→  onStabilizerChange,
   422→  eraserMode = 'stroke',
   423→  onEraserModeChange,
   424→  eraserSize = 20,
   425→  onEraserSizeChange,
   426→  highlighterColor = '#facc15',
   427→  onHighlighterColorChange,
   428→  highlighterOpacity = 40,
   429→  onHighlighterOpacityChange,
   430→}: FloatingPenToolbarProps) {
   431→  const [isOpen, setIsOpen] = useState(false);
   432→  const [subPanel, setSubPanel] = useState<SubPanel>(null);
   433→  const containerRef = useRef<HTMLDivElement>(null);
   434→
   435→  /* Determine which sub-panel to show based on active tool */
   436→  useEffect(() => {
   437→    if (!isOpen) {
   438→      setSubPanel(null);
   439→      return;
   440→    }
   441→    const shapeTools = ['line', 'arrow', 'rectangle', 'circle', 'triangle'];
   442→    if (shapeTools.includes(activeTool)) {
   443→      setSubPanel('shapes');
   444→    } else if (activeTool === 'pen') {
   445→      setSubPanel('pen');
   446→    } else if (activeTool === 'highlighter') {
   447→      setSubPanel('highlighter');
   448→    } else if (activeTool === 'eraser') {
   449→      setSubPanel('eraser');
   450→    } else if (activeTool === 'text') {
   451→      setSubPanel('text');
   452→    } else {
   453→      setSubPanel(null);
   454→    }
   455→  }, [activeTool, isOpen]);
   456→
   457→  /* Keyboard shortcuts */
   458→  const handleKeyDown = useCallback(
   459→    (e: KeyboardEvent) => {
   460→      const target = e.target as HTMLElement;
   461→      const isInput = target.tagName === 'INPUT' || target.tagName === 'TEXTAREA' || target.isContentEditable;
   462→      if (isInput) return;
   463→
   464→      if (e.key === 'p' || e.key === 'P') {
   465→        e.preventDefault();
   466→        setIsOpen((prev) => !prev);
   467→ }
   468→      if (e.key === 'e' || e.key === 'E') {
   469→        e.preventDefault();
   470→        onToolChange('eraser');
   471→        if (!isOpen) setIsOpen(true);
   472→      }
   473→      if (e.key === 't' || e.key === 'T') {
   474→        e.preventDefault();
   475→        onToolChange('text');
   476→        if (!isOpen) setIsOpen(true);
   477→      }
   478→      if (e.key === 'v' || e.key === 'V') {
   479→        e.preventDefault();
   480→        onToolChange('select');
   481→      }
   482→      if (e.key === 'Escape') {
   483→        setIsOpen(false);
   484→      }
   485→    },
   486→    [onToolChange, isOpen],
   487→  );
   488→
   489→  useEffect(() => {
   490→    window.addEventListener('keydown', handleKeyDown);
   491→    return () => window.removeEventListener('keydown', handleKeyDown);
   492→  }, [handleKeyDown]);
   493→
   494→  /* Click outside to close */
   495→  useEffect(() => {
   496→    if (!isOpen) return;
   497→    const handleClick = (e: MouseEvent) => {
   498→      if (containerRef.current && !containerRef.current.contains(e.target as Node)) {
   499→        setIsOpen(false);
   500→      }
   501→    };
   502→    const timer = setTimeout(() => {
   503→      document.addEventListener('mousedown', handleClick);
   504→    }, 0);
   505→    return () => {
   506→      clearTimeout(timer);
   507→      document.removeEventListener('mousedown', handleClick);
   508→    };
   509→  }, [isOpen]);
   510→
   511→  /* Handle tool click */
   512→  const handleToolClick = useCallback(
   513→    (toolId: string) => {
   514→      onToolChange(toolId);
   515→    },
   516→    [onToolChange],
   517→  );
   518→
   519→  /* Handle preset */
   520→  const handlePreset = useCallback(
   521→    (preset: (typeof PRESETS)[number]) => {
   522→      onToolChange(preset.tool);
   523→      onStrokeColorChange(preset.color);
   524→      onStrokeWidthChange(preset.width);
   525→      onSmoothingChange(preset.smoothing);
   526→    },
   527→    [onToolChange, onStrokeColorChange, onStrokeWidthChange, onSmoothingChange],
   528→  );
   529→
   530→  /* Render sub-panel */
   531→  const renderSubPanel = () => {
   532→    switch (subPanel) {
   533→      case 'pen':
   534→        return (
   535→          <PenSubPanel
   536→            strokeColor={strokeColor}
   537→            onStrokeColorChange={onStrokeColorChange}
   538→            strokeWidth={strokeWidth}
   539→            onStrokeWidthChange={onStrokeWidthChange}
   540→            smoothing={smoothing}
   541→            onSmoothingChange={onSmoothingChange}
   542→            stabilizer={stabilizer}
   543→            onStabilizerChange={onStabilizerChange}
   544→          />
   545→        );
   546→      case 'highlighter':
   547→        return (
   548→          <HighlighterSubPanel
   549→            highlighterColor={highlighterColor}
   550→            onHighlighterColorChange={onHighlighterColorChange}
   551→            highlighterOpacity={highlighterOpacity}
   552→            onHighlighterOpacityChange={onHighlighterOpacityChange}
   553→          />
   554→        );
   555→      case 'eraser':
   556→        return (
   557→          <EraserSubPanel
   558→            eraserMode={eraserMode}
   559→            onEraserModeChange={onEraserModeChange}
   560→            eraserSize={eraserSize}
   561→            onEraserSizeChange={onEraserSizeChange}
   562→          />
   563→        );
   564→      case 'text':
   565→        return <TextSubPanel />;
   566→      case 'shapes':
   567→        return <ShapesSubPanel activeTool={activeTool} onToolChange={onToolChange} />;
   568→      default:
   569→        return null;
   570→    }
   571→  };
   572→
   573→  return (
   574→    <div
   575→      ref={containerRef}
   576→      className="fixed bottom-5 right-5 z-[100] flex flex-col items-end"
   577→    >
   578→      {/* Expanded toolbar panel */}
   579→      <AnimatePresence>
   580→        {isOpen && (
   581→          <motion.div
   582→            initial={{ opacity: 0, y: 20, scale: 0.95 }}
   583→            animate={{ opacity: 1, y: 0, scale: 1 }}
   584→            exit={{ opacity: 0, y: 20, scale: 0.95 }}
   585→            transition={{ duration: 0.2, ease: [0.4, 0, 0.2, 1] }}
   586→            className="glass-enhanced mb-3 flex flex-col overflow-hidden rounded-ws-card"
   587→            style={{
   588→              maxWidth: 260,
   589→              boxShadow: 'var(--ws-shadow-medium)',
   590→            }}
   591→          >
   592→            <div className="flex">
   593→              {/* Tool list column */}
   594→              <div className="flex flex-col gap-0.5 p-1.5">
   595→                {/* Drawing tools */}
   596→                <ToolButton
   597→                  icon={<PenTool size={18} />}
   598→                  label="Caneta"
   599→                  active={activeTool === 'pen'}
   600→                  onClick={() => handleToolClick('pen')}
   601→                />
   602→                <ToolButton
   603→                  icon={<Highlighter size={18} />}
   604→                  label="Marca-texto"
   605→                  active={activeTool === 'highlighter'}
   606→                  onClick={() => handleToolClick('highlighter')}
   607→                />
   608→                <ToolButton
   609→                  icon={<Eraser size={18} />}
   610→                  label="Borracha"
   611→                  active={activeTool === 'eraser'}
   612→                  onClick={() => handleToolClick('eraser')}
   613→                />
   614→                <ToolButton
   615→                  icon={<Type size={18} />}
   616→                  label="Texto"
   617→                  active={activeTool === 'text'}
   618→                  onClick={() => handleToolClick('text')}
   619→                />
   620→                <ToolButton
   621→                  icon={<MousePointer2 size={18} />}
   622→                  label="Selecionar"
   623→                  active={activeTool === 'select'}
   624→                  onClick={() => handleToolClick('select')}
   625→                />
   626→                <ToolButton
   627→                  icon={<Pentagon size={18} />}
   628→                  label="Formas"
   629→                  active={['line', 'arrow', 'rectangle', 'circle', 'triangle'].includes(activeTool)}
   630→                  onClick={() => handleToolClick('rectangle')}
   631→                />
   632→
   633→                {/* Separator */}
   634→                <div className="my-1 h-px w-full" style={{ backgroundColor: 'var(--ws-glass-border)' }} />
   635→
   636→                {/* Actions */}
   637→                <ToolButton
   638→                  icon={<Undo2 size={18} />}
   639→                  label="Desfazer"
   640→                  active={false}
   641→                  onClick={onUndo}
   642→                />
   643→                <ToolButton
   644→                  icon={<Redo2 size={18} />}
   645→                  label="Refazer"
   646→                  active={false}
   647→                  onClick={onRedo}
   648→                />
   649→
   650→                <div className="my-1 h-px w-full" style={{ backgroundColor: 'var(--ws-glass-border)' }} />
   651→
   652→                <ToolButton
   653→                  icon={<ImagePlus size={18} />}
   654→                  label="Imagem"
   655→                  active={false}
   656→                  onClick={onImageImport}
   657→                />
   658→
   659→                <div className="my-1 h-px w-full" style={{ backgroundColor: 'var(--ws-glass-border)' }} />
   660→
   661→                <ToolButton
   662→                  icon={<Settings size={18} />}
   663→                  label="Config"
   664→                  active={false}
   665→                  onClick={() => {}}
   666→                />
   667→              </div>
   668→
   669→              {/* Separator line */}
   670→              <div
   671→                className="w-px self-stretch"
   672→                style={{ backgroundColor: 'var(--ws-glass-border)' }}
   673→              />
   674→
   675→              {/* Sub-panel content */}
   676→              <div
   677→                className="flex-1 overflow-y-auto"
   678→                style={{ maxHeight: 420 }}
   679→              >
   680→                {subPanel ? (
   681→                  <motion.div
   682→                    key={subPanel}
   683→                    initial={{ opacity: 0, x: 8 }}
   684→                    animate={{ opacity: 1, x: 0 }}
   685→                    exit={{ opacity: 0, x: 8 }}
   686→                    transition={{ duration: 0.15 }}
   687→                  >
   688→                    {renderSubPanel()}
   689→                  </motion.div>
   690→                ) : (
   691→                  <div className="flex items-center justify-center p-4" style={{ minWidth: 180, minHeight: 60 }}>
   692→                    <p className="text-center text-xs" style={{ color: 'var(--ws-text-tertiary)' }}>
   693→                      Selecione uma ferramenta
   694→                    </p>
   695→                  </div>
   696→                )}
   697→              </div>
   698→            </div>
   699→
   700→            {/* Divider */}
   701→            <div className="mx-3 h-px" style={{ backgroundColor: 'var(--ws-glass-border)' }} />
   702→
   703→            {/* Presets section */}
   704→            <div className="flex gap-1.5 p-2">
   705→              {PRESETS.map((preset) => (
   706→                <button
   707→                  key={preset.id}
   708→                  onClick={() => handlePreset(preset)}
   709→                  className="flex flex-1 flex-col items-center gap-1 rounded-ws-button px-2 py-1.5 text-center transition-colors hover:opacity-80"
   710→                  title={preset.label}
   711→                >
   712→                  <span className="text-base leading-none">{preset.emoji}</span>
   713→                  <span className="text-[10px] leading-tight" style={{ color: 'var(--ws-text-secondary)' }}>
   714→                    {preset.label}
   715→                  </span>
   716→                </button>
   717→              ))}
   718→            </div>
   719→          </motion.div>
   720→        )}
   721→      </AnimatePresence>
   722→
   723→      {/* Floating toggle button */}
   724→      <motion.button
   725→        whileHover={{ scale: 1.05 }}
   726→        whileTap={{ scale: 0.95 }}
   727→        onClick={() => setIsOpen((prev) => !prev)}
   728→        className={isOpen
   729→          ? 'glass-enhanced flex h-12 w-12 items-center justify-center rounded-full transition-all'
   730→          : 'glass-enhanced animate-pulse-glow flex h-12 w-12 items-center justify-center rounded-full transition-all'
   731→        }
   732→        style={{
   733→          boxShadow: 'var(--ws-shadow-medium)',
   734→          color: isOpen ? 'var(--ws-accent)' : 'var(--ws-text-primary)',
   735→        }}
   736→        aria-label={isOpen ? 'Fechar barra de ferramentas' : 'Abrir barra de ferramentas'}
   737→        title="Tecla P para alternar"
   738→      >
   739→        <AnimatePresence mode="wait" initial={false}>
   740→          <motion.span
   741→            key={isOpen ? 'close' : 'open'}
   742→            initial={{ rotate: -90, opacity: 0 }}
   743→            animate={{ rotate: 0, opacity: 1 }}
   744→            exit={{ rotate: 90, opacity: 0 }}
   745→            transition={{ duration: 0.15 }}
   746→            className="flex items-center justify-center"
   747→          >
   748→            {isOpen ? <X size={20} /> : <PenTool size={20} />}
   749→          </motion.span>
   750→        </AnimatePresence>
   751→      </motion.button>
   752→    </div>
   753→  );
   754→}
