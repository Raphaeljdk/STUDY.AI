   171→// ===== MAIN COMPONENT =====
   172→export function DiscoverView({ onNavigate }: DiscoverViewProps) {
   173→  // State
   174→  const [items, setItems] = useState<DiscoverItem[]>([]);
   175→  const [loading, setLoading] = useState(true);
   176→  const [activeFilter, setActiveFilter] = useState('todos');
   177→  const [searchQuery, setSearchQuery] = useState('');
   178→  const [expandedId, setExpandedId] = useState<string | null>(null);
   179→  const [savingId, setSavingId] = useState<string | null>(null);
   180→  const [page, setPage] = useState(1);
   181→  const [totalPages, setTotalPages] = useState(1);
   182→  const [subjects, setSubjects] = useState<Subject[]>([]);
   183→
   184→  // AI Generate dialog state
   185→  const [showAIDialog, setShowAIDialog] = useState(false);
   186→  const [aiType, setAiType] = useState('dica');
   187→  const [aiSubject, setAiSubject] = useState('');
   188→  const [aiDifficulty, setAiDifficulty] = useState('medio');
   189→  const [aiGenerating, setAiGenerating] = useState(false);
   190→
   191→  // Subject selector state
   192→  const [showSubjectDialog, setShowSubjectDialog] = useState(false);
   193→  const [subjectSelectItemId, setSubjectSelectItemId] = useState<string | null>(null);
   194→  const [selectedSubjectId, setSelectedSubjectId] = useState<string | '');
   195→
   196→  const filterRef = useRef<HTMLDivElement>(null);
   197→
   198→  // ===== DATA FETCHING =====
   199→  const fetchItems = useCallback(async (pageNum: number, type?: string) => {
   200→    try {
   201→      const params = new URLSearchParams({ page: String(pageNum), limit: '20' });
   202→      if (type && type !== 'todos') params.set('type', type);
   203→      const res = await fetch(`/api/discover?${params.toString()}`);
   204→      if (!res.ok) throw new Error('Failed to fetch');
   205→      const data = await res.json();
   206→      if (pageNum === 1) {
   207→        setItems(data.items || []);
   208→      } else {
   209→        setItems(prev => [...prev, ...(data.items || [])]);
   210→      }
   211→      setTotalPages(data.pagination?.totalPages || 1);
   212→    } catch {
   213→      toast({ title: 'Erro ao carregar conteúdo', description: 'Tente novamente mais tarde.' });
   214→    }
   215→  }, []);
   216→
   217→  const fetchSubjects = useCallback(async () => {
   218→    try {
   219→      const res = await fetch('/api/subjects');
   220→      if (!res.ok) return;
   221→      const data = await res.json();
   222→      setSubjects(Array.isArray(data) ? data : data.subjects || []);
   223→    } catch {
   224→      // Silent fail
   225→    }
   226→  }, []);
   227→
   228→  useEffect(() => {
   229→    const load = async () => {
   230→      setLoading(true);
   231→      setPage(1);
   232→      await fetchItems(1, activeFilter);
   233→      setLoading(false);
   234→    };
   235→    load();
   236→  }, [activeFilter, fetchItems]);
   237→
   238→  useEffect(() => {
   239→    fetchSubjects();
   240→  }, [fetchSubjects]);
   241→
   242→  // Infinite scroll
   243→  const loadMore = useCallback(async () => {
   244→    if (loading || page >= totalPages) return;
   245→    const nextPage = page + 1;
   246→    setPage(nextPage);
   247→    await fetchItems(nextPage, activeFilter);
   248→  }, [loading, page, totalPages, activeFilter, fetchItems]);
   249→
   250→  // ===== FILTERED ITEMS =====
   251→  const filteredItems = useMemo(() => {
   252→    if (!searchQuery.trim()) return items;
   253→    const q = searchQuery.toLowerCase();
   254→    return items.filter(
   255→      item =>
   256→        item.title.toLowerCase().includes(q) ||
   257→        item.content.toLowerCase().includes(q) ||
   258→        item.summary?.toLowerCase().includes(q) ||
   259→        item.subject?.toLowerCase().includes(q) ||
   260→        item.tags?.toLowerCase().includes(q)
   261→    );
   262→  }, [items, searchQuery]);
   263→
   264→  // ===== SAVE/UNSAVE =====
   265→  const handleSave = async (e: React.MouseEvent, itemId: string) => {
   266→    e.stopPropagation();
   267→    setSavingId(itemId);
   268→    try {
   269→      const res = await fetch(`/api/discover/${itemId}/save`, { method: 'POST' });
   270→      if (!res.ok) throw new Error('Failed');
   271→      const data = await res.json();
   272→      setItems(prev =>
   273→        prev.map(item =>
   274→          item.id === itemId ? { ...item, isSaved: data.isSaved, saves: data.isSaved ? item.saves + 1 : Math.max(0, item.saves - 1) } : item
   275→        )
   276→      );
   277→      toast({
   278→        title: data.isSaved ? 'Salvo!' : 'Removido dos salvos',
   279→        description: data.isSaved ? 'Conteúdo adicionado aos seus salvos.' : 'Conteúdo removido dos salvos.',
   280→      });
   281→    } catch {
   282→      toast({ title: 'Erro', description: 'Não foi possível salvar o conteúdo.' });
   283→    } finally {
   284→      setSavingId(null);
   285→    }
   286→  };
   287→
   288→  // ===== AI GENERATE =====
   289→  const handleAIGenerate = async () => {
   290→    if (!aiSubject.trim()) {
   291→      toast({ title: 'Informe o assunto', description: 'Digite um assunto para gerar conteúdo.' });
   292→      return;
   293→    }
   294→    setAiGenerating(true);
   295→    try {
   296→      const res = await fetch('/api/discover', {
   297→        method: 'POST',
   298→        headers: { 'Content-Type': 'application/json' },
   299→        body: JSON.stringify({
   300→          type: aiType,
   301→          subject: aiSubject.trim(),
   302→          difficulty: aiDifficulty,
   303→          generateWithAI: true,
   304→        }),
   305→      });
   306→      if (!res.ok) {
   307→        const err = await res.json().catch(() => ({}));
   308→        throw new Error(err.error || 'Failed to generate');
   309→      }
   310→      const data = await res.json();
   311→      if (data.item) {
   312→        setItems(prev => [data.item, ...prev]);
   313→        toast({ title: 'Conteúdo gerado! ✨', description: data.item.title });
   314→      }
   315→      setShowAIDialog(false);
   316→      setAiSubject('');
   317→    } catch (err: any) {
   318→      toast({ title: 'Erro ao gerar', description: err.message || 'Tente novamente.' });
   319→    } finally {
   320→      setAiGenerating(false);
   321→    }
   322→  };
   323→
   324→  // ===== ADD TO SUBJECT =====
   325→  const openSubjectSelector = (e: React.MouseEvent, itemId: string) => {
   326→    e.stopPropagation();
   327→    setSubjectSelectItemId(itemId);
   328→    setSelectedSubjectId('');
   329→    setShowSubjectDialog(true);
   330→  };
   331→
   332→  const handleAddToSubject = async () => {
   333→    if (!selectedSubjectId || !subjectSelectItemId) return;
   334→    try {
   335→      const item = items.find(i => i.id === subjectSelectItemId);
   336→      if (!item) return;
   337→      // Create a note in the subject with the discover content
   338→      const res = await fetch('/api/notes', {
   339→        method: 'POST',
   340→        headers: { 'Content-Type': 'application/json' },
   341→        body: JSON.stringify({
   342→          title: `[Discover] ${item.title}`,
   343→          content: item.content,
   344→          subjectId: selectedSubjectId,
   345→        }),
   346→      });
   347→      if (res.ok) {
   348→        toast({ title: 'Adicionado à matéria! 📚', description: 'Conteúdo salvo na matéria selecionada.' });
   349→        setShowSubjectDialog(false);
   350→      } else {
   351→        throw new Error('Failed');
   352→      }
   353→    } catch {
   354→      toast({ title: 'Erro', description: 'Não foi possível adicionar à matéria.' });
   355→    }
   356→  };
   357→
   358→  // ===== ACTION BUTTONS =====
   359→  const handleGenerateFlashcards = (e: React.MouseEvent, item: DiscoverItem) => {
   360→    e.stopPropagation();
   361→    toast({ title: 'Flashcards em breve!', description: 'Funcionalidade de gerar flashcards será liberada em breve.' });
   362→  };
   363→
   364→  const handleGenerateQuestions = (e: React.MouseEvent, item: DiscoverItem) => {
   365→    e.stopPropagation();
   366→    toast({ title: 'Questões em breve!', description: 'Funcionalidade de gerar questões será liberada em breve.' });
   367→  };
   368→
   369→  // ===== HELPERS =====
   370→  const getTypeConfig = (type: string) => CARD_TYPES[type] || CARD_TYPES.dica;
   371→  const getDifficultyConfig = (d: string) => DIFFICULTY_CONFIG[d] || DIFFICULTY_CONFIG.medio;
   372→  const formatDuration = (seconds: number | null) => {
   373→    if (!seconds) return null;
   374→    if (seconds < 60) return `${seconds}s`;
   375→    const m = Math.floor(seconds / 60);
   376→    const s = seconds % 60;
   377→    return s > 0 ? `${m}m ${s}s` : `${m}min`;
   378→  };
   379→
   380→  // ===== RENDER =====
   381→  return (
   382→    <div className="relative min-h-full pb-24">
   383→      {/* Header */}
   384→      <div className="mb-6">
   385→        <h1
   386→          className="font-serif-jp text-2xl font-bold mb-1"
   387→          style={{ color: 'var(--ws-text-primary)' }}
   388→        >
   389→          Discover
   390→        </h1>
   391→        <p className="text-sm" style={{ color: 'var(--ws-text-tertiary)' }}>
   392→          Explore conteúdo educacional gerado pela comunidade e IA
   393→        </p>
   394→      </div>
   395→
   396→      {/* Search Bar */}
   397→      <div className="relative mb-4">
   398→        <Search
   399→          className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2"
   400→          style={{ color: 'var(--ws-text-tertiary)' }}
   401→        />
   402→        <Input
   403→          placeholder="Buscar conteúdo..."
   404→          value={searchQuery}
   405→          onChange={e => setSearchQuery(e.target.value)}
   406→          className="pl-10 pr-4 h-11 w-full rounded-ws-button border-[var(--ws-glass-border)] bg-[var(--ws-glass)] backdrop-blur-xl text-[var(--ws-text-primary)] placeholder:text-[var(--ws-text-tertiary)] focus-visible:ring-[var(--ws-accent)]/30 focus-visible:border-[var(--ws-accent)]/40"
   407→          style={{ fontSize: '14px' }}
   408→        />
   409→        {searchQuery && (
   410→          <button
   411→            onClick={() => setSearchQuery('')}
   412→            className="absolute right-3 top-1/2 -translate-y-1/2 rounded-full p-0.5 transition-ws"
   413→            style={{ color: 'var(--ws-text-tertiary)' }}
   414→            aria-label="Limpar busca"
   415→          >
   416→            <X className="h-4 w-4" />
   417→          </button>
   418→        )}
   419→      </div>
   420→
   421→      {/* Filter Bar */}
   422→      <div
   423→        ref={filterRef}
   424→        className="mb-6 -mx-4 px-4 overflow-x-auto no-scrollbar"
   425→      >
   426→        <div className="flex gap-2 pb-1" style={{ minWidth: 'max-content' }}>
   427→          {FILTER_OPTIONS.map(opt => {
   428→            const isActive = activeFilter === opt.key;
   429→            const typeConf = opt.key !== 'todos' ? CARD_TYPES[opt.key] : null;
   430→            return (
   431→              <button
   432→                key={opt.key}
   433→                onClick={() => {
   434→                  setActiveFilter(opt.key);
   435→                  setPage(1);
   436→                }}
   437→                className="flex items-center gap-1.5 whitespace-nowrap rounded-ws-button px-3.5 py-2 text-xs font-medium transition-ws"
   438→                style={{
   439→                  background: isActive
   440→                    ? typeConf
   441→                      ? typeConf.accentBg
   442→                      : 'color-mix(in srgb, var(--ws-accent) 10%, transparent)'
   443→                    : 'var(--ws-glass)',
   444→                  color: isActive
   445→                    ? typeConf
   446→                      ? typeConf.accent
   447→                      : 'var(--ws-accent)'
   448→                    : 'var(--ws-text-secondary)',
   449→                  border: `1px solid ${isActive
   450→                    ? typeConf
   451→                      ? typeConf.accentBorder
   452→                      : 'color-mix(in srgb, var(--ws-accent) 25%, transparent)'
   453→                    : 'var(--ws-glass-border)'}`,
   454→                  boxShadow: isActive ? 'var(--ws-shadow-soft)' : 'none',
   455→                }}
   456→                aria-pressed={isActive}
   457→              >
   458→                <span>{opt.emoji}</span>
   459→                <span>{opt.label}</span>
   460→              </button>
   461→            );
   462→          })}
   463→        </div>
   464→      </div>
   465→
   466→      {/* Content Feed */}
   467→      {loading && items.length === 0 ? (
   468→        <div className="space-y-4">
   469→          {Array.from({ length: 4 }).map((_, i) => (
   470→            <SkeletonCard key={i} />
   471→          ))}
   472→        </div>
   473→      ) : filteredItems.length === 0 ? (
   474→        <EmptyState hasFilter={activeFilter !== 'todos' || !!searchQuery.trim()} />
   475→      ) : (
   476→        <div className="space-y-4">
   477→          <AnimatePresence mode="popLayout">
   478→            {filteredItems.map((item, index) => {
   479→              const typeConf = getTypeConfig(item.type);
   480→              const diffConf = getDifficultyConfig(item.difficulty);
   481→              const isExpanded = expandedId === item.id;
   482→              const duration = formatDuration(item.duration);
   483→
   484→              return (
   485→                <motion.div
   486→                  key={item.id}
   487→                  layout
   488→                  initial={{ opacity: 0, y: 16 }}
   489→                  animate={{ opacity: 1, y: 0 }}
   490→                  exit={{ opacity: 0, y: -10 }}
   491→                  transition={{ duration: 0.35, delay: Math.min(index * 0.04, 0.2), ease: [0.4, 0, 0.2, 1] }}
   492→                  className="hover-lift cursor-pointer"
   493→                  style={{ borderRadius: 'var(--ws-radius-card)' }}
   494→                  onClick={() => setExpandedId(isExpanded ? null : item.id)}
   495→                >
   496→                  <div
   497→                    className="border backdrop-blur-xl p-5 transition-ws"
   498→                    style={{
   499→                      borderRadius: 'var(--ws-radius-card)',
