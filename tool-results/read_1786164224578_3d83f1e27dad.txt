    29→    29→
    30→    30→class SmoothBrush extends PencilBrush {
    31→    31→  private buffer: Array<{ x: number; y: number }> = [];
    32→    32→  private bufferSize = 4;
    33→    33→
    34→    34→  override onPointerDown(pointer: any) {
    35→    35→    this.buffer = [];
    36→    36→    super.onPointerDown(pointer);
    37→    37→  }
    38→    38→
    39→    39→  override onPointerMove(pointer: any) {
    40→    40→    this.buffer.push({ x: pointer.x, y: pointer.y });
    41→    41→    if (this.buffer.length > this.bufferSize) {
    42→    42→      this.buffer.shift();
    43→    43→    }
    44→    44→    if (this.buffer.length > 1) {
    45→    45→      const avg = this.buffer.reduce(
    46→    46→        (acc, p) => ({ x: acc.x + p.x / this.buffer.length, y: acc.y + p.y / this.buffer.length }),
    47→    47→        { x: 0, y: 0 },
    48→    48→      );
    49→    49→      super.onPointerMove(avg);
    50→    50→    } else {
    51→    51→      super.onPointerMove(pointer);
    52→    52→    }
    53→    53→  }
    54→    54→}
