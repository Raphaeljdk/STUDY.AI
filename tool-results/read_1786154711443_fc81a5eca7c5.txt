    77→    77→function getLevelName(level: number): string {
    78→    78→  if (level <= 1) return 'Iniciante';
    79→    79→  if (level <= 2) return 'Aprendiz';
    80→    80→  if (level <= 3) return 'Estudante';
    81→    81→  if (level <= 4) return 'Dedicado';

    83→    83→  return 'Mestre';

    86→    86→function getXpForLevel(level: number): number {
    87→    87→  // XP thresholds: 0->100->250->500->850->1300...
    88→    88→  const thresholds = [0, 100, 250, 500, 850, 1300];
    89→    89→  if (level <= 0) return 0;
    90→    90→  if (level >= thresholds.length) {
    91→    91→    // Beyond defined levels: 450 per level
    92→    92→    const lastDefined = thresholds[thresholds.length - 1];
    93→    93→    return lastDefined + (level - (thresholds.length - 1)) * 450;
    94→    94→  }
    95→    95→  return thresholds[level];
    96→    96→}
